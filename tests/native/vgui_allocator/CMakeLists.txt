cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseVguiAllocatorFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
option(CSE_VGUI_ALLOCATOR_SANITIZERS "Instrument production VGUI allocation/formatting and fixtures" OFF)
foreach(fixture allocator printf)
  add_executable(vgui_${fixture}_tests ${fixture}_tests.cpp
    "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/lib-src/vgui/vgui.cpp")
endforeach()
add_library(vgui_taskbar_compile OBJECT "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/lib-src/vgui/TaskBar.cpp")
add_dependencies(vgui_allocator_tests vgui_taskbar_compile)
foreach(target vgui_allocator_tests vgui_printf_tests vgui_taskbar_compile)
  target_include_directories(${target} PRIVATE
    "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/lib-src/vgui/include")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE VGUI_EXPORTS
    _CRT_SECURE_NO_WARNINGS "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
  set_property(TARGET ${target} PROPERTY
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /utf-8
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
  if(CSE_VGUI_ALLOCATOR_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME native-vgui-allocation-size COMMAND vgui_allocator_tests)
add_test(NAME native-vgui-printf-defined COMMAND vgui_printf_tests defined)
add_test(NAME native-vgui-printf-overlong COMMAND vgui_printf_tests overlong)
