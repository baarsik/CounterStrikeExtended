cmake_minimum_required(VERSION 3.25)
project(native_server_browser_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CSE_REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_BROWSER_SOURCE_ROOT "${CSE_REPO}" CACHE PATH "Actual source snapshot root")
set(CSE_SDL3_LIBRARY "" CACHE FILEPATH "Qualified SDL3 library for filesystem API only")
option(CSE_BROWSER_SANITIZERS "Address and undefined behavior sanitizers" OFF)
set(sdk "${CSE_BROWSER_SOURCE_ROOT}/src")
add_library(browser_model STATIC "${sdk}/GameUI/native/ServerBrowserModel.cpp")
add_library(browser_config STATIC "${sdk}/GameUI/native/ServerBrowserConfig.cpp")
add_library(browser_config_macro_contract OBJECT config_macro_contract.cpp)
target_include_directories(browser_config_macro_contract PRIVATE "${sdk}/GameUI/native"
    "${sdk}/external/SDL3/include" "${sdk}/public" "${sdk}/common")
target_include_directories(browser_model PUBLIC "${sdk}/GameUI/native")
target_include_directories(browser_config PUBLIC "${sdk}/GameUI/native" PRIVATE "${sdk}/external/SDL3/include"
    "${sdk}/public" "${sdk}/common")
target_link_libraries(browser_config PUBLIC "${CSE_SDL3_LIBRARY}")
add_executable(model_tests model_tests.cpp)
target_link_libraries(model_tests PRIVATE browser_model)
add_executable(favorite_adapter_tests favorite_adapter_tests.cpp)
target_link_libraries(favorite_adapter_tests PRIVATE browser_model)
add_executable(ping_lifetime_tests ping_lifetime_tests.cpp)
target_link_libraries(ping_lifetime_tests PRIVATE browser_model)
foreach(case pump retire inline failed-retire pending-retire)
    add_test(NAME server_browser_ping_${case} COMMAND ping_lifetime_tests ${case})
endforeach()
add_executable(config_tests config_tests.cpp)
target_link_libraries(config_tests PRIVATE browser_config)
target_include_directories(config_tests PRIVATE "${sdk}/public" "${sdk}/common")
add_executable(provider_probe provider_probe.cpp)
target_include_directories(provider_probe PRIVATE "${sdk}/public")
target_link_libraries(provider_probe PRIVATE browser_model "${CMAKE_DL_LIBS}")
foreach(branch little big)
    add_executable(address_formatter_${branch}_tests address_formatter_tests.cpp)
    target_include_directories(address_formatter_${branch}_tests PRIVATE "${sdk}/external/steamworks")
    add_test(NAME server_browser_address_formatter_${branch} COMMAND address_formatter_${branch}_tests)
endforeach()
target_compile_definitions(address_formatter_big_tests PRIVATE VALVE_BIG_ENDIAN=1)
add_executable(masterserver_tests masterserver_tests.cpp "${sdk}/GameUI/native/ServerBrowserMasterserverQuery.cpp")
target_link_libraries(masterserver_tests PRIVATE browser_config browser_model)
add_executable(masterserver_directory_tests masterserver_directory_tests.cpp)
target_link_libraries(masterserver_directory_tests PRIVATE browser_config browser_model)
add_test(NAME server_browser_master_publication COMMAND masterserver_directory_tests)
if(WIN32)
    target_link_libraries(masterserver_tests PRIVATE ws2_32)
    target_link_libraries(masterserver_directory_tests PRIVATE ws2_32)
endif()
foreach(target browser_model browser_config browser_config_macro_contract model_tests config_tests favorite_adapter_tests provider_probe masterserver_tests masterserver_directory_tests
        address_formatter_little_tests address_formatter_big_tests ping_lifetime_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(CSE_BROWSER_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    target_compile_definitions(${target} PRIVATE "$<$<CONFIG:Debug>:_DEBUG>")
endforeach()
add_test(NAME server_browser_model COMMAND model_tests)
add_test(NAME server_browser_config COMMAND config_tests "${CMAKE_CURRENT_BINARY_DIR}/config-fixture")
add_test(NAME server_browser_favorite_adapter COMMAND favorite_adapter_tests)

add_test(NAME server_browser_config_macro_contract
    COMMAND "${CMAKE_COMMAND}" --build "${CMAKE_BINARY_DIR}"
        --target browser_config_macro_contract --config "$<CONFIG>")

find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(case pagination backup cancel reentry cancel-reentry failed-empty-refresh
        completed-pump completed-cancel completed-inline failed-ping-cancel cancel-completes-other large-quick-refresh
        directory-limit-exceeded)
    add_test(NAME server_browser_master_${case} COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/masterserver_fixture.py" "$<TARGET_FILE:masterserver_tests>" ${case})
endforeach()
