cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseClientMouseCoordinates LANGUAGES CXX)
set(CSE_SDL_INCLUDE "" CACHE PATH "Pinned SDL3.4.20 include directory; no provider is linked")
set(CSE_MOUSE_WINDOW_HEADER "${CSE_ROOT}/src/engine/sdl3_window.h" CACHE FILEPATH "Actual owned header")
if(NOT EXISTS "${CSE_SDL_INCLUDE}/SDL3/SDL.h")
    message(FATAL_ERROR "Provide the pinned SDL3.4.20 include directory")
endif()
add_executable(client_mouse_coordinates_tests coordinate_tests.cpp controlled_window.cpp)
target_include_directories(client_mouse_coordinates_tests PRIVATE "${CSE_SDL_INCLUDE}"
    "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/common" "${CSE_ROOT}/src/public")
target_compile_definitions(client_mouse_coordinates_tests PRIVATE CSE_MOUSE_WINDOW_HEADER="${CSE_MOUSE_WINDOW_HEADER}")
set_target_properties(client_mouse_coordinates_tests PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
option(CSE_MOUSE_COORDINATES_SANITIZERS "Instrument only this fake-service component fixture" OFF)
if(MSVC)
    target_compile_options(client_mouse_coordinates_tests PRIVATE /W4 /WX /fp:precise /EHs-c- /GR
        $<$<CONFIG:Release>:/O2> $<$<CONFIG:Debug>:/Od> $<$<CONFIG:Debug>:/Zi>)
    target_compile_definitions(client_mouse_coordinates_tests PRIVATE NOMINMAX)
    set_property(TARGET client_mouse_coordinates_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(client_mouse_coordinates_tests PRIVATE -Wall -Wextra -Werror -ffp-model=precise
        -ffp-contract=off -Wno-overriding-option -fno-exceptions -frtti -funwind-tables
        $<$<CONFIG:Release>:-O2> $<$<CONFIG:Debug>:-O0> $<$<CONFIG:Debug>:-g>)
    if(CSE_MOUSE_COORDINATES_SANITIZERS)
        target_compile_options(client_mouse_coordinates_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(client_mouse_coordinates_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME coherent-engine-mouse-position-and-center COMMAND client_mouse_coordinates_tests)
set_tests_properties(coherent-engine-mouse-position-and-center PROPERTIES TIMEOUT 15)
