cmake_minimum_required(VERSION 3.25)
project(CsePathResolution LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(owner "${CSE_ROOT}/GoldSrc-SDK/engine/common.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${owner}" extract.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py"
  "${owner}" "${CMAKE_CURRENT_BINARY_DIR}/actual_resolution.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(path_resolution_tests resolution_tests.cpp)
target_include_directories(path_resolution_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(path_resolution_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(path_resolution_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(path_resolution_tests PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME engine-path-resolution COMMAND path_resolution_tests)
