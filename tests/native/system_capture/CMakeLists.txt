cmake_minimum_required(VERSION 3.25)
project(CseSystemCapture LANGUAGES C CXX OBJCXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CSE_SDL_INCLUDE "" CACHE PATH "Pinned SDL public headers")
set(CSE_SDL_CONFIG "" CACHE PATH "Qualified provider build_config directory")
set(CSE_CAPTURE_PROVIDER_SOURCE "${CSE_ROOT}/src/common/sdl3_native_cocoa_mouse.mm" CACHE FILEPATH
    "Complete actual owning Cocoa provider source")
option(CSE_CAPTURE_ORIGINAL_BOUNDARY "Exercise original identity-only boundary for behavioral RED" OFF)
option(CSE_CAPTURE_SANITIZERS "Instrument actual source with ASan/UBSan" OFF)
add_library(interaction_c_import OBJECT api_probe.c)
target_include_directories(interaction_c_import PRIVATE "${CSE_ROOT}/src/common")
set_target_properties(interaction_c_import PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
add_executable(interaction_tests interaction_tests.mm "${CSE_CAPTURE_PROVIDER_SOURCE}")
target_include_directories(interaction_tests PRIVATE "${CSE_ROOT}/src/common"
    "${CSE_SDL_INCLUDE}" "${CSE_SDL_CONFIG}")
target_compile_options(interaction_tests PRIVATE -Wall -Wextra -Werror -fobjc-arc -fvisibility=hidden
    -fno-exceptions -funwind-tables -ffp-model=precise -ffp-contract=off -Wno-overriding-option
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
target_link_libraries(interaction_tests PRIVATE "-framework Cocoa")
set_target_properties(interaction_tests PROPERTIES OBJCXX_STANDARD 14 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO)
if(CSE_CAPTURE_ORIGINAL_BOUNDARY)
    target_compile_definitions(interaction_tests PRIVATE CSE_ORIGINAL_OWNER_BOUNDARY=1)
    target_compile_options(interaction_tests PRIVATE -Wno-unused-function)
endif()
if(CSE_CAPTURE_SANITIZERS)
    target_compile_options(interaction_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    target_link_options(interaction_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME interaction-absolute COMMAND interaction_tests)
add_test(NAME interaction-relative COMMAND interaction_tests --relative)
