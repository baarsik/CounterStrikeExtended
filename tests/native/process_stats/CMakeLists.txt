cmake_minimum_required(VERSION 3.25)
project(CseProcessStats LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CSE_STATS_SANITIZERS "Instrument the POSIX process-stat boundary" OFF)
if(WIN32)
    message(FATAL_ERROR "The retained Windows process-stat provider is unchanged")
endif()
foreach(target process_stats_tests process_stats_real)
    if(target STREQUAL process_stats_tests)
        set(source stats_tests.cpp)
    else()
        set(source real_tests.cpp)
    endif()
    add_executable(${target} ${source})
    set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_include_directories(${target} PRIVATE "${CSE_ROOT}/src/engine")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion
        -fno-exceptions -fno-rtti)
    if(CSE_STATS_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
endforeach()
enable_testing()
add_test(NAME process-stats-controlled COMMAND process_stats_tests)
add_test(NAME process-stats-real COMMAND process_stats_real)
