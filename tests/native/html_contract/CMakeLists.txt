cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseHtmlContractFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(widget common cstrike)
  if(widget STREQUAL "cstrike")
    set(widget_source "${CSE_ROOT}/GoldSrc-SDK/cstrike/vgui2/controls/HTML.cpp")
  else()
    set(widget_source "${CSE_ROOT}/GoldSrc-SDK/vgui2/controls/HTML.cpp")
  endif()
  set(widget_output "${CMAKE_CURRENT_BINARY_DIR}/${widget}")
  add_custom_command(OUTPUT "${widget_output}/popup_input.inc"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${widget_output}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_popup_input.py"
      "${widget_source}" "${widget_output}/popup_input.inc"
    DEPENDS extract_popup_input.py "${widget_source}" VERBATIM)
  add_executable(html_popup_${widget}_tests popup_input_tests.cpp "${widget_output}/popup_input.inc")
  target_compile_features(html_popup_${widget}_tests PRIVATE cxx_std_17)
  target_include_directories(html_popup_${widget}_tests PRIVATE "${widget_output}")
  if(MSVC)
    target_compile_options(html_popup_${widget}_tests PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(html_popup_${widget}_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME html-popup-input-${widget} COMMAND html_popup_${widget}_tests)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(host_source "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/cef_host.cpp")
    add_custom_command(OUTPUT "${widget_output}/scrollbars.inc" "${widget_output}/host_geometry.inc"
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_scrollbars.py"
        "${widget_source}" "${host_source}" "${widget_output}"
      DEPENDS extract_scrollbars.py "${widget_source}" "${host_source}" VERBATIM)
    add_executable(html_scrollbar_${widget}_tests scrollbar_tests.cpp
      "${widget_output}/scrollbars.inc" "${widget_output}/host_geometry.inc")
    target_compile_features(html_scrollbar_${widget}_tests PRIVATE cxx_std_17)
    target_include_directories(html_scrollbar_${widget}_tests PRIVATE "${widget_output}"
      "${CSE_ROOT}/GoldSrc-SDK/engine")
    if(MSVC)
      target_compile_options(html_scrollbar_${widget}_tests PRIVATE /W4 /WX /EHsc)
    else()
      target_compile_options(html_scrollbar_${widget}_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    foreach(scenario fit vertical horizontal both disabled)
      add_test(NAME html-scrollbar-${widget}-${scenario} COMMAND html_scrollbar_${widget}_tests ${scenario})
    endforeach()
  endif()
endforeach()
add_executable(html_contract_tests contract_tests.cpp)
target_compile_features(html_contract_tests PRIVATE cxx_std_17)
target_include_directories(html_contract_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
if(MSVC)
  target_compile_options(html_contract_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(html_contract_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME html-controller-contract COMMAND html_contract_tests)
