cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseGPUDeviceProbe LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_GPU_SDL_LIBRARY "" CACHE FILEPATH "Pinned matching-config source-built SDL3 shared/import library")
set(CSE_GPU_SDL_RUNTIME "" CACHE FILEPATH "Pinned matching-config source-built SDL3 runtime")
if(NOT EXISTS "${CSE_GPU_SDL_LIBRARY}" OR NOT EXISTS "${CSE_GPU_SDL_RUNTIME}")
  message(FATAL_ERROR "GPU probe requires matching pinned source-built SDL library/runtime")
endif()
add_library(probe_sdl SHARED IMPORTED)
if(WIN32)
  set_target_properties(probe_sdl PROPERTIES IMPORTED_IMPLIB "${CSE_GPU_SDL_LIBRARY}"
    IMPORTED_LOCATION "${CSE_GPU_SDL_RUNTIME}")
  set(probe_driver direct3d12)
else()
  set_target_properties(probe_sdl PROPERTIES IMPORTED_LOCATION "${CSE_GPU_SDL_LIBRARY}")
  set(probe_driver metal)
endif()
add_executable(gpu_device_probe gpu_probe.cpp)
set_target_properties(gpu_device_probe PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(gpu_device_probe PRIVATE "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
target_link_libraries(gpu_device_probe PRIVATE probe_sdl)
if(MSVC)
  set_property(TARGET gpu_device_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_definitions(gpu_device_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(gpu_device_probe PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
else()
  target_compile_options(gpu_device_probe PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
add_custom_command(TARGET gpu_device_probe POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
  "${CSE_GPU_SDL_RUNTIME}" "$<TARGET_FILE_DIR:gpu_device_probe>" VERBATIM)
enable_testing()
add_test(NAME gpu-device-capabilities COMMAND gpu_device_probe ${probe_driver})
if(WIN32)
  add_test(NAME gpu-device-fewer-resource-slots COMMAND gpu_device_probe direct3d12 fewer-resource-slots)
  add_test(NAME gpu-device-dxbc-fewer-resource-slots COMMAND gpu_device_probe direct3d12 dxbc-fewer-resource-slots)
  add_test(NAME gpu-device-sampled-triangle COMMAND gpu_device_probe direct3d12 dxbc-fewer-resource-slots
    "${CMAKE_CURRENT_SOURCE_DIR}/shaders/FullscreenVert.dxbc" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/BlitFrom2D.dxbc")
else()
  add_test(NAME gpu-device-sampled-triangle COMMAND gpu_device_probe metal offline-triangle
    "${CMAKE_CURRENT_SOURCE_DIR}/shaders/FullscreenVert.metallib"
    "${CMAKE_CURRENT_SOURCE_DIR}/shaders/BlitFrom2D.metallib")
endif()
add_test(NAME gpu-device-offline-shader-inputs COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_shaders.py")
get_property(probe_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${probe_tests} PROPERTIES TIMEOUT 30)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selected-$<CONFIG>.txt" CONTENT
  "sdl_library=${CSE_GPU_SDL_LIBRARY}\nsdl_runtime=${CSE_GPU_SDL_RUNTIME}\n")
