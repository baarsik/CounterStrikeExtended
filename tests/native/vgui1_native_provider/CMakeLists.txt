cmake_minimum_required(VERSION 3.25)
project(CseVgui1NativeProvider LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_inputs.py"
    "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs" COMMAND_ERROR_IS_FATAL ANY)
set(CSE_FREETYPE_PREFIX "" CACHE PATH "Exact pinned source-built FreeType dependency prefix")
set(VCPKG_IMPORT_PREFIX "${CSE_FREETYPE_PREFIX}")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG TRUE)
list(PREPEND CMAKE_PREFIX_PATH "${CSE_FREETYPE_PREFIX}")
find_package(ZLIB CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/zlib" NO_DEFAULT_PATH)
if(NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB ALIAS ZLIB::ZLIBSTATIC)
endif()
find_library(BZIP2_LIBRARY_RELEASE NAMES bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(BZIP2_LIBRARY_DEBUG NAMES bz2d bzip2d libbz2d libbzip2d bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/debug/lib" NO_DEFAULT_PATH REQUIRED)
find_path(BZIP2_INCLUDE_DIR NAMES bzlib.h PATHS "${CSE_FREETYPE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_package(BZip2 REQUIRED)
find_package(PNG CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/png" NO_DEFAULT_PATH)
find_package(freetype CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/freetype" NO_DEFAULT_PATH)
set(vgui_source "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/lib-src/vgui")
set(native_source "${vgui_source}/posix/native")
add_library(native_face STATIC "${native_source}/VGui1NativeFontFace.cpp"
    "${native_source}/VGui1FixedSysFontFace.cpp"
    "${native_source}/VGui1LegacyRaster.cpp")
target_include_directories(native_face PUBLIC "${native_source}" "${vgui_source}/posix" "${vgui_source}/include")
target_link_libraries(native_face PUBLIC Freetype::Freetype)
add_library(native_provider STATIC "${native_source}/VGui1MacFontProvider.cpp")
target_link_libraries(native_provider PUBLIC native_face)
set_target_properties(native_provider PROPERTIES CXX_STANDARD 17)
if(APPLE)
    set(discovery "${CSE_ROOT}/GoldSrc-SDK/vgui2/vgui_surfacelib/native")
    add_library(provider_discovery STATIC "${discovery}/MacFontDiscovery.cpp")
    target_include_directories(provider_discovery PUBLIC "${discovery}")
    target_link_libraries(provider_discovery PUBLIC Freetype::Freetype "-framework CoreText" "-framework CoreFoundation")
    include("${CSE_ROOT}/cmake/MacFontService.cmake")
    cse_define_mac_font_registration_service("${discovery}/MacFontRegistration.cpp")
    target_link_libraries(native_provider PRIVATE provider_discovery cse_mac_font_registration)
    set_target_properties(provider_discovery PROPERTIES CXX_STANDARD 17)
endif()
add_executable(native_face_tests face_tests.cpp)
target_link_libraries(native_face_tests PRIVATE native_face)
add_executable(native_provider_tests provider_tests.cpp)
target_link_libraries(native_provider_tests PRIVATE native_provider)
if(APPLE)
    add_executable(native_real_probe real_probe.cpp)
    target_include_directories(native_real_probe PRIVATE "${CSE_ROOT}/tests/native/vgui1_legacy_raster")
    target_link_libraries(native_real_probe PRIVATE native_provider provider_discovery)
    set_target_properties(native_real_probe PROPERTIES CXX_STANDARD 17)
    add_executable(native_owner_tests owner_tests.cpp "${vgui_source}/posix/Font.cpp"
        "${vgui_source}/posix/VGui1FontServices.cpp" "${vgui_source}/Font.cpp"
        "${vgui_source}/common/fileimage.cpp" "${vgui_source}/common/vfontdata.cpp")
    target_include_directories(native_owner_tests PRIVATE "${vgui_source}/common")
    target_link_libraries(native_owner_tests PRIVATE native_provider)
    target_compile_options(native_owner_tests PRIVATE -fno-exceptions)
endif()
set(test_targets native_face native_provider native_face_tests native_provider_tests)
if(APPLE)
    list(APPEND test_targets native_real_probe native_owner_tests provider_discovery cse_mac_font_registration)
endif()
option(CSE_PROVIDER_SANITIZERS "ASan/UBSan owned provider/face fixtures" OFF)
foreach(target IN LISTS test_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off
            -ffp-model=strict)
    endif()
    if(CSE_PROVIDER_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
endforeach()
foreach(target native_face native_provider)
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHs-c-)
        target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
    else()
        target_compile_options(${target} PRIVATE -fno-exceptions)
    endif()
endforeach()
enable_testing()
add_test(NAME native-face-authored COMMAND native_face_tests "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs")
add_test(NAME native-provider-injected COMMAND native_provider_tests "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs")
set_tests_properties(native-face-authored native-provider-injected PROPERTIES TIMEOUT 30)
# No real discovery, registration or font-resource execution is an automatic test.
if(APPLE)
    add_test(NAME native-owner-bitmap-lifecycle COMMAND native_owner_tests
        "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs")
    set_tests_properties(native-owner-bitmap-lifecycle PROPERTIES TIMEOUT 30)
endif()
