cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseTier0PthreadFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT APPLE)
  message(FATAL_ERROR "This fixture executes actual Apple pthread ownership")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CSE_TIER0_SPIKE_DATABASE "" CACHE FILEPATH "Actual generated configuration compile_commands.json")
if(NOT EXISTS "${CSE_TIER0_SPIKE_DATABASE}")
  message(FATAL_ERROR "The generated spike compiler database is required")
endif()
option(CSE_TIER0_PTHREAD_SANITIZERS "Instrument the full actual TU and fixture" OFF)
set(object "${CMAKE_CURRENT_BINARY_DIR}/threadtools.o")
set(sanitizers)
if(CSE_TIER0_PTHREAD_SANITIZERS)
  set(sanitizers --sanitizers)
endif()
file(GLOB headers "${CSE_ROOT}/GoldSrc-SDK/public/tier0/*.h")
add_custom_command(OUTPUT "${object}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compile_spike.py"
    --database "${CSE_TIER0_SPIKE_DATABASE}" --sdk "${CSE_ROOT}/GoldSrc-SDK" --output "${object}" ${sanitizers}
  DEPENDS "${CSE_ROOT}/GoldSrc-SDK/tier0/threadtools.cpp" ${headers}
    "${CMAKE_CURRENT_SOURCE_DIR}/compile_spike.py" "${CSE_TIER0_SPIKE_DATABASE}"
  DEPFILE "${object}.d" VERBATIM)
set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(tier0_pthread_tests pthread_tests.cpp "${object}")
set_target_properties(tier0_pthread_tests PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(tier0_pthread_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
target_compile_definitions(tier0_pthread_tests PRIVATE TIER0_DLL POSIX OSX)
target_compile_options(tier0_pthread_tests PRIVATE -Wall -Wextra -Werror
  "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
target_link_libraries(tier0_pthread_tests PRIVATE Threads::Threads)
if(CSE_TIER0_PTHREAD_SANITIZERS)
  target_compile_options(tier0_pthread_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
    -fno-omit-frame-pointer)
  target_link_options(tier0_pthread_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(scenario current ownership worker repeated)
  add_test(NAME tier0-pthread-${scenario} COMMAND tier0_pthread_tests ${scenario})
  set_tests_properties(tier0-pthread-${scenario} PROPERTIES TIMEOUT 20)
endforeach()
