cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseCdTrackValues LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(CSE_ENABLE_SANITIZERS "Check defined native track arithmetic" OFF)
set(CSE_CD_SOURCE "${CSE_ROOT}/src/engine/cd.cpp" CACHE FILEPATH "Actual CD command source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_CD_SOURCE}" extract_track.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_track.py"
  "${CSE_CD_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/actual_native_track.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(cd_track_tests track_tests.cpp)
target_compile_features(cd_track_tests PRIVATE cxx_std_14)
target_include_directories(cd_track_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(cd_track_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/RTC1>")
else()
  target_compile_options(cd_track_tests PRIVATE -Wall -Wextra -Werror -Wshorten-64-to-32
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  if(CSE_ENABLE_SANITIZERS)
    target_compile_options(cd_track_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    target_link_options(cd_track_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_test(NAME cd-native-track-values COMMAND cd_track_tests)
