cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseModuleLoadingFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
set(CSE_MODULE_SOURCE_ROOT "${CSE_ROOT}/GoldSrc-SDK" CACHE PATH "Production or explicitly pinned loader sources")
enable_testing()
add_library(loader_leaf SHARED leaf.cpp)
option(CSE_MODULE_SANITIZE_LEAF "Apply selected sanitizer flags to the reloadable external fixture library" ON)
if(NOT CSE_MODULE_SANITIZE_LEAF)
  if(MSVC)
    message(FATAL_ERROR "The uninstrumented-leaf fixture mode is only defined for Clang/GNU")
  endif()
  target_compile_options(loader_leaf PRIVATE -fno-sanitize=all)
  target_link_options(loader_leaf PRIVATE -fno-sanitize=all)
endif()
if(WIN32)
  target_sources(loader_leaf PRIVATE leaf.def)
endif()
set_target_properties(loader_leaf PROPERTIES PREFIX "" OUTPUT_NAME "fixture leaf")
set(loader_variants engine public cstrike vgui2)
if(NOT WIN32)
  list(APPEND loader_variants hltv_core demoplayer)
endif()
set(loader_targets loader_leaf)
foreach(variant IN LISTS loader_variants)
  if(variant STREQUAL engine)
    set(source engine/interface.cpp)
  elseif(variant STREQUAL public)
    set(source public/interface.cpp)
  elseif(variant STREQUAL cstrike)
    set(source cstrike/public/interface.cpp)
  elseif(variant STREQUAL vgui2)
    set(source vgui2/src/public/interface.cpp)
  elseif(variant STREQUAL hltv_core)
    set(source HLTV/Core/private/public/interface.cpp)
  else()
    set(source HLTV/DemoPlayer/private/public/interface.cpp)
  endif()
  set(target module_loading_${variant}_tests)
  add_executable(${target} loader_tests.cpp "${CSE_MODULE_SOURCE_ROOT}/${source}")
  if(variant STREQUAL hltv_core)
    target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/HLTV/Core/private")
  elseif(variant STREQUAL demoplayer)
    target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/HLTV/DemoPlayer/private")
  endif()
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine" "${CSE_ROOT}/GoldSrc-SDK/public")
  # These fixtures do not publish the launcher's F entry point.
  target_compile_definitions(${target} PRIVATE CLIENT_DLL "$<$<PLATFORM_ID:Windows>:WIN32>")
  if(variant STREQUAL public AND WIN32)
    target_compile_definitions(${target} PRIVATE CSE_TEST_BORROWED_NAME_API=0)
  else()
    target_compile_definitions(${target} PRIVATE CSE_TEST_BORROWED_NAME_API=1)
  endif()
  if(variant STREQUAL engine OR variant STREQUAL public OR variant STREQUAL cstrike OR variant STREQUAL vgui2)
    target_compile_definitions(${target} PRIVATE CSE_TEST_NULL_FACTORY_MAIN=1)
  else()
    target_compile_definitions(${target} PRIVATE CSE_TEST_NULL_FACTORY_MAIN=0)
  endif()
  if(variant STREQUAL cstrike OR variant STREQUAL vgui2)
    target_compile_definitions(${target} PRIVATE CSE_TEST_MISSING_FACTORY_MAIN=1)
  else()
    target_compile_definitions(${target} PRIVATE CSE_TEST_MISSING_FACTORY_MAIN=0)
  endif()
  set_target_properties(${target} PROPERTIES ENABLE_EXPORTS ON)
  add_dependencies(${target} loader_leaf)
  target_link_libraries(${target} PRIVATE "${CMAKE_DL_LIBS}")
  list(APPEND loader_targets ${target})
  add_test(NAME actual-${variant}-module-lifecycle COMMAND ${target} "$<TARGET_FILE:loader_leaf>"
    "${CMAKE_CURRENT_BINARY_DIR}/module-trace-${variant}.bin")
endforeach()
foreach(target IN LISTS loader_targets)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
endforeach()
