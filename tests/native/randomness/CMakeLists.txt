cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseRandomnessBoundary LANGUAGES CXX)
set(CSE_RANDOM_IMPLEMENTATION "${CSE_ROOT}/src/cstrike/common/Random.cpp" CACHE FILEPATH "Actual random source")
option(CSE_ENABLE_SANITIZERS "Instrument actual RNG integer arithmetic" OFF)
add_executable(random_tests random_tests.cpp)
target_compile_features(random_tests PRIVATE cxx_std_14)
target_compile_definitions(random_tests PRIVATE CSE_RANDOM_IMPLEMENTATION="${CSE_RANDOM_IMPLEMENTATION}")
if(MSVC)
  set_property(TARGET random_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(random_tests PRIVATE /W4 /WX /fp:precise /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/RTC1>")
else()
  target_compile_options(random_tests PRIVATE -Wall -Wextra -Werror -Wshorten-64-to-32 -ffp-model=precise -ffp-contract=off
    -Wno-overriding-option "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  if(CSE_ENABLE_SANITIZERS)
    target_compile_options(random_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    target_link_options(random_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME random-win32-value-parity COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_LIST_DIR}/verify_output.py" "$<TARGET_FILE:random_tests>")
set_tests_properties(random-win32-value-parity PROPERTIES TIMEOUT 20)
