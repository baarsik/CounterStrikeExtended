cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(CseServerFormsTests LANGUAGES C CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_SRC "${CSE_ROOT}/src")
include("${CSE_ROOT}/cmake/WebPDependency.cmake")
include("${CSE_ROOT}/cmake/ExpatDependency.cmake")
option(CSE_SERVER_FORMS_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

enable_testing()
function(forms_test name)
    add_executable(${name} "${name}_tests.cpp" ${ARGN})
    target_compile_features(${name} PRIVATE cxx_std_17)
    target_include_directories(${name} PRIVATE "${CSE_SRC}/public" "${CSE_SRC}/amxx/csextended/forms")
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        if(CSE_SERVER_FORMS_SANITIZERS)
            target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${name} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME ${name} COMMAND ${name} "${CMAKE_CURRENT_BINARY_DIR}")
endfunction()

set(protocol "${CSE_SRC}/public/cse_server_forms.cpp" "${CSE_SRC}/public/cse_server_forms_xaml.cpp")
set(images "${CSE_SRC}/public/cse_server_forms_image.cpp" "${CSE_SRC}/common/image_codec.cpp")
set(cache "${CSE_SRC}/cstrike/cl_dll/server_form_cache.cpp")
set(session "${CSE_SRC}/cstrike/cl_dll/server_form_session.cpp")

forms_test(protocol "${protocol}")

    CSE_FORMS_SHOWCASE="${CSE_SRC}/amxx/csextended/deploy/forms/showcase.xaml")

forms_test(server "${protocol}" "${server}")
forms_test(key)
