cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseSteamServerProviderFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
include("${CSE_ROOT}/cmake/SteamDependency.cmake")
set(CSE_STEAM_FIXTURE_PROVIDER "" CACHE FILEPATH "Explicit pinned provider; execution separately authorized")
if(NOT CSE_STEAM_FIXTURE_PROVIDER)
  message(FATAL_ERROR "Server provider fixture requires an explicit pinned provider")
endif()
cse_verify_steam_provider("${CSE_STEAM_FIXTURE_PROVIDER}" "${CSE_TARGET_OS}" "${CSE_TARGET_ARCH}")
add_library(steam_server_owned_module SHARED server_probe.cpp "${CSE_ROOT}/GoldSrc-SDK/common/steam_service.cpp")
target_compile_features(steam_server_owned_module PRIVATE cxx_std_17)
target_compile_definitions(steam_server_owned_module PRIVATE "CSE_SERVER_PROVIDER_SOURCE_PATH=\"${CSE_STEAM_FIXTURE_PROVIDER}\"")
target_include_directories(steam_server_owned_module PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
target_include_directories(steam_server_owned_module SYSTEM PRIVATE "${CSE_ROOT}/GoldSrc-SDK/external/steamworks")
target_compile_definitions(steam_server_owned_module PRIVATE _CRT_SECURE_NO_WARNINGS CSE_OWNED_STEAM_PROVIDER=1
  "CSE_STEAM_PROVIDER_PATH=\"${CSE_STEAM_FIXTURE_PROVIDER}\"")
cse_force_steam_initialization(steam_server_owned_module)
target_link_libraries(steam_server_owned_module PRIVATE "${CMAKE_DL_LIBS}")
if(MSVC)
  target_compile_options(steam_server_owned_module PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  target_compile_options(steam_server_owned_module PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
endif()
target_compile_definitions(steam_server_owned_module PRIVATE CSE_SERVER_MODULE_FIXTURE)
add_executable(steam_server_probe module_probe.cpp)
target_compile_features(steam_server_probe PRIVATE cxx_std_17)
target_include_directories(steam_server_probe PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
target_link_libraries(steam_server_probe PRIVATE "${CMAKE_DL_LIBS}")
add_dependencies(steam_server_probe steam_server_owned_module)
# Real server initialization/login is never registered as an automatic CTest.

# Optional raw-SDK control distinguishes provider behavior from owned-service routing.
option(CSE_STEAM_SERVER_BUILD_DIRECT_CONTROL "Build a direct SDK comparison executable" OFF)
set(CSE_STEAM_DIRECT_IMPORT_FILE "" CACHE FILEPATH "Pinned Windows x64 import archive for direct control")
if(CSE_STEAM_SERVER_BUILD_DIRECT_CONTROL)
  add_executable(steam_server_direct_probe server_probe.cpp)
  target_compile_features(steam_server_direct_probe PRIVATE cxx_std_17)
  target_include_directories(steam_server_direct_probe PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
  target_include_directories(steam_server_direct_probe SYSTEM PRIVATE "${CSE_ROOT}/GoldSrc-SDK/external/steamworks")
  target_compile_definitions(steam_server_direct_probe PRIVATE _CRT_SECURE_NO_WARNINGS)
  if(WIN32)
    if(NOT CSE_TARGET_ARCH STREQUAL "x64" OR NOT EXISTS "${CSE_STEAM_DIRECT_IMPORT_FILE}")
      message(FATAL_ERROR "Direct Windows control requires the pinned x64 import archive")
    endif()
    file(READ "${CSE_ROOT}/tools/qualification/fixtures/steam/lock.json" steam_lock)
    string(JSON expected_import_sha GET "${steam_lock}" windows_x64 import_library sha256)
    file(SHA256 "${CSE_STEAM_DIRECT_IMPORT_FILE}" actual_import_sha)
    if(NOT expected_import_sha STREQUAL actual_import_sha)
      message(FATAL_ERROR "Direct-control import archive does not match the provider pin")
    endif()
    target_link_libraries(steam_server_direct_probe PRIVATE "${CSE_STEAM_DIRECT_IMPORT_FILE}")
    set(direct_provider_name "steam_api64.dll")
  elseif(APPLE)
    target_link_libraries(steam_server_direct_probe PRIVATE "${CSE_STEAM_FIXTURE_PROVIDER}")
    set(direct_provider_name "libsteam_api.dylib")
  else()
    message(FATAL_ERROR "Direct control is limited to the two currently pinned native providers")
  endif()
  if(MSVC)
    target_compile_options(steam_server_direct_probe PRIVATE /W4 /WX /EHsc
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(steam_server_direct_probe PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
  target_compile_definitions(steam_server_direct_probe PRIVATE
    "CSE_SERVER_PROVIDER_SOURCE_PATH=\"${CSE_STEAM_FIXTURE_PROVIDER}\""
    "CSE_SERVER_PROVIDER_STAGED_PATH=\"$<TARGET_FILE_DIR:steam_server_direct_probe>/${direct_provider_name}\"")
  target_link_libraries(steam_server_direct_probe PRIVATE "${CMAKE_DL_LIBS}")
  add_custom_command(TARGET steam_server_direct_probe POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${CSE_STEAM_FIXTURE_PROVIDER}" "$<TARGET_FILE_DIR:steam_server_direct_probe>/${direct_provider_name}")
endif()

  target_compile_features(steam_server_direct_module PRIVATE cxx_std_17)

    "CSE_SERVER_PROVIDER_SOURCE_PATH=\"${CSE_STEAM_FIXTURE_PROVIDER}\""
    "CSE_SERVER_PROVIDER_STAGED_PATH=\"$<TARGET_FILE_DIR:steam_server_direct_module>/${direct_provider_name}\"")
  target_link_libraries(steam_server_direct_module PRIVATE "${CMAKE_DL_LIBS}")
  if(WIN32)
    target_link_libraries(steam_server_direct_module PRIVATE "${CSE_STEAM_DIRECT_IMPORT_FILE}")
  else()

# Explicit lifecycle control: retire the module-owned transport outside the
# loader lock before dlclose; the late finalizer is an idempotent fallback.
add_library(steam_shutdown_owned_module SHARED shutdown_probe.cpp "${CSE_ROOT}/GoldSrc-SDK/common/steam_service.cpp")
target_compile_features(steam_shutdown_owned_module PRIVATE cxx_std_17)
target_include_directories(steam_shutdown_owned_module PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
target_include_directories(steam_shutdown_owned_module SYSTEM PRIVATE "${CSE_ROOT}/GoldSrc-SDK/external/steamworks")
target_compile_definitions(steam_shutdown_owned_module PRIVATE CSE_OWNED_STEAM_PROVIDER=1
  CSE_SHUTDOWN_MODULE_FIXTURE=1 "CSE_STEAM_PROVIDER_PATH=\"${CSE_STEAM_FIXTURE_PROVIDER}\"")
cse_force_steam_initialization(steam_shutdown_owned_module)
target_link_libraries(steam_shutdown_owned_module PRIVATE "${CMAKE_DL_LIBS}")
set_target_properties(steam_shutdown_owned_module PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(steam_shutdown_module_probe shutdown_module_probe.cpp)
target_compile_features(steam_shutdown_module_probe PRIVATE cxx_std_17)
target_include_directories(steam_shutdown_module_probe PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
target_link_libraries(steam_shutdown_module_probe PRIVATE "${CMAKE_DL_LIBS}")
add_dependencies(steam_shutdown_module_probe steam_shutdown_owned_module)
