cmake_minimum_required(VERSION 3.25)
project(CseVgui1LegacyRaster LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_owned_font.py"
    "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_reference.py"
    "${CSE_ROOT}/src/utils/vgui/lib-src/vgui/win32/Font.cpp" "${CMAKE_CURRENT_BINARY_DIR}/extracted"
    COMMAND_ERROR_IS_FATAL ANY)
option(CSE_RASTER_REFERENCE_ONLY "Only compile the real Windows GDI authored-font reference" OFF)
if(WIN32)
    add_executable(raster_gdi_reference gdi_reference.cpp)
    target_include_directories(raster_gdi_reference PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/extracted")
    target_compile_definitions(raster_gdi_reference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_compile_options(raster_gdi_reference PRIVATE /W4 /WX /EHsc /fp:precise)
    target_link_libraries(raster_gdi_reference PRIVATE gdi32 user32 bcrypt)
endif()
if(CSE_RASTER_REFERENCE_ONLY)
    return()
endif()
set(raster_source "${CSE_ROOT}/src/utils/vgui/lib-src/vgui/posix/native")
add_library(legacy_raster STATIC "${raster_source}/VGui1LegacyRaster.cpp")
target_include_directories(legacy_raster PUBLIC "${raster_source}")
set(CSE_FREETYPE_PREFIX "" CACHE PATH "Exact pinned source-built FreeType dependency prefix")
set(VCPKG_IMPORT_PREFIX "${CSE_FREETYPE_PREFIX}")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG TRUE)
list(PREPEND CMAKE_PREFIX_PATH "${CSE_FREETYPE_PREFIX}")
find_package(ZLIB CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/zlib" NO_DEFAULT_PATH)
if(NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB ALIAS ZLIB::ZLIBSTATIC)
endif()
find_library(BZIP2_LIBRARY_RELEASE NAMES bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(BZIP2_LIBRARY_DEBUG NAMES bz2d bzip2d libbz2d libbzip2d bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/debug/lib" NO_DEFAULT_PATH REQUIRED)
find_path(BZIP2_INCLUDE_DIR NAMES bzlib.h PATHS "${CSE_FREETYPE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_package(BZip2 REQUIRED)
find_package(PNG CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/png" NO_DEFAULT_PATH)
find_package(freetype CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/freetype" NO_DEFAULT_PATH)
target_link_libraries(legacy_raster PRIVATE Freetype::Freetype)
add_executable(raster_pixel_tests pixel_tests.cpp)
add_executable(raster_candidate_tests candidate_tests.cpp)
target_link_libraries(raster_candidate_tests PRIVATE legacy_raster)
add_executable(raster_replay_tests replay_tests.cpp)
target_link_libraries(raster_replay_tests PRIVATE legacy_raster)
target_include_directories(raster_pixel_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/extracted")
target_link_libraries(raster_pixel_tests PRIVATE legacy_raster)
if(WIN32)
    target_compile_definitions(raster_pixel_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(raster_pixel_tests PRIVATE bcrypt)
    target_compile_options(legacy_raster PRIVATE /EHs-c-)
    target_compile_definitions(legacy_raster PRIVATE _HAS_EXCEPTIONS=0)
endif()
foreach(target legacy_raster raster_pixel_tests raster_candidate_tests raster_replay_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off
            -ffp-model=strict)
    endif()
endforeach()
option(CSE_RASTER_SANITIZERS "Instrument owned raster source and fixtures" OFF)
if(CSE_RASTER_SANITIZERS AND NOT MSVC)
    foreach(target legacy_raster raster_pixel_tests raster_candidate_tests raster_replay_tests)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endforeach()
endif()
if(NOT MSVC)
    target_compile_options(legacy_raster PRIVATE -fno-exceptions)
endif()
enable_testing()
add_test(NAME legacy-pixel-transfer COMMAND raster_pixel_tests)
add_test(NAME legacy-pixel-rgb-domain COMMAND raster_pixel_tests rgb)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/candidate-output")
add_test(NAME legacy-candidate-data COMMAND raster_candidate_tests
    "${CMAKE_CURRENT_BINARY_DIR}/owned-inputs/owned-raster.ttf" "${CMAKE_CURRENT_BINARY_DIR}/candidate-output")
set(CSE_GDI_REFERENCE_DIRECTORY "" CACHE PATH "Frozen authored GDI input for pure transfer replay")
if(CSE_GDI_REFERENCE_DIRECTORY)
    add_test(NAME legacy-gdi-transfer-replay COMMAND raster_replay_tests "${CSE_GDI_REFERENCE_DIRECTORY}")
    set_tests_properties(legacy-gdi-transfer-replay PROPERTIES TIMEOUT 60)
endif()
set_tests_properties(legacy-pixel-transfer legacy-pixel-rgb-domain legacy-candidate-data PROPERTIES TIMEOUT 60)
