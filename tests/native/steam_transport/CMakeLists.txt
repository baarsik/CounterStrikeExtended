cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseSteamServerTransportFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_library(transport_provider SHARED fake_provider.cpp)
add_library(transport_incomplete_provider SHARED fake_provider.cpp)
target_compile_definitions(transport_incomplete_provider PRIVATE CSE_INCOMPLETE_TRANSPORT_PROVIDER)
add_executable(transport_tests transport_tests.cpp)
foreach(target transport_provider transport_incomplete_provider transport_tests)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
endforeach()
target_link_libraries(transport_tests PRIVATE "${CMAKE_DL_LIBS}")
add_test(NAME native-steam-server-transport COMMAND transport_tests
  "$<TARGET_FILE:transport_provider>" "$<TARGET_FILE:transport_incomplete_provider>")
