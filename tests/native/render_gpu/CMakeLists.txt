cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseRenderGPU LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(CSE_RENDER_GPU_SANITIZERS "Instrument owned CPU preparation/adapter code" OFF)
set(CSE_RENDER_GPU_OFFLINE_DIRECTORY "" CACHE PATH "Actual offline FXC producer output for reflection checks")
set(CSE_RENDER_GPU_METAL_DIRECTORY "" CACHE PATH "Actual offline Metal producer output for AIR layout checks")
if(CSE_RENDER_GPU_SANITIZERS AND MSVC)
  message(FATAL_ERROR "This fixture ASan/UBSan option requires Clang/GNU")
endif()
set(native "${CSE_ROOT}/GoldSrc-SDK/engine/render/native")
add_library(render_gpu STATIC "${native}/legacy_state.cpp" "${native}/legacy_commands.cpp"
  "${native}/legacy_texture.cpp" "${native}/gpu_capture.cpp")
target_include_directories(render_gpu PUBLIC "${native}")
set(CSE_RENDER_GPU_SDL_LIBRARY "" CACHE FILEPATH "Matching pinned SDL3 import/shared library")
set(CSE_RENDER_GPU_SDL_RUNTIME "" CACHE FILEPATH "Matching pinned SDL3 runtime")
if(NOT EXISTS "${CSE_RENDER_GPU_SDL_LIBRARY}" OR NOT EXISTS "${CSE_RENDER_GPU_SDL_RUNTIME}")
  message(FATAL_ERROR "Explicit matching source-built SDL3 library/runtime required")
endif()
add_library(render_gpu_sdl SHARED IMPORTED)
if(WIN32)
  set_target_properties(render_gpu_sdl PROPERTIES IMPORTED_IMPLIB "${CSE_RENDER_GPU_SDL_LIBRARY}"
    IMPORTED_LOCATION "${CSE_RENDER_GPU_SDL_RUNTIME}")
else()
  set_target_properties(render_gpu_sdl PROPERTIES IMPORTED_LOCATION "${CSE_RENDER_GPU_SDL_LIBRARY}")
endif()
target_sources(render_gpu PRIVATE "${native}/gpu_adapter.cpp")
target_include_directories(render_gpu PUBLIC "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
target_link_libraries(render_gpu PUBLIC render_gpu_sdl)
add_executable(render_gpu_capture_tests capture_tests.cpp)
target_link_libraries(render_gpu_capture_tests PRIVATE render_gpu)
add_executable(render_gpu_probe gpu_probe.cpp)
target_link_libraries(render_gpu_probe PRIVATE render_gpu)
set(render_gpu_targets render_gpu render_gpu_capture_tests render_gpu_probe)
set(render_gpu_executables render_gpu_capture_tests render_gpu_probe)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(render_gpu_matrix_probe gpu_matrix_probe.cpp)
  target_link_libraries(render_gpu_matrix_probe PRIVATE render_gpu)
  list(APPEND render_gpu_targets render_gpu_matrix_probe)
  list(APPEND render_gpu_executables render_gpu_matrix_probe)
endif()
foreach(target IN LISTS render_gpu_targets)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /fp:precise
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
  if(CSE_RENDER_GPU_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
foreach(target IN LISTS render_gpu_executables)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${CSE_RENDER_GPU_SDL_RUNTIME}" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
endforeach()
enable_testing()
add_test(NAME render-gpu-capture COMMAND render_gpu_capture_tests)
if(APPLE)
  add_test(NAME render-gpu-metal-aliases COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}" -p test_metal_toolchain.py -v)
endif()
if(CSE_RENDER_GPU_OFFLINE_DIRECTORY)
  add_test(NAME render-gpu-offline-reflection COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/verify_offline.py" --directory "${CSE_RENDER_GPU_OFFLINE_DIRECTORY}"
    --source "${native}/shaders/legacy_first.hlsl")
endif()
if(CSE_RENDER_GPU_METAL_DIRECTORY)
  add_test(NAME render-gpu-offline-metal COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/verify_metal.py" --directory "${CSE_RENDER_GPU_METAL_DIRECTORY}"
    --source "${native}/shaders/legacy_first.metal")
endif()
