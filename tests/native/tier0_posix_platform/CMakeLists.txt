cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseTier0PosixPlatform LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT APPLE)
  message(FATAL_ERROR "This fixture qualifies the actual Mac POSIX platform implementation")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_POSIX_PLATFORM_DATABASE "" CACHE FILEPATH "Actual generated compile_commands.json")
set(CSE_POSIX_PLATFORM_SOURCE "${CSE_ROOT}/src/tier0/platform_posix.cpp" CACHE FILEPATH
  "Actual production source or an explicitly preserved negative reference")
option(CSE_POSIX_PLATFORM_SANITIZERS "Instrument the complete actual TU and fixture" OFF)
option(CSE_POSIX_PLATFORM_LEGACY_REFERENCE "Demote visible width/strncat warnings only in a preserved negative reference" OFF)
if(NOT EXISTS "${CSE_POSIX_PLATFORM_DATABASE}")
  message(FATAL_ERROR "The actual generated compiler database is required")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_bindings.py"
  "${CSE_POSIX_PLATFORM_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/vcr_bindings.h" COMMAND_ERROR_IS_FATAL ANY)
function(platform_fixture target)
  set(object "${CMAKE_CURRENT_BINARY_DIR}/${target}-platform.o")
  set(options)
  if(target STREQUAL posix_time_tests)
    list(APPEND options --controlled-clock)
  endif()
  if(CSE_POSIX_PLATFORM_SANITIZERS)
    list(APPEND options --sanitizers)
  endif()
  if(CSE_POSIX_PLATFORM_LEGACY_REFERENCE)
    list(APPEND options --legacy-reference)
  endif()
  add_custom_command(OUTPUT "${object}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py"
      --database "${CSE_POSIX_PLATFORM_DATABASE}" --sdk "${CSE_ROOT}/src"
      --source "${CSE_POSIX_PLATFORM_SOURCE}" --output "${object}" ${options}
    DEPENDS "${CSE_POSIX_PLATFORM_SOURCE}" "${CSE_ROOT}/src/public/tier0/platform.h"
      "${CSE_ROOT}/src/public/cse_posix_crt.h" "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py"
      "${CSE_POSIX_PLATFORM_DATABASE}" DEPFILE "${object}.d" VERBATIM)
  set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  add_executable(${target} platform_tests.cpp "${object}")
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/src/public" "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(${target} PRIVATE TIER0_DLL POSIX OSX)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  if(CSE_POSIX_PLATFORM_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
      -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
platform_fixture(posix_platform_tests)
platform_fixture(posix_time_tests)
enable_testing()
foreach(case command-valid command-limits live)
  add_test(NAME tier0-posix-platform-${case} COMMAND posix_platform_tests ${case})
endforeach()
foreach(case time-normal time-high vcr zero-epoch)
  add_test(NAME tier0-posix-platform-${case} COMMAND posix_time_tests ${case})
endforeach()
get_directory_property(test_names TESTS)
set_tests_properties(${test_names} PROPERTIES TIMEOUT 20)
