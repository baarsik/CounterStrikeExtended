cmake_minimum_required(VERSION 3.25)
project(CseMuzzleFlashFixture LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(source "${CSE_ROOT}/src/engine/cl_tent.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}" extract.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py"
    "${source}" "${CMAKE_CURRENT_BINARY_DIR}/muzzle_flash.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(muzzle_flash_tests muzzle_flash_tests.cpp)
target_compile_features(muzzle_flash_tests PRIVATE cxx_std_17)
set_target_properties(muzzle_flash_tests PROPERTIES CXX_EXTENSIONS NO)
target_include_directories(muzzle_flash_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/common" "${CSE_ROOT}/src/public"
    "${CSE_ROOT}/src/cl_dll" "${CSE_ROOT}/src/pm_shared" "${CSE_ROOT}/src/external/SDL3/include")
if(MSVC)
    target_compile_options(muzzle_flash_tests PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(muzzle_flash_tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
option(CSE_MUZZLE_SANITIZERS "Enable ASan and UBSan" OFF)
if(CSE_MUZZLE_SANITIZERS)
    target_compile_options(muzzle_flash_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(muzzle_flash_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME muzzle-flash-origin COMMAND muzzle_flash_tests)
