cmake_minimum_required(VERSION 3.25)
project(cse_mac_settings_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(WIN32)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
    add_custom_command(OUTPUT "${generated}/app_registry.inc"
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/extract_windows_fixture.py"
            --root "${source_root}" --output "${generated}"
        DEPENDS extract_windows_fixture.py
            "${source_root}/src/utils/vgui/lib-src/vgui/win32/App.cpp"
            "${source_root}/src/utils/vgui/lib-src/vgui/vgui.cpp"
        VERBATIM)
    add_executable(mac_settings_win32_raw windows_registry_fixture.cpp "${generated}/app_registry.inc")
    target_include_directories(mac_settings_win32_raw PRIVATE "${generated}")
    target_compile_definitions(mac_settings_win32_raw PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(mac_settings_win32_raw PRIVATE advapi32)
    if(MSVC)
        target_compile_options(mac_settings_win32_raw PRIVATE /W4 /WX)
    endif()
    enable_testing()
    add_test(NAME mac_settings_actual_win32_app COMMAND mac_settings_win32_raw)
    set_tests_properties(mac_settings_actual_win32_app PROPERTIES TIMEOUT 30)
    return()
endif()
if(NOT APPLE)
    message(FATAL_ERROR "This fixture requires Apple's CoreFoundation or the explicit Win32 reference")
endif()
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
option(CSE_MAC_SETTINGS_SANITIZERS "Instrument the nonpersistent model fixture" OFF)
find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
add_executable(mac_settings_model model_tests.cpp)
target_link_libraries(mac_settings_model PRIVATE ${COREFOUNDATION_FRAMEWORK})
target_compile_options(mac_settings_model PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Werror)
if(CSE_MAC_SETTINGS_SANITIZERS)
    target_compile_options(mac_settings_model PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mac_settings_model PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME mac_settings_nonpersistent_model COMMAND mac_settings_model)

# Build actual production symbols, but do not execute product-domain preferences.
add_library(mac_settings_store_compile STATIC ../../../src/common/mac_settings_store.cpp)
target_link_libraries(mac_settings_store_compile PRIVATE ${COREFOUNDATION_FRAMEWORK})
target_compile_options(mac_settings_store_compile PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Werror)
add_executable(mac_settings_persistent persistent_fixture.cpp)
target_link_libraries(mac_settings_persistent PRIVATE ${COREFOUNDATION_FRAMEWORK})
target_compile_options(mac_settings_persistent PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Werror)
# There is intentionally no add_test for the persistent fixture.
add_executable(mac_settings_header header_compile.cpp)
target_compile_options(mac_settings_header PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Werror)
add_test(NAME mac_settings_header_cxx14 COMMAND mac_settings_header)
set_tests_properties(mac_settings_nonpersistent_model mac_settings_header_cxx14 PROPERTIES TIMEOUT 30)
