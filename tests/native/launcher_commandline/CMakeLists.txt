cmake_minimum_required(VERSION 3.25)
project(CseLauncherCommandLine LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CSE_COMMANDLINE_SANITIZERS "Instrument the actual launcher command-line implementation" OFF)
set(CSE_COMMANDLINE_REFERENCE "" CACHE FILEPATH "Preserved Windows parser source for behavior comparison")
add_executable(launcher_commandline_tests commandline_tests.cpp
  "${CSE_ROOT}/GoldSrc-SDK/launcher/common/commandline.cpp")
set(parser_targets launcher_commandline_tests)
if(WIN32 AND CSE_COMMANDLINE_REFERENCE)
  add_executable(launcher_commandline_reference commandline_tests.cpp "${CSE_COMMANDLINE_REFERENCE}")
  list(APPEND parser_targets launcher_commandline_reference)
endif()
foreach(target IN LISTS parser_targets)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/launcher/common")
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas)
    if(CSE_COMMANDLINE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
set(scenarios basic mutators quoted response whitespace limits)
if(APPLE)
  list(APPEND scenarios canonical-restart)
endif()
if(NOT WIN32)
  list(APPEND scenarios argv argv-boundary argv-overflow)
endif()
foreach(scenario IN LISTS scenarios)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/work-${scenario}")
  add_test(NAME launcher-commandline-${scenario} COMMAND launcher_commandline_tests "${scenario}")
  set_tests_properties(launcher-commandline-${scenario} PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/work-${scenario}")
endforeach()
if(TARGET launcher_commandline_reference)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME launcher-commandline-windows-reference COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/compare_windows.py" "$<TARGET_FILE:launcher_commandline_tests>"
    "$<TARGET_FILE:launcher_commandline_reference>")
endif()
