cmake_minimum_required(VERSION 3.25)
project(CseVoiceBanFiles LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CSE_BAN_SANITIZERS "Instrument the complete voice-ban owners" OFF)
set(CSE_BAN_REFERENCE_ROOT "" CACHE PATH "Preserved legacy HL/CS source for valid-file comparison")
enable_testing()
foreach(game hl cs)
    if(game STREQUAL hl)
        set(owner "${CSE_ROOT}/GoldSrc-SDK/game_shared")
    else()
        set(owner "${CSE_ROOT}/GoldSrc-SDK/cstrike/game_shared")
    endif()
    set(variants current)
    if(CSE_BAN_REFERENCE_ROOT)
        list(APPEND variants reference)
    endif()
    foreach(variant IN LISTS variants)
        set(target voice_bans_${game}_${variant})
        if(variant STREQUAL reference)
            set(source "${CSE_BAN_REFERENCE_ROOT}/${game}.cpp")
            set(mode valid)
        else()
            set(source "${owner}/voice_banmgr.cpp")
            set(mode full)
        endif()
        add_executable(${target} file_tests.cpp "${source}")
        set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
        target_include_directories(${target} PRIVATE "${owner}")
        if(MSVC)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Werror=shorten-64-to-32
                -include "${CSE_ROOT}/GoldSrc-SDK/public/cse_posix_crt.h")
            if(variant STREQUAL reference)
                # Baseline diagnostic only; execute complete valid files, never malformed legacy reads.
                target_compile_options(${target} PRIVATE -Wno-error=shorten-64-to-32)
            endif()
            if(CSE_BAN_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
                    -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
            endif()
        endif()
        set(work "${CMAKE_CURRENT_BINARY_DIR}/${target}-owned")
        file(MAKE_DIRECTORY "${work}")
        file(WRITE "${work}/owner.marker" "CSE voice-ban authored fixture V1")
        add_test(NAME ${target} COMMAND ${target} "${work}" "${mode}")
    endforeach()
endforeach()
