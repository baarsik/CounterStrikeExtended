cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseTier0AllocatorFixtures LANGUAGES C CXX RC)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "The retained small-block heap uses Windows VirtualAlloc; use a matching MSVC target")
endif()
set(CSE_SDK_DIR "${CSE_ROOT}/src")
set(CSE_TIER0_EXTERNAL_MODULE "" CACHE FILEPATH "Preserved complete tier0 module for the negative reference")
if(CSE_TIER0_EXTERNAL_MODULE)
  set(tier0_module "${CSE_TIER0_EXTERNAL_MODULE}")
else()
  set(CSE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime/$<CONFIG>")
  include("${CSE_ROOT}/cmake/RuntimeTargets.cmake")
  add_subdirectory("${CSE_ROOT}/cmake/targets/tier0" "${CMAKE_CURRENT_BINARY_DIR}/tier0")
  # Diagnostic symbols change no source/FP/EH/assert policy and remain outside runtime outputs.
  target_link_options(tier0 PRIVATE /DEBUG:FULL)
  set(tier0_module "$<TARGET_FILE:tier0>")
endif()
add_executable(tier0_allocator_tests allocator_tests.cpp)
target_compile_features(tier0_allocator_tests PRIVATE cxx_std_14)
target_include_directories(tier0_allocator_tests PRIVATE "${CSE_SDK_DIR}/public")
target_compile_definitions(tier0_allocator_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(tier0_allocator_tests PRIVATE /W4 /WX /EHsc
  "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
if(TARGET tier0)
  add_dependencies(tier0_allocator_tests tier0)
endif()
enable_testing()
foreach(scenario startup sizes reuse realloc)
  add_test(NAME "tier0-allocator-${scenario}" COMMAND tier0_allocator_tests "${tier0_module}" "${scenario}")
  set_tests_properties("tier0-allocator-${scenario}" PROPERTIES TIMEOUT 30
    PASS_REGULAR_EXPRESSION "tier0_allocator_passed=1")
endforeach()
