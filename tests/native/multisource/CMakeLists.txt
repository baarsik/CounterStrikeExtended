cmake_minimum_required(VERSION 3.25)
project(CseMultisource LANGUAGES CXX)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_MULTISOURCE_SOURCE "${CSE_ROOT}/GoldSrc-SDK/dlls/buttons.cpp" CACHE FILEPATH "Actual source under test")
option(CSE_MULTISOURCE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_MULTISOURCE_SOURCE}"
    "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${CSE_ROOT}/GoldSrc-SDK/dlls/cbase.h")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py"
    "${CSE_MULTISOURCE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/imported/multisource_bodies.inc"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(multisource_tests multisource_tests.cpp)
target_include_directories(multisource_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/imported")
target_compile_features(multisource_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(multisource_tests PRIVATE /W4 /WX /wd4100 /EHsc)
else()
    target_compile_options(multisource_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
    if(CSE_MULTISOURCE_SANITIZERS)
        target_compile_options(multisource_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(multisource_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME multisource-behavior COMMAND multisource_tests)
