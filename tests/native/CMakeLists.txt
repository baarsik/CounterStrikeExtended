cmake_minimum_required(VERSION 3.25)
project(CseNativeBoundaryTests LANGUAGES CXX)
enable_testing()
if(MSVC)
  foreach(mode enabled ndebug)
    add_executable(utlmemory_${mode} utlmemory_assert_tests.cpp)
    target_compile_features(utlmemory_${mode} PRIVATE cxx_std_17)
    target_compile_options(utlmemory_${mode} PRIVATE /W4 /WX /EHsc /analyze /analyze:external-)
    target_compile_definitions(utlmemory_${mode} PRIVATE UTL_ENGINE_ASSERTS)
    if(mode STREQUAL enabled)
      target_compile_options(utlmemory_${mode} PRIVATE /UNDEBUG)
    else()
      target_compile_definitions(utlmemory_${mode} PRIVATE NDEBUG)
    endif()
    add_test(NAME utlmemory_${mode} COMMAND utlmemory_${mode})
  endforeach()
endif()
