cmake_minimum_required(VERSION 3.25)
project(a3d_platform LANGUAGES CXX)
enable_testing()
set(CSE_TARGET_ARCH "" CACHE STRING "Fixture target architecture")
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(fixture layout interface)
    add_executable(${fixture}_tests ${fixture}_tests.cpp)
    target_include_directories(${fixture}_tests PRIVATE "${CSE_ROOT}/src/a3dapi")
    target_compile_features(${fixture}_tests PRIVATE cxx_std_17)
    add_test(NAME ${fixture}_tests COMMAND ${fixture}_tests)
endforeach()

target_sources(interface_tests PRIVATE "${CSE_ROOT}/src/a3dapi/a3d_guids.cpp")
if(WIN32)
    target_link_libraries(interface_tests PRIVATE uuid dxguid)
endif()
