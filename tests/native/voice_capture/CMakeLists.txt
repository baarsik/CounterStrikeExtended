cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseVoiceCaptureReferences LANGUAGES CXX)
find_package(Threads REQUIRED)

set(engine "${CSE_ROOT}/GoldSrc-SDK/engine")
add_executable(voice_openal_historical_reference openal_historical_reference.cpp)
target_compile_definitions(voice_openal_historical_reference PRIVATE
    CSE_VOICE_OPENAL_SOURCE="${engine}/voice_record_openal.cpp")
set(reference_targets voice_openal_historical_reference)
if(WIN32)
    add_executable(voice_windows_reference legacy_windows_reference.cpp)
    target_compile_definitions(voice_windows_reference PRIVATE WIN32 DLL_EXPORT= _CRT_SECURE_NO_WARNINGS
        CSE_VOICE_WAVEIN_SOURCE="${engine}/voice_record_wavein.cpp"
        CSE_VOICE_DSOUND_SOURCE="${engine}/voice_record_dsound.cpp"
        CSE_VOICE_CIRCULAR_SOURCE="${engine}/CircularBuffer.cpp")
    target_include_directories(voice_windows_reference PRIVATE "${engine}" "${CSE_ROOT}/GoldSrc-SDK/common"
        "${CSE_ROOT}/GoldSrc-SDK/public")
    list(APPEND reference_targets voice_windows_reference)
endif()

option(CSE_VOICE_COMPONENT_BASELINE "Build the failing missing-recorder test baseline" OFF)
add_library(capture_fake_sdl SHARED fake_sdl.cpp)
target_compile_definitions(capture_fake_sdl PRIVATE CSE_BUILD_FAKE_SDL)
target_include_directories(capture_fake_sdl PUBLIC "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include"
    "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(capture_fake_sdl PRIVATE Threads::Threads)
if(WIN32)
    target_sources(capture_fake_sdl PRIVATE fake_sdl_exports.def)
endif()

if(CSE_VOICE_COMPONENT_BASELINE)
    add_library(capture_component OBJECT missing_capture_baseline.cpp)
    target_compile_definitions(capture_component PRIVATE CSE_CAPTURE_BASELINE)
else()
    add_library(capture_component OBJECT "${engine}/voice_record_sdl.cpp")
endif()
set_property(TARGET capture_component PROPERTY POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
    # Bind the module's own factory/callback implementation locally; the driver
    # checks that its callback address belongs to the actual loaded image.
    target_compile_options(capture_component PRIVATE -fvisibility=hidden)
endif()
target_include_directories(capture_component PRIVATE "${engine}" "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include" "${CSE_ROOT}/GoldSrc-SDK/common" "${CSE_ROOT}/GoldSrc-SDK/public")
add_executable(voice_capture_tests component_tests.cpp allocation_fault.cpp $<TARGET_OBJECTS:capture_component>)
target_include_directories(voice_capture_tests PRIVATE "${engine}")
target_link_libraries(voice_capture_tests PRIVATE capture_fake_sdl Threads::Threads)
if(CSE_VOICE_COMPONENT_BASELINE)
    target_compile_definitions(voice_capture_tests PRIVATE CSE_CAPTURE_BASELINE)
endif()
list(APPEND reference_targets capture_fake_sdl capture_component voice_capture_tests)
if(NOT CSE_VOICE_COMPONENT_BASELINE)
    add_library(capture_fixture_module SHARED capture_fixture_module.cpp $<TARGET_OBJECTS:capture_component>)
    target_include_directories(capture_fixture_module PRIVATE "${engine}")
    target_link_libraries(capture_fixture_module PRIVATE capture_fake_sdl)
    if(WIN32)
        target_sources(capture_fixture_module PRIVATE capture_fixture_module.def)
    endif()
    add_executable(voice_capture_module_tests module_lifecycle_tests.cpp)
    target_include_directories(voice_capture_module_tests PRIVATE "${engine}")
    target_link_libraries(voice_capture_module_tests PRIVATE capture_fake_sdl Threads::Threads ${CMAKE_DL_LIBS})
    list(APPEND reference_targets capture_fixture_module voice_capture_module_tests)
endif()

option(CSE_VOICE_REFERENCE_SANITIZERS "Instrument only the owned historical-source reference fixture" OFF)
foreach(target IN LISTS reference_targets)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE WIN32)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:precise)
        target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O2> $<$<CONFIG:Debug>:/Od>
            $<$<CONFIG:Debug>:/Zi>)
        target_compile_definitions(${target} PRIVATE $<$<CONFIG:Release>:NDEBUG> $<$<CONFIG:Debug>:_DEBUG>)
        if(target STREQUAL "capture_component")
            target_compile_options(${target} PRIVATE /EHs-c-)
        else()
            target_compile_options(${target} PRIVATE /EHsc)
        endif()
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-model=precise -ffp-contract=off
            -Wno-overriding-option $<$<CONFIG:Release>:-O2> $<$<CONFIG:Debug>:-O0> $<$<CONFIG:Debug>:-g>)
        if(target STREQUAL "capture_component")
            target_compile_options(${target} PRIVATE -fno-exceptions -funwind-tables)
        endif()
    endif()
    if(CSE_VOICE_REFERENCE_SANITIZERS)
        if(MSVC)
            message(FATAL_ERROR "Reference sanitizer fixture requires Clang")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
if(WIN32)
    # The actual legacy engine sources retain their C++17 compilation policy.
    set_property(TARGET voice_windows_reference PROPERTY CXX_STANDARD 17)
endif()
