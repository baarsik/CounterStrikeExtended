cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseNativeSymbolAddresses LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_library(symbol_fixture SHARED module.cpp)
target_compile_features(symbol_fixture PRIVATE cxx_std_17)
add_executable(symbol_address_tests symbol_address_tests.cpp)
add_executable(prediction_callback_tests prediction_callback_tests.cpp)
target_compile_features(symbol_address_tests PRIVATE cxx_std_17)
target_compile_features(prediction_callback_tests PRIVATE cxx_std_17)
target_include_directories(symbol_address_tests PRIVATE
  "${CSE_ROOT}/GoldSrc-SDK/public" "${CSE_ROOT}/GoldSrc-SDK/common")
target_link_libraries(symbol_address_tests PRIVATE ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(symbol_address_tests PRIVATE /W4 /WX /EHsc)
  target_compile_options(symbol_fixture PRIVATE /W4 /WX /EHsc)
  target_compile_options(prediction_callback_tests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME symbol-addresses COMMAND symbol_address_tests "$<TARGET_FILE:symbol_fixture>")
add_test(NAME prediction-callbacks COMMAND prediction_callback_tests)
