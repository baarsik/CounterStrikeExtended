cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseBrowserFactoryFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(CSE_TARGET_ARCH STREQUAL "x86")
  message(FATAL_ERROR "The owned browser factory fixture is native64 only")
endif()
enable_testing()
set(CSE_BROWSER_SDL_PROVIDER "" CACHE FILEPATH "Explicit pinned source-built SDL3 runtime")
set(CSE_BROWSER_SDL_IMPORT_ARCHIVE "" CACHE FILEPATH "Matching Windows SDL3 import archive")
if(NOT CSE_BROWSER_SDL_PROVIDER)
  message(FATAL_ERROR "Provide the source-built pinned SDL3 runtime")
endif()
include("${CSE_ROOT}/cmake/BinaryArchitecture.cmake")
cse_verify_binary("${CSE_BROWSER_SDL_PROVIDER}" "${CSE_TARGET_ARCH}")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CSE_ROOT}/tools/sdl3_dependency.py"
  RESULT_VARIABLE header_status OUTPUT_VARIABLE header_identity ERROR_VARIABLE header_error)
if(NOT header_status EQUAL 0)
  message(FATAL_ERROR "Pinned SDL3 headers do not match: ${header_error}")
endif()
add_library(browser_factory_sdl SHARED IMPORTED)
set_target_properties(browser_factory_sdl PROPERTIES IMPORTED_LOCATION "${CSE_BROWSER_SDL_PROVIDER}"
  INTERFACE_INCLUDE_DIRECTORIES "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
if(WIN32)
  cse_verify_binary("${CSE_BROWSER_SDL_IMPORT_ARCHIVE}" "${CSE_TARGET_ARCH}")
  set_property(TARGET browser_factory_sdl PROPERTY IMPORTED_IMPLIB "${CSE_BROWSER_SDL_IMPORT_ARCHIVE}")
endif()
set(CSE_BROWSER_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native")
foreach(target IN ITEMS factory_fixture_host browser_factory_tests)
  if(target STREQUAL "factory_fixture_host")
    add_executable(${target} "${CSE_ROOT}/tests/native/browser_wire/controller_tests.cpp")
  else()
    add_executable(${target} factory_tests.cpp "${CSE_BROWSER_SOURCE}/controller_factory.cpp")
  endif()
  target_sources(${target} PRIVATE "${CSE_BROWSER_SOURCE}/browser_controller.cpp"
    "${CSE_BROWSER_SOURCE}/process_channel.cpp")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CSE_BROWSER_SOURCE}" "${CSE_ROOT}/GoldSrc-SDK/public")
  target_link_libraries(${target} PRIVATE browser_factory_sdl)
  target_compile_definitions(${target} PRIVATE "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
  if(WIN32)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${CSE_BROWSER_SDL_PROVIDER}" "$<TARGET_FILE_DIR:${target}>/SDL3.dll" VERBATIM)
  endif()
endforeach()
target_compile_definitions(browser_factory_tests PRIVATE
  CSE_BROWSER_HOST_RELATIVE_PATH="$<TARGET_FILE_NAME:factory_fixture_host>"
  CSE_BROWSER_DEBUG_PROFILE=$<IF:$<CONFIG:Debug>,1,0>
  CSE_FACTORY_FIXTURE_HOST="$<TARGET_FILE:factory_fixture_host>")
add_dependencies(browser_factory_tests factory_fixture_host)
add_test(NAME native-browser-factory COMMAND browser_factory_tests "${CMAKE_CURRENT_BINARY_DIR}/factory-profile")
set_tests_properties(native-browser-factory PROPERTIES TIMEOUT 45)
