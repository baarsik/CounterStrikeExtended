cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseColorHeaderBindings LANGUAGES CXX)
option(CSE_ENABLE_SANITIZERS "Instrument native color storage and copying" OFF)
enable_testing()
foreach(binding engine vgui2 cstrike)
  add_executable(color_header_${binding} color_headers.cpp)
  target_compile_features(color_header_${binding} PRIVATE cxx_std_17)
  if(binding STREQUAL engine)
    set(first_include "${CSE_ROOT}/src/engine")
  elseif(binding STREQUAL vgui2)
    set(first_include "${CSE_ROOT}/src/vgui2/src/public")
  else()
    set(first_include "${CSE_ROOT}/src/cstrike/public")
  endif()
  if(NOT binding STREQUAL vgui2)
    target_compile_definitions(color_header_${binding} PRIVATE CSE_EXPECT_ENGINE_COLOR)
  endif()
  target_include_directories(color_header_${binding} PRIVATE "${first_include}"
    "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/common"
    "${CSE_ROOT}/src/public" "${CSE_ROOT}/src/vgui2/src/public")
  if(MSVC)
    target_compile_options(color_header_${binding} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(color_header_${binding} PRIVATE -Wall -Wextra -Werror
      -Wno-error=inconsistent-missing-override -Wno-error=overloaded-virtual)
    if(CSE_ENABLE_SANITIZERS)
      target_compile_options(color_header_${binding} PRIVATE -fsanitize=address,undefined
        -fno-sanitize-recover=all -fno-omit-frame-pointer)
      target_link_options(color_header_${binding} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME color-binding-${binding} COMMAND color_header_${binding})
endforeach()
