cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseGameDispatch LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_DISPATCH_SOURCE "${CSE_ROOT}/src/engine/sys_dll.cpp" CACHE FILEPATH "Actual dispatch implementation")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_DISPATCH_SOURCE}" extract_dispatch.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_dispatch.py"
    "${CSE_DISPATCH_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/actual_dispatch.inc" COMMAND_ERROR_IS_FATAL ANY)
option(CSE_DISPATCH_SANITIZERS "Instrument owned modules and actual dispatch" OFF)
foreach(name first second)
    add_library(dispatch_${name} SHARED module.cpp)
    target_compile_definitions(dispatch_${name} PRIVATE "$<$<STREQUAL:${name},first>:CSE_FIRST_MODULE>")
    if(WIN32)
        if(name STREQUAL "second")
            target_sources(dispatch_${name} PRIVATE second.def)
        else()
            target_sources(dispatch_${name} PRIVATE module.def)
        endif()
    endif()
endforeach()
add_executable(game_dispatch_tests dispatch_tests.cpp)
target_include_directories(game_dispatch_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(game_dispatch_tests PRIVATE ${CMAKE_DL_LIBS})
foreach(target dispatch_first dispatch_second game_dispatch_tests)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${CSE_ROOT}/src/engine"
        "${CSE_ROOT}/src/public")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc
            "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/RTC1>")
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
            "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
        if(CSE_DISPATCH_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
enable_testing()
add_test(NAME actual-typed-game-dispatch COMMAND game_dispatch_tests
    "$<TARGET_FILE:dispatch_first>" "$<TARGET_FILE:dispatch_second>")
