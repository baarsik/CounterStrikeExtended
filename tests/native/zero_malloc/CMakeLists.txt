cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseZeroMalloc LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(zero_malloc_tests allocation_tests.cpp)
target_compile_features(zero_malloc_tests PRIVATE cxx_std_14)
target_include_directories(zero_malloc_tests PRIVATE "${CSE_ROOT}/src/engine")
option(CSE_ZERO_MALLOC_SANITIZERS "Instrument the controlled allocator fixture" OFF)
if(MSVC)
    message(FATAL_ERROR "This fixture checks the POSIX assertion boundary; Windows retains its UCRT assertion")
else()
    target_compile_options(zero_malloc_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions -funwind-tables
        "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
    if(CSE_ZERO_MALLOC_SANITIZERS)
        target_compile_options(zero_malloc_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(zero_malloc_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME actual-zero-allocation-boundary
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_allocation.py" "$<TARGET_FILE:zero_malloc_tests>")
set_tests_properties(actual-zero-allocation-boundary PROPERTIES TIMEOUT 10)
