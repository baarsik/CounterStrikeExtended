cmake_minimum_required(VERSION 3.25)
project(cse_network_platform LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CSE_REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_NETWORK_SOURCE "${CSE_REPOSITORY}/GoldSrc-SDK/engine/net_ws.cpp" CACHE FILEPATH "Actual transport source")
set(CSE_NETWORK_COMMON "${CSE_REPOSITORY}/GoldSrc-SDK/common" CACHE PATH "Actual common headers")
option(CSE_NETWORK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
endif()
set(extracted "${CMAKE_CURRENT_BINARY_DIR}/extracted")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_functions.py"
    "${CSE_NETWORK_SOURCE}" "${extracted}" COMMAND_ERROR_IS_FATAL ANY)
get_filename_component(CSE_ENGINE_DIRECTORY "${CSE_NETWORK_SOURCE}" DIRECTORY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_NETWORK_SOURCE}"
    "${CSE_ENGINE_DIRECTORY}/cl_main.cpp" "${CSE_ENGINE_DIRECTORY}/client.h" "${CMAKE_CURRENT_LIST_DIR}/extract_functions.py")
function(network_target target source policy)
    add_executable(${target} "${source}")
    target_include_directories(${target} PRIVATE "${extracted}" "${CSE_NETWORK_COMMON}")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(${target} PRIVATE WIN32 _CRT_SECURE_NO_WARNINGS ${policy})
        if(policy STREQUAL "CSE_ENGINE_WSOCK32")
            target_link_libraries(${target} PRIVATE wsock32)
        else()
            target_link_libraries(${target} PRIVATE ws2_32)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
        if(CSE_NETWORK_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
if(WIN32)
    set(policies CSE_ENGINE_WSOCK32 CSE_ENGINE_WINSOCK2)
else()
    set(policies POSIX)
endif()
foreach(policy IN LISTS policies)
    network_target(network_open_${policy} open_tests.cpp "${policy}")
    add_test(NAME open_${policy} COMMAND network_open_${policy})
    set_tests_properties(open_${policy} PROPERTIES TIMEOUT 10)
    network_target(network_transport_${policy} transport_tests.cpp "${policy}")
    foreach(scenario read-set sleep stagger lengths wire)
        add_test(NAME ${scenario}_${policy} COMMAND network_transport_${policy} ${scenario})
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${scenario}_${policy}")
        set_tests_properties(${scenario}_${policy} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${scenario}_${policy}" TIMEOUT 10)
    endforeach()
    network_target(network_send_${policy} send_tests.cpp "${policy}")
    foreach(scenario host-unreachable network-unreachable diagnostics errors packet resend-host resend-network exhaust-host exhaust-network resend-timing resend-unexpected-paths)
        add_test(NAME send_${scenario}_${policy} COMMAND network_send_${policy} ${scenario})
        set_tests_properties(send_${scenario}_${policy} PROPERTIES TIMEOUT 10)
    endforeach()
    if(NOT WIN32)
        add_test(NAME pipe_POSIX COMMAND network_transport_${policy} pipe)
        set_tests_properties(pipe_POSIX PROPERTIES TIMEOUT 10)
    endif()
endforeach()
if(NOT WIN32)
    network_target(network_loopback_probe loopback_probe.cpp POSIX)
    # Deliberately no CTest entry: socket execution requires explicit fixture authorization.
endif()

if(APPLE)
    network_target(network_send_non_apple send_tests.cpp POSIX)
    target_compile_definitions(network_send_non_apple PRIVATE CSE_NETWORK_NON_APPLE_CONTROL=1)
    foreach(scenario host-unreachable network-unreachable errors packet resend-host resend-network exhaust-host exhaust-network resend-timing resend-unexpected-paths)
        add_test(NAME send_${scenario}_non_apple COMMAND network_send_non_apple ${scenario})
        set_tests_properties(send_${scenario}_non_apple PROPERTIES TIMEOUT 10)
    endforeach()
endif()
