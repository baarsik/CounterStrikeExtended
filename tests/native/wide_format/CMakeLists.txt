cmake_minimum_required(VERSION 3.25)
project(CseWideFormatReference LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(wide_format_reference reference.cpp)
if(MSVC)
    target_compile_options(wide_format_reference PRIVATE /W4 /WX /EHsc)
    target_compile_definitions(wide_format_reference PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(wide_format_reference PRIVATE bcrypt version)
else()
    target_compile_options(wide_format_reference PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
endif()

execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_numeric_thresholds.py"
    --output "${CMAKE_CURRENT_BINARY_DIR}/numeric-thresholds" COMMAND_ERROR_IS_FATAL ANY)
add_executable(wide_numeric_reference numeric_reference.cpp)
target_include_directories(wide_numeric_reference PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/numeric-thresholds")
if(MSVC)
    target_compile_options(wide_numeric_reference PRIVATE /W4 /WX /EHsc)
    target_compile_definitions(wide_numeric_reference PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(wide_numeric_reference PRIVATE bcrypt version)
else()
    target_compile_options(wide_numeric_reference PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
endif()

option(CSE_WIDE_BUILD_HELPER "Build the unwired UTF16 formatter" OFF)
option(CSE_WIDE_SANITIZERS "Instrument helper formatter fixtures" OFF)
option(CSE_WIDE_FAST_FP "Compile the formatter core with inherited fast FP policy" OFF)
if(CSE_WIDE_BUILD_HELPER)
    get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    add_library(legacy_wide_format STATIC "${CSE_ROOT}/GoldSrc-SDK/common/legacy_wide_format.cpp")
    target_include_directories(legacy_wide_format PUBLIC "${CSE_ROOT}/GoldSrc-SDK/common")
    add_executable(wide_format_helper reference.cpp)
    target_compile_definitions(wide_format_helper PRIVATE CSE_WIDE_HELPER)
    target_link_libraries(wide_format_helper PRIVATE legacy_wide_format)
    if(MSVC)
        target_compile_options(legacy_wide_format PRIVATE /W4 /WX /EHs-c-)
        target_compile_definitions(legacy_wide_format PRIVATE _HAS_EXCEPTIONS=0)
        if(CSE_WIDE_FAST_FP)
            target_compile_options(legacy_wide_format PRIVATE /fp:fast)
        else()
            target_compile_options(legacy_wide_format PRIVATE /fp:precise)
        endif()
        target_compile_options(wide_format_helper PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(wide_format_helper PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
        target_link_libraries(wide_format_helper PRIVATE bcrypt version)
    else()
        foreach(target legacy_wide_format wide_format_helper)
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
            if(CSE_WIDE_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
                    -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endforeach()
    endif()
endif()

if(CSE_WIDE_BUILD_HELPER)
    add_executable(wide_numeric_helper numeric_reference.cpp)
    target_include_directories(wide_numeric_helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/numeric-thresholds")
    target_compile_definitions(wide_numeric_helper PRIVATE CSE_WIDE_HELPER)
    target_link_libraries(wide_numeric_helper PRIVATE legacy_wide_format)
    if(MSVC)
        target_compile_options(wide_numeric_helper PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(wide_numeric_helper PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
        target_link_libraries(wide_numeric_helper PRIVATE bcrypt version)
    else()
        target_compile_options(wide_numeric_helper PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
        if(CSE_WIDE_SANITIZERS)
            target_compile_options(wide_numeric_helper PRIVATE -fsanitize=address,undefined
                -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(wide_numeric_helper PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()
if(CSE_WIDE_BUILD_HELPER)
    find_package(Threads REQUIRED)
    add_executable(wide_format_edge_tests edge_tests.cpp)
    target_link_libraries(wide_format_edge_tests PRIVATE legacy_wide_format Threads::Threads)
    if(MSVC)
        target_compile_options(wide_format_edge_tests PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(legacy_wide_format PRIVATE -fno-exceptions)
        if(CSE_WIDE_FAST_FP)
            target_compile_options(legacy_wide_format PRIVATE -ffp-model=fast)
        else()
            target_compile_options(legacy_wide_format PRIVATE -ffp-model=precise)
        endif()
        target_compile_options(wide_format_edge_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
        if(CSE_WIDE_SANITIZERS)
            target_compile_options(wide_format_edge_tests PRIVATE -fsanitize=address,undefined
                -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(wide_format_edge_tests PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()
if(CSE_WIDE_BUILD_HELPER)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_wrappers.py"
        "${CSE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/actual_wrappers.h" COMMAND_ERROR_IS_FATAL ANY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CSE_ROOT}/GoldSrc-SDK/cstrike/game_shared/shared_util.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/extract_wrappers.py")
    add_executable(wide_format_wrapper_tests wrapper_tests.cpp)
    target_include_directories(wide_format_wrapper_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(wide_format_wrapper_tests PRIVATE legacy_wide_format)
    if(MSVC)
        target_compile_options(wide_format_wrapper_tests PRIVATE /W4 /WX /EHsc)
    else()
        target_compile_options(wide_format_wrapper_tests PRIVATE -Wall -Wextra -Werror)
        if(CSE_WIDE_SANITIZERS)
            target_compile_options(wide_format_wrapper_tests PRIVATE -fsanitize=address,undefined
                -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(wide_format_wrapper_tests PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()
