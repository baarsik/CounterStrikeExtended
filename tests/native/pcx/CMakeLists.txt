cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CsePcxFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(CSE_PCX_SOURCE "${CSE_ROOT}/src/engine/screen.cpp" CACHE FILEPATH "Current or preserved screen.cpp")
option(CSE_PCX_REFERENCE "Allow the original implicit length narrowing for a reference comparison" OFF)
option(CSE_PCX_SANITIZERS "Instrument the exact writer and allocation/file hooks" OFF)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/production_bodies.inc"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_sources.py"
    "${CSE_PCX_SOURCE}" "${CSE_ROOT}/src/engine/zone.h" "${CMAKE_CURRENT_BINARY_DIR}"
  DEPENDS "${CSE_PCX_SOURCE}" "${CSE_ROOT}/src/engine/zone.h" extract_sources.py VERBATIM)
add_executable(pcx_tests pcx_tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/production_bodies.inc")
target_compile_features(pcx_tests PRIVATE cxx_std_17)
target_include_directories(pcx_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(pcx_tests PRIVATE "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
if(MSVC)
  target_compile_options(pcx_tests PRIVATE /W4 /WX /EHsc "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  if(CSE_PCX_REFERENCE)
    target_compile_options(pcx_tests PRIVATE /wd4244 /wd4267)
  endif()
else()
  target_compile_options(pcx_tests PRIVATE "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" -Wall -Wextra -Werror -Wshorten-64-to-32)
  if(CSE_PCX_REFERENCE)
    target_compile_options(pcx_tests PRIVATE -Wno-shorten-64-to-32)
  endif()
  if(CSE_PCX_SANITIZERS)
    target_compile_options(pcx_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pcx_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
foreach(case IN ITEMS exact-bytes row-stride word-fields allocation-failure large-count-rejection invalid-dimensions)
  add_test(NAME pcx-${case} COMMAND pcx_tests ${case})
endforeach()
