cmake_minimum_required(VERSION 3.25)
project(fullscreen_refresh LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/src/engine/sys_getmodes.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py"
    "${root}" "${CMAKE_CURRENT_BINARY_DIR}/fullscreen.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(fullscreen_refresh tests.cpp)
target_include_directories(fullscreen_refresh PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${root}/src/external/SDL3/include")
target_compile_features(fullscreen_refresh PRIVATE cxx_std_17)
target_compile_options(fullscreen_refresh PRIVATE -Wall -Wextra -Werror)
option(SANITIZERS "Address and undefined behavior sanitizers" OFF)
if(SANITIZERS)
    target_compile_options(fullscreen_refresh PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fullscreen_refresh PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(case supported retina retina-only wrong-density absent unsupported resolution unavailable rejected windowed fallback)
    add_test(NAME fullscreen_${case} COMMAND fullscreen_refresh ${case})
endforeach()
