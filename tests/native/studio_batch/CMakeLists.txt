cmake_minimum_required(VERSION 3.25)
project(CseStudioBatchTests LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CSE_STUDIO_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_executable(studio_topology_tests topology_tests.cpp)
add_executable(studio_submission_tests submission_tests.cpp)
add_executable(studio_windows_macro_tests windows_macro_tests.cpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_ROOT}/src/engine/r_studio.cpp" "${CSE_ROOT}/src/engine/r_local.h" extract_engine.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_engine.py"
  "${CSE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/engine_attributes_extracted.inc"
  RESULT_VARIABLE extraction_status COMMAND_ERROR_IS_FATAL ANY)
add_executable(studio_engine_attributes engine_attributes.cpp
  "${CSE_ROOT}/src/engine/render/native/legacy_state.cpp")
target_include_directories(studio_engine_attributes PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CSE_ROOT}/src/common")
foreach(target studio_topology_tests studio_submission_tests studio_windows_macro_tests studio_engine_attributes)
target_include_directories(${target} PRIVATE "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/external/SDL3/include")
target_compile_features(${target} PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(CSE_STUDIO_SANITIZERS)
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
endif()
endforeach()
enable_testing()
add_test(NAME studio-indexed-topology COMMAND studio_topology_tests)
add_test(NAME studio-indexed-submission COMMAND studio_submission_tests)
add_test(NAME studio-windows-sdk-macros COMMAND studio_windows_macro_tests)
add_test(NAME studio-engine-attributes COMMAND studio_engine_attributes)
add_test(NAME studio-engine-alignment COMMAND studio_engine_attributes alignment)

option(CSE_STUDIO_GL_PROBE "Build the separately selected real OpenGL parity/benchmark probe" OFF)
set(CSE_STUDIO_SDL_LIBRARY "" CACHE FILEPATH "Matching native SDL3 shared/import library for the GL probe")
if(CSE_STUDIO_GL_PROBE)
  if(NOT EXISTS "${CSE_STUDIO_SDL_LIBRARY}")
    message(FATAL_ERROR "CSE_STUDIO_GL_PROBE requires CSE_STUDIO_SDL_LIBRARY")
  endif()
  find_package(OpenGL REQUIRED)
  add_executable(studio_gl_probe gl_probe.cpp)
  target_compile_features(studio_gl_probe PRIVATE cxx_std_17)
  target_include_directories(studio_gl_probe PRIVATE "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/external/SDL3/include")
  target_link_libraries(studio_gl_probe PRIVATE "${CSE_STUDIO_SDL_LIBRARY}" OpenGL::GL)
  if(MSVC)
    target_compile_options(studio_gl_probe PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(studio_gl_probe PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
  add_test(NAME studio-real-gl-parity COMMAND studio_gl_probe)
  set_tests_properties(studio-real-gl-parity PROPERTIES TIMEOUT 30)
endif()
