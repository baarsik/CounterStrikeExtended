function(add_service_fixture target provider)
  if(target MATCHES "^steam_service_owned_module")
    add_library(${target} SHARED late_global_owner.cpp service_probe.cpp "${CSE_ROOT}/GoldSrc-SDK/common/steam_service.cpp")
    target_compile_definitions(${target} PRIVATE CSE_SERVICE_MODULE_FIXTURE=1)
  else()
    add_executable(${target} late_global_owner.cpp service_probe.cpp "${CSE_ROOT}/GoldSrc-SDK/common/steam_service.cpp")
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  cse_force_steam_initialization(${target})
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
  target_include_directories(${target} SYSTEM PRIVATE "${CSE_ROOT}/GoldSrc-SDK/external/steamworks")
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS CSE_OWNED_STEAM_PROVIDER=1 "CSE_STEAM_PROVIDER_PATH=\"${provider}\"")
  target_link_libraries(${target} PRIVATE "${CMAKE_DL_LIBS}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
endfunction()
