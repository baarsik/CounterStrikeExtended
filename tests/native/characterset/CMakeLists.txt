cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseCharacterSet LANGUAGES CXX)
option(CSE_ENABLE_SANITIZERS "Instrument the owned character-table fixture" OFF)
add_executable(character_set_tests character_set_tests.cpp "${CSE_ROOT}/GoldSrc-SDK/public/characterset.cpp")
target_compile_features(character_set_tests PRIVATE cxx_std_17)
target_include_directories(character_set_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
if(MSVC)
  target_compile_options(character_set_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  target_compile_options(character_set_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  if(CSE_ENABLE_SANITIZERS)
    target_compile_options(character_set_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(character_set_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
target_compile_definitions(character_set_tests PRIVATE "$<$<CONFIG:Release>:NDEBUG>" "$<$<CONFIG:Debug>:_DEBUG>")
enable_testing()
add_test(NAME byte_domain_and_null_inputs COMMAND character_set_tests)
