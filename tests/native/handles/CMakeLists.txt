cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseNativeHandles LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_executable(command_handle_tests command_handle_tests.cpp)
target_compile_features(command_handle_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(command_handle_tests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME native-command-handles COMMAND command_handle_tests)
