cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseNativeFontDiscovery LANGUAGES C CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT APPLE)
  message(FATAL_ERROR "This standalone fixture exercises the macOS CoreText discovery boundary")
endif()
include("${CSE_ROOT}/cmake/NativeThirdParty.cmake")
cse_load_native_third_party()
set(font_source "${CSE_ROOT}/GoldSrc-SDK/vgui2/vgui_surfacelib")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_file(CSE_FONT_FIXTURE_SOURCE NAMES Arial.ttf
  PATHS /System/Library/Fonts/Supplemental NO_DEFAULT_PATH REQUIRED)
set(unicode_fixture "${CMAKE_CURRENT_BINARY_DIR}/unicode-fixture.ttf")
add_custom_command(OUTPUT "${unicode_fixture}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_unicode_fixture.py"
    "${CSE_FONT_FIXTURE_SOURCE}" "${unicode_fixture}"
  DEPENDS make_unicode_fixture.py "${CSE_FONT_FIXTURE_SOURCE}" VERBATIM)
add_custom_target(unicode_font_fixture DEPENDS "${unicode_fixture}")
add_executable(font_discovery_tests font_discovery_tests.cpp "${font_source}/native/MacFontDiscovery.cpp")
add_dependencies(font_discovery_tests unicode_font_fixture)
target_compile_features(font_discovery_tests PRIVATE cxx_std_17)
target_compile_definitions(font_discovery_tests PRIVATE CSE_UNICODE_FONT_FIXTURE="${unicode_fixture}")
target_include_directories(font_discovery_tests PRIVATE "${font_source}")
target_compile_options(font_discovery_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(font_discovery_tests PRIVATE Freetype::Freetype "-framework CoreText" "-framework CoreFoundation")
option(CSE_FONT_SANITIZERS "Run the fixture with address/undefined sanitizers" OFF)
if(CSE_FONT_SANITIZERS)
  target_compile_options(font_discovery_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(font_discovery_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(scenario family style unicode invalid fallback utf16 lifetime)
  add_test(NAME font-discovery-${scenario} COMMAND font_discovery_tests "${scenario}")
endforeach()

add_executable(font_runtime_tests font_runtime_tests.cpp
  "${font_source}/linuxfont.cpp" "${font_source}/FontManager.cpp" "${font_source}/FontAmalgam.cpp"
  "${font_source}/native/MacFontDiscovery.cpp")
add_dependencies(font_runtime_tests unicode_font_fixture)
target_compile_features(font_runtime_tests PRIVATE cxx_std_17)
target_compile_definitions(font_runtime_tests PRIVATE CSE_UNICODE_FONT_FIXTURE="${unicode_fixture}")
target_include_directories(font_runtime_tests PRIVATE "${font_source}"
  "${CSE_ROOT}/GoldSrc-SDK/vgui2/src/public" "${CSE_ROOT}/GoldSrc-SDK/public")
target_link_libraries(font_runtime_tests PRIVATE Freetype::Freetype "-framework CoreText" "-framework CoreFoundation")
if(CSE_FONT_SANITIZERS)
  target_compile_options(font_runtime_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(font_runtime_tests PRIVATE -fsanitize=address,undefined)
endif()
foreach(scenario invalid rendering fallback unicode lifetime)
  add_test(NAME font-runtime-${scenario} COMMAND font_runtime_tests "${scenario}")
endforeach()
