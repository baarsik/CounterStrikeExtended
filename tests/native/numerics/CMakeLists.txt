cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseNumericFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_executable(numeric_tests numeric_tests.cpp)
target_compile_features(numeric_tests PRIVATE cxx_std_17)
target_compile_definitions(numeric_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(numeric_tests PRIVATE /W4 /WX /EHsc /fp:strict
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  target_compile_options(numeric_tests PRIVATE -Wall -Wextra -Werror -ffp-model=strict
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
endif()
add_test(NAME legacy-numeric-conversion COMMAND numeric_tests "${CMAKE_CURRENT_BINARY_DIR}/numeric-results.bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_NUMERIC_SOURCE_DIR "${CSE_ROOT}/GoldSrc-SDK/engine" CACHE PATH "Production or pinned reference source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_NUMERIC_SOURCE_DIR}/common.cpp" "${CSE_NUMERIC_SOURCE_DIR}/cvar.cpp" extract_functions.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_functions.py"
  "${CSE_NUMERIC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}"
  RESULT_VARIABLE extract_status ERROR_VARIABLE extract_error)
if(NOT extract_status EQUAL 0)
  message(FATAL_ERROR "Cannot extract production numeric functions: ${extract_error}")
endif()
add_executable(numeric_engine_tests engine_tests.cpp angle_body.cpp cvar_body.cpp)
target_include_directories(numeric_engine_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(numeric_engine_tests PRIVATE cxx_std_17)
target_compile_definitions(numeric_engine_tests PRIVATE HL25_HW_ENGINE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  set_source_files_properties(engine_tests.cpp angle_body.cpp PROPERTIES COMPILE_OPTIONS /fp:strict)
  set_source_files_properties(cvar_body.cpp PROPERTIES COMPILE_OPTIONS /fp:precise)
  target_compile_options(numeric_engine_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  set_source_files_properties(engine_tests.cpp angle_body.cpp PROPERTIES COMPILE_OPTIONS -ffp-model=strict)
  set_source_files_properties(cvar_body.cpp PROPERTIES COMPILE_OPTIONS "-ffp-model=precise;-ffp-contract=off;-Wno-overriding-option")
  target_compile_options(numeric_engine_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
endif()
add_test(NAME actual-angle-and-cvar COMMAND numeric_engine_tests "${CMAKE_CURRENT_BINARY_DIR}/engine-results.bin")
