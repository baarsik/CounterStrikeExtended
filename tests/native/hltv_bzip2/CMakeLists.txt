cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseHltvBzip2 LANGUAGES C CXX)
set(CSE_HLTV_BZIP2_SOURCE "${CSE_ROOT}/GoldSrc-SDK/HLTV/Core/private/utils/bzip2" CACHE PATH "Retained codec source")
option(CSE_HLTV_BZIP2_SANITIZERS "Instrument retained codec" OFF)
set(codec_sources blocksort.c bz_error.c bzlib.c compress.c crctable.c decompress.c huffman.c randtable.c)
list(TRANSFORM codec_sources PREPEND "${CSE_HLTV_BZIP2_SOURCE}/")
add_library(retained_bzip2 STATIC ${codec_sources})
target_include_directories(retained_bzip2 PUBLIC "${CSE_HLTV_BZIP2_SOURCE}")
add_executable(hltv_bzip2_tests codec_tests.cpp)
target_link_libraries(hltv_bzip2_tests PRIVATE retained_bzip2)
target_compile_features(hltv_bzip2_tests PRIVATE cxx_std_14)
foreach(target retained_bzip2 hltv_bzip2_tests)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
        target_compile_options(${target} PRIVATE /W3 "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
        if(target STREQUAL "hltv_bzip2_tests")
            target_compile_options(${target} PRIVATE /EHsc)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Werror=shorten-64-to-32
            "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
        if(CSE_HLTV_BZIP2_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
enable_testing()
add_test(NAME actual-hltv-bzip2 COMMAND hltv_bzip2_tests "${CMAKE_CURRENT_BINARY_DIR}")
