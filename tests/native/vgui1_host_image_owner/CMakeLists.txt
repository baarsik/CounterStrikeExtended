cmake_minimum_required(VERSION 3.25)
project(CseVgui1HostImageOwner LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_OWNER_SOURCE_ROOT "${CSE_ROOT}" CACHE PATH "Frozen production source root")
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(owner_source "${CSE_OWNER_SOURCE_ROOT}/GoldSrc-SDK/launcher/native/VGui1HostImageOwner.cpp")
foreach(mode native injected)
    add_library(owner_${mode} STATIC)
    if(EXISTS "${owner_source}")
        target_sources(owner_${mode} PRIVATE "${owner_source}")
    else()
        target_sources(owner_${mode} PRIVATE missing.cpp)
    endif()
    add_executable(host_${mode} host.cpp allocation_guard.cpp)
    target_link_libraries(host_${mode} PRIVATE owner_${mode})
    if(mode STREQUAL "injected")
        target_compile_definitions(owner_${mode} PRIVATE dlclose=CseFixtureDlclose)
        target_compile_definitions(host_${mode} PRIVATE CSE_RELEASE_INJECTION=1)
        target_sources(host_${mode} PRIVATE release_injection.cpp)
    endif()
endforeach()
add_library(root_fixture SHARED root.cpp)
add_library(foreign_fixture SHARED root.cpp)
add_library(proxy_fixture SHARED proxy.cpp)
target_link_libraries(proxy_fixture PRIVATE root_fixture)
add_library(misplaced_owner SHARED misplaced.cpp)
add_library(benign_insertion SHARED benign_insertion.cpp)
if(EXISTS "${owner_source}")
    target_sources(misplaced_owner PRIVATE "${owner_source}")
endif()
option(CSE_OWNER_SANITIZERS "Instrument owned host and real dylibs" OFF)
option(CSE_OWNER_TSAN "Check invalid worker callbacks against real owner mutations" OFF)
if(CSE_OWNER_SANITIZERS AND CSE_OWNER_TSAN)
    message(FATAL_ERROR "Run TSan separately from ASan/UBSan")
endif()
if(CSE_OWNER_TSAN)
    target_compile_definitions(host_native PRIVATE CSE_TSAN_SHORT_BURST)
endif()
foreach(target owner_native owner_injected host_native host_injected root_fixture foreign_fixture proxy_fixture
        misplaced_owner benign_insertion)
    target_include_directories(${target} PRIVATE
        "${CSE_OWNER_SOURCE_ROOT}/GoldSrc-SDK/launcher/native" "${CSE_ROOT}/GoldSrc-SDK/public")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti
        -fno-fast-math -ffp-contract=off -fvisibility=hidden -fvisibility-inlines-hidden)
    if(CSE_OWNER_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
    if(CSE_OWNER_TSAN)
        target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=thread)
    endif()
endforeach()
enable_testing()
add_test(NAME real-native-owner COMMAND host_native "$<TARGET_FILE:root_fixture>"
    "$<TARGET_FILE:foreign_fixture>" "$<TARGET_FILE:misplaced_owner>" "$<TARGET_FILE:proxy_fixture>" 1)
add_test(NAME injected-release-controls COMMAND host_injected "$<TARGET_FILE:root_fixture>"
    "$<TARGET_FILE:foreign_fixture>" "$<TARGET_FILE:misplaced_owner>" "$<TARGET_FILE:proxy_fixture>" 1)
add_test(NAME actual-reload-cycles COMMAND host_native "$<TARGET_FILE:root_fixture>"
    "$<TARGET_FILE:foreign_fixture>" "$<TARGET_FILE:misplaced_owner>" "$<TARGET_FILE:proxy_fixture>" 4)
add_test(NAME inserted-native-owner COMMAND "${CMAKE_COMMAND}" -E env
    "DYLD_INSERT_LIBRARIES=$<TARGET_FILE:benign_insertion>" CSE_OWNER_EXPECT_INSERTION=1
    "$<TARGET_FILE:host_native>" "$<TARGET_FILE:root_fixture>"
    "$<TARGET_FILE:foreign_fixture>" "$<TARGET_FILE:misplaced_owner>" "$<TARGET_FILE:proxy_fixture>" 1)
