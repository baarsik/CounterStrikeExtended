cmake_minimum_required(VERSION 3.25)
project(CseWideFormatCallers LANGUAGES C CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_CALLER_INPUT_ROOT "${CSE_ROOT}" CACHE PATH "Read-only production input for RED/current extraction")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_callers.py"
    --root "${CSE_CALLER_INPUT_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/extracted"
    COMMAND_ERROR_IS_FATAL ANY)
add_library(wide_core STATIC "${CSE_ROOT}/GoldSrc-SDK/common/legacy_wide_format.cpp")
target_include_directories(wide_core PUBLIC "${CSE_ROOT}/GoldSrc-SDK/common")
add_executable(chat_controls chat_controls.cpp)
target_include_directories(chat_controls PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/extracted")
target_link_libraries(chat_controls PRIVATE wide_core)
add_executable(alias_controls alias_controls.cpp)
target_include_directories(alias_controls PRIVATE "${CSE_CALLER_INPUT_ROOT}/GoldSrc-SDK/public")
target_link_libraries(alias_controls PRIVATE wide_core)
add_library(c_prefix OBJECT c_controls.c)
target_include_directories(c_prefix PRIVATE "${CSE_CALLER_INPUT_ROOT}/GoldSrc-SDK/public")
option(CSE_CALLER_SANITIZERS "Instrument owned caller fixtures" OFF)
foreach(target wide_core chat_controls alias_controls c_prefix)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(CSE_CALLER_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
target_compile_options(wide_core PRIVATE -fno-exceptions)
target_compile_options(alias_controls PRIVATE -fno-exceptions)
enable_testing()
add_test(NAME chat-hud-wrapper-callers COMMAND chat_controls)
add_test(NAME direct-va-prefix-aliases COMMAND alias_controls)
set_tests_properties(chat-hud-wrapper-callers direct-va-prefix-aliases PROPERTIES TIMEOUT 15)
