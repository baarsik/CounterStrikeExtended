cmake_minimum_required(VERSION 3.25)
project(CseVgui1NativeFontProbe LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "This compile-only preparation targets the macOS font boundary")
endif()
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CSE_FREETYPE_PREFIX "" CACHE PATH "Exact pinned source-built FreeType prefix")
set(VCPKG_IMPORT_PREFIX "${CSE_FREETYPE_PREFIX}")
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG TRUE)
list(PREPEND CMAKE_PREFIX_PATH "${CSE_FREETYPE_PREFIX}")
find_package(ZLIB CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/zlib" NO_DEFAULT_PATH)
if(NOT TARGET ZLIB::ZLIB)
    add_library(ZLIB::ZLIB ALIAS ZLIB::ZLIBSTATIC)
endif()
find_library(BZIP2_LIBRARY_RELEASE NAMES bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(BZIP2_LIBRARY_DEBUG NAMES bz2d bzip2d libbz2d libbzip2d bz2 bzip2 libbz2 libbzip2
    PATHS "${CSE_FREETYPE_PREFIX}/debug/lib" NO_DEFAULT_PATH REQUIRED)
find_path(BZIP2_INCLUDE_DIR NAMES bzlib.h PATHS "${CSE_FREETYPE_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_package(BZip2 REQUIRED)
find_package(PNG CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/png" NO_DEFAULT_PATH)
find_package(freetype CONFIG REQUIRED PATHS "${CSE_FREETYPE_PREFIX}/share/freetype" NO_DEFAULT_PATH)
set(discovery "${CSE_ROOT}/src/vgui2/vgui_surfacelib/native")
set(raster "${CSE_ROOT}/src/utils/vgui/lib-src/vgui/posix/native")
include("${CSE_ROOT}/cmake/MacFontService.cmake")
cse_define_mac_font_registration_service("${discovery}/MacFontRegistration.cpp")
add_library(probe_discovery STATIC "${discovery}/MacFontDiscovery.cpp")
target_include_directories(probe_discovery PUBLIC "${discovery}")
target_link_libraries(probe_discovery PUBLIC Freetype::Freetype "-framework CoreText" "-framework CoreFoundation")
add_library(probe_legacy_raster STATIC "${raster}/VGui1LegacyRaster.cpp")
target_include_directories(probe_legacy_raster PUBLIC "${raster}")
target_link_libraries(probe_legacy_raster PUBLIC Freetype::Freetype)
target_compile_options(probe_legacy_raster PRIVATE -fno-fast-math -ffp-contract=off -ffp-model=strict)
add_executable(vgui1_native_font_probe probe.cpp)
target_include_directories(vgui1_native_font_probe PRIVATE "${CSE_ROOT}/tests/native/vgui1_legacy_raster")
target_link_libraries(vgui1_native_font_probe PRIVATE probe_discovery probe_legacy_raster
    cse_mac_font_registration)
option(CSE_PROBE_SANITIZERS "Compile the owned probe and services with ASan/UBSan" OFF)
foreach(target probe_discovery probe_legacy_raster vgui1_native_font_probe cse_mac_font_registration)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(NOT target STREQUAL "cse_mac_font_registration")
        target_compile_options(${target} PRIVATE -fno-exceptions)
    endif()
    if(CSE_PROBE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
endforeach()
# Deliberately no add_test, discovery, font-file search, or post-build execution.
