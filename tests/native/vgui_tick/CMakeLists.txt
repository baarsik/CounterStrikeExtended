cmake_minimum_required(VERSION 3.25)
project(CseVguiTick LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(vgui_source "${CSE_ROOT}/src/utils/vgui/lib-src/vgui")
option(CSE_VGUI_TICK_SANITIZERS "Instrument the actual VGUI App scheduler" OFF)
option(CSE_VGUI_TICK_LEGACY_RED "Allow the original LP64 narrowing to expose behavioral RED" OFF)
add_library(vgui_tick_public_layout OBJECT public_layout.cpp)
target_include_directories(vgui_tick_public_layout PRIVATE "${CSE_ROOT}/src/utils/vgui/include")
add_executable(vgui_tick_tests tick_tests.cpp controlled_bindings.cpp "${vgui_source}/App.cpp"
  $<TARGET_OBJECTS:vgui_tick_public_layout>)
target_include_directories(vgui_tick_tests PRIVATE "${vgui_source}/include")
foreach(target vgui_tick_tests vgui_tick_public_layout)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas
    -Werror=shorten-64-to-32 -Werror=pointer-to-int-cast)
  if(CSE_VGUI_TICK_LEGACY_RED)
    target_compile_options(${target} PRIVATE -Wno-error=shorten-64-to-32)
  endif()
  if(CSE_VGUI_TICK_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endforeach()
if(CSE_VGUI_TICK_SANITIZERS)
  target_link_options(vgui_tick_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(scenario ordinary completion-clock initial-sign signed-wrap full-wrap negative-clock
    multiple-wraps negative-interval maximum-interval minimum-interval long-limits layout)
  add_test(NAME vgui-tick-${scenario} COMMAND vgui_tick_tests "${scenario}")
  if(CSE_VGUI_TICK_SANITIZERS)
    set_tests_properties(vgui-tick-${scenario} PROPERTIES
      ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0:halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
  endif()
endforeach()
