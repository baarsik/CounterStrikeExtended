cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseSoundTransferFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
set(CSE_SOUND_TRANSFER_IMPLEMENTATION "${CSE_ROOT}/GoldSrc-SDK/engine/sound_transfer_impl.h" CACHE FILEPATH
  "Production transfer methods; a pinned archived implementation may be selected for reference evidence")
configure_file(transfer_header.h.in transfer_header.h @ONLY)
option(CSE_TEST_NATIVE_SOUND_TRANSFER "Test the native SDL lock boundary as well as the legacy reference" ON)
set(transfer_modes legacy)
if(CSE_TEST_NATIVE_SOUND_TRANSFER)
  list(APPEND transfer_modes native)
endif()
enable_testing()
foreach(mode IN LISTS transfer_modes)
  if(mode STREQUAL legacy)
    set(target sound_transfer_tests)
    set(use_sdl 0)
  else()
    set(target sound_transfer_native_tests)
    set(use_sdl 1)
  endif()
  add_executable(${target} transfer_tests.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE CSE_SOUND_USE_SDL=${use_sdl})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    # The engine's snd_mix.cpp uses precise FP, MinSpace in Release and Disabled in Debug.
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /fp:precise
      "$<$<CONFIG:Release>:/O1>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-model=precise -ffp-contract=off
      -Wno-overriding-option "$<$<CONFIG:Release>:-Os>" "$<$<CONFIG:Debug>:-O0>")
  endif()
  add_test(NAME sound-transfer-${mode} COMMAND ${target} "${CMAKE_CURRENT_BINARY_DIR}/transferred-${mode}.pcm")
endforeach()
add_executable(ring_samples_tests ring_samples_tests.cpp)
target_compile_features(ring_samples_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ring_samples_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(ring_samples_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native-ring-samples COMMAND ring_samples_tests)
add_executable(legacy_ring_tests "${CSE_ROOT}/tests/sdl3/audio_ring_tests.cpp")
target_compile_features(legacy_ring_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(legacy_ring_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(legacy_ring_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME retained-ring-contract COMMAND legacy_ring_tests)
set(CSE_SOUND_SDL_PROVIDER "" CACHE FILEPATH "Explicit pinned SDL3 provider for separately authorized fixtures")
set(CSE_SOUND_SDL_IMPORT_ARCHIVE "" CACHE FILEPATH "The selected Windows provider's native import archive")
if(CSE_SOUND_SDL_PROVIDER)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  include("${CSE_ROOT}/cmake/BinaryArchitecture.cmake")
  cse_verify_binary("${CSE_SOUND_SDL_PROVIDER}" "${CSE_TARGET_ARCH}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CSE_ROOT}/tools/sdl3_dependency.py"
    RESULT_VARIABLE header_status OUTPUT_VARIABLE header_identity ERROR_VARIABLE header_error)
  if(NOT header_status EQUAL 0)
    message(FATAL_ERROR "Pinned SDL3 header validation failed: ${header_error}")
  endif()
  add_library(sound_sdl_provider SHARED IMPORTED)
  set_target_properties(sound_sdl_provider PROPERTIES IMPORTED_LOCATION "${CSE_SOUND_SDL_PROVIDER}"
    INTERFACE_INCLUDE_DIRECTORIES "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
  if(WIN32)
    cse_verify_binary("${CSE_SOUND_SDL_IMPORT_ARCHIVE}" "${CSE_TARGET_ARCH}")
    set_property(TARGET sound_sdl_provider PROPERTY IMPORTED_IMPLIB "${CSE_SOUND_SDL_IMPORT_ARCHIVE}")
  endif()
  find_package(Threads REQUIRED)
  foreach(target sound_provider_tests retained_stream_tests dma_provider_tests)
    if(target STREQUAL sound_provider_tests)
      set(source provider_tests.cpp)
    elseif(target STREQUAL dma_provider_tests)
      set(source dma_provider_tests.cpp)
    else()
      set(source "${CSE_ROOT}/tests/sdl3/audio_stream_tests.cpp")
    endif()
    add_executable(${target} "${source}")
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(target STREQUAL dma_provider_tests)
      target_sources(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine/snd_sdl.cpp")
      target_compile_definitions(${target} PRIVATE CSE_SOUND_USE_SDL=1
        "$<$<PLATFORM_ID:Windows>:WIN32>" DLL_EXPORT=)
      target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine"
        "${CSE_ROOT}/GoldSrc-SDK/common" "${CSE_ROOT}/GoldSrc-SDK/public")
    endif()
    target_link_libraries(${target} PRIVATE sound_sdl_provider Threads::Threads)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /EHsc /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "$<TARGET_FILE:sound_sdl_provider>" "$<TARGET_FILE_DIR:${target}>")
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
endif()
