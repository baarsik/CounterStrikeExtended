cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseNetworkWorkerFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Threads REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
enable_testing()
set(CSE_NETWORK_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/net_ws.cpp" CACHE FILEPATH
  "Production or pinned reference network source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_NETWORK_SOURCE}" extract_functions.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_functions.py"
  "${CSE_NETWORK_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/actual_network.h"
  RESULT_VARIABLE extract_status ERROR_VARIABLE extract_error)
if(NOT extract_status EQUAL 0)
  message(FATAL_ERROR "Cannot extract production network functions: ${extract_error}")
endif()
set(CSE_NETWORK_WORKER_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/network_worker.cpp" CACHE FILEPATH
  "Production worker or a temporary mutation for fixture sensitivity checks")
set(CSE_NETWORK_SANITIZER "" CACHE STRING "Optional fixture sanitizer: address or thread")
set_property(CACHE CSE_NETWORK_SANITIZER PROPERTY STRINGS "" address thread)
set(worker_targets network_worker_tests)
if(WIN32)
  list(APPEND worker_targets network_worker_wsock32_tests)
endif()
foreach(target IN LISTS worker_targets)
  add_executable(${target} worker_tests.cpp "${CSE_NETWORK_WORKER_SOURCE}")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${CSE_ROOT}/GoldSrc-SDK/engine")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
    if(target STREQUAL network_worker_wsock32_tests)
      target_link_libraries(${target} PRIVATE wsock32)
    else()
      target_link_libraries(${target} PRIVATE ws2_32)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
  if(CSE_NETWORK_SANITIZER STREQUAL address)
    if(MSVC)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    else()
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  elseif(CSE_NETWORK_SANITIZER STREQUAL thread AND NOT MSVC)
    target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fsanitize=thread)
  elseif(CSE_NETWORK_SANITIZER)
    message(FATAL_ERROR "Unsupported network fixture sanitizer for this compiler: ${CSE_NETWORK_SANITIZER}")
  endif()
  add_test(NAME ${target} COMMAND ${target})
  set_tests_properties(${target} PROPERTIES TIMEOUT 30)
endforeach()
