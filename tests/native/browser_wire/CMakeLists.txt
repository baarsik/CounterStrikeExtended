cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseBrowserWireFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_executable(browser_wire_tests wire_tests.cpp)
target_compile_features(browser_wire_tests PRIVATE cxx_std_17)
target_include_directories(browser_wire_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native")
if(MSVC)
  target_compile_options(browser_wire_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(browser_wire_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native-browser-wire COMMAND browser_wire_tests)
add_executable(browser_key_tests key_tests.cpp
  "${CSE_ROOT}/GoldSrc-SDK/vgui2/vgui_surfacelib/vgui_key_translation.cpp")
target_compile_features(browser_key_tests PRIVATE cxx_std_17)
target_include_directories(browser_key_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public"
  "${CSE_ROOT}/GoldSrc-SDK/vgui2/vgui_surfacelib" "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
if(MSVC)
  target_compile_options(browser_key_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(browser_key_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME native-browser-key-namespace COMMAND browser_key_tests)
set(CSE_BROWSER_SDL_PROVIDER "" CACHE FILEPATH "Explicit pinned source-built SDL3 runtime")
set(CSE_BROWSER_SDL_IMPORT_ARCHIVE "" CACHE FILEPATH "Matching Windows SDL3 import archive")
if(CSE_BROWSER_SDL_PROVIDER)
  include("${CSE_ROOT}/cmake/BinaryArchitecture.cmake")
  cse_verify_binary("${CSE_BROWSER_SDL_PROVIDER}" "${CSE_TARGET_ARCH}")
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CSE_ROOT}/tools/sdl3_dependency.py"
    RESULT_VARIABLE header_status OUTPUT_VARIABLE header_identity ERROR_VARIABLE header_error)
  if(NOT header_status EQUAL 0)
    message(FATAL_ERROR "Pinned SDL3 header validation failed: ${header_error}")
  endif()
  add_library(browser_sdl SHARED IMPORTED)
  set_target_properties(browser_sdl PROPERTIES IMPORTED_LOCATION "${CSE_BROWSER_SDL_PROVIDER}"
    INTERFACE_INCLUDE_DIRECTORIES "${CSE_ROOT}/GoldSrc-SDK/external/SDL3/include")
  if(WIN32)
    cse_verify_binary("${CSE_BROWSER_SDL_IMPORT_ARCHIVE}" "${CSE_TARGET_ARCH}")
    set_property(TARGET browser_sdl PROPERTY IMPORTED_IMPLIB "${CSE_BROWSER_SDL_IMPORT_ARCHIVE}")
  endif()
  add_executable(browser_process_tests process_tests.cpp
    "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/process_channel.cpp")
  target_compile_features(browser_process_tests PRIVATE cxx_std_17)
  target_include_directories(browser_process_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native")
  target_link_libraries(browser_process_tests PRIVATE browser_sdl)
  if(MSVC)
    target_compile_options(browser_process_tests PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(browser_process_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  if(WIN32)
    add_custom_command(TARGET browser_process_tests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${CSE_BROWSER_SDL_PROVIDER}" "$<TARGET_FILE_DIR:browser_process_tests>/SDL3.dll" VERBATIM)
  endif()
  add_test(NAME native-browser-process COMMAND browser_process_tests)
  set_tests_properties(native-browser-process PROPERTIES TIMEOUT 30)
endif()
if(CSE_BROWSER_SDL_PROVIDER AND NOT CSE_TARGET_ARCH STREQUAL "x86")
  add_executable(browser_controller_tests controller_tests.cpp
    "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/browser_controller.cpp"
    "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/process_channel.cpp")
  target_compile_features(browser_controller_tests PRIVATE cxx_std_17)
  target_include_directories(browser_controller_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native"
    "${CSE_ROOT}/GoldSrc-SDK/public")
  target_link_libraries(browser_controller_tests PRIVATE browser_sdl)
  if(MSVC)
    target_compile_options(browser_controller_tests PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(browser_controller_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  add_dependencies(browser_controller_tests browser_process_tests)
  add_test(NAME native-browser-controller COMMAND browser_controller_tests "${CMAKE_CURRENT_BINARY_DIR}/controller-profile")
  set_tests_properties(native-browser-controller PROPERTIES TIMEOUT 45)
endif()
if(TARGET browser_controller_tests)
  add_executable(browser_cef_probe cef_controller_probe.cpp
    "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/browser_controller.cpp"
    "${CSE_ROOT}/GoldSrc-SDK/engine/html/native/process_channel.cpp")
  target_compile_features(browser_cef_probe PRIVATE cxx_std_17)
  target_include_directories(browser_cef_probe PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native"
    "${CSE_ROOT}/GoldSrc-SDK/public")
  target_link_libraries(browser_cef_probe PRIVATE browser_sdl)
  add_dependencies(browser_cef_probe browser_process_tests)
  if(MSVC)
    target_compile_options(browser_cef_probe PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(browser_cef_probe PRIVATE -Wall -Wextra -Werror)
  endif()
endif()
