cmake_minimum_required(VERSION 3.25)
project(cse_socket_lifetime LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OSX_ARCHITECTURES arm64)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(EXTRACTED "${CMAKE_CURRENT_BINARY_DIR}/extracted")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tests/native/network_platform/extract_functions.py"
    "${ROOT}/src/engine/net_ws.cpp" "${EXTRACTED}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/src/engine/net_ws.cpp")
add_library(socket_provider SHARED provider.cpp "${ROOT}/src/engine/native_socket_lifetime.cpp")
target_include_directories(socket_provider PRIVATE "${EXTRACTED}" "${ROOT}/tests/native/network_platform"
    "${ROOT}/src/common" "${ROOT}/src/engine")
target_compile_options(socket_provider PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
add_executable(socket_inheritance inheritance.cpp "${ROOT}/src/launcher/native/SocketImageOwner.cpp")
target_link_options(socket_inheritance PRIVATE -Wl,-export_dynamic)
target_include_directories(socket_inheritance PRIVATE "${ROOT}/src/launcher/native")
target_include_directories(socket_inheritance PRIVATE "${EXTRACTED}" "${ROOT}/tests/native/network_platform"
    "${ROOT}/src/common")
target_compile_options(socket_inheritance PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
foreach(mode fork exec)
    add_test(NAME inheritance_${mode} COMMAND socket_inheritance ${mode} $<TARGET_FILE:socket_provider>)
    set_tests_properties(inheritance_${mode} PROPERTIES TIMEOUT 10)
endforeach()
add_library(socket_instrumented_host OBJECT "${ROOT}/src/launcher/native/SocketImageOwner.cpp")
target_compile_options(socket_instrumented_host PRIVATE -Wall -Wextra -Werror -finstrument-functions)
add_executable(socket_contract contract.cpp $<TARGET_OBJECTS:socket_instrumented_host>)
target_include_directories(socket_contract PRIVATE "${ROOT}/src/launcher/native")
target_compile_options(socket_contract PRIVATE -Wall -Wextra -Werror)
target_link_options(socket_contract PRIVATE -Wl,-export_dynamic)
add_test(NAME canonical_roles_and_lifecycle COMMAND socket_contract $<TARGET_FILE:socket_provider>)
set_tests_properties(canonical_roles_and_lifecycle PROPERTIES TIMEOUT 15)

add_library(socket_controlled_host OBJECT "${ROOT}/src/launcher/native/SocketImageOwner.cpp")
target_compile_options(socket_controlled_host PRIVATE -Wall -Wextra -Werror
    -include "${CMAKE_CURRENT_SOURCE_DIR}/syscall_redirect.h")
add_executable(socket_failures failures.cpp $<TARGET_OBJECTS:socket_controlled_host>)
target_include_directories(socket_failures PRIVATE "${ROOT}/src/launcher/native")
target_compile_options(socket_failures PRIVATE -Wall -Wextra -Werror)
target_link_options(socket_failures PRIVATE -Wl,-export_dynamic)
foreach(mode failures consumed races)
    add_test(NAME owner_${mode} COMMAND socket_failures ${mode} $<TARGET_FILE:socket_provider> $<TARGET_FILE:socket_shadow>)
    set_tests_properties(owner_${mode} PROPERTIES TIMEOUT 15)
endforeach()
add_library(socket_shadow SHARED shadow.cpp)
target_include_directories(socket_shadow PRIVATE "${ROOT}/src/public")
add_library(socket_hidden_host OBJECT "${ROOT}/src/launcher/native/SocketImageOwner.cpp")
target_compile_options(socket_hidden_host PRIVATE -Wall -Wextra -Werror -finstrument-functions)
target_compile_definitions(socket_hidden_host PRIVATE CseNativeSocketHostV1=CseTestHiddenSocketHost)
foreach(kind RANGE 0 3)
    add_executable(socket_table_${kind} table_rejection.cpp $<TARGET_OBJECTS:socket_hidden_host>)
    target_include_directories(socket_table_${kind} PRIVATE "${ROOT}/src/launcher/native")
    target_compile_definitions(socket_table_${kind} PRIVATE CSE_BAD_TABLE=${kind})
    target_compile_options(socket_table_${kind} PRIVATE -Wall -Wextra -Werror)
    target_link_options(socket_table_${kind} PRIVATE -Wl,-export_dynamic)
    add_test(NAME canonical_table_${kind} COMMAND socket_table_${kind}
        $<TARGET_FILE:socket_provider> $<TARGET_FILE:socket_shadow>)
endforeach()

add_executable(socket_network_state network_state.cpp "${ROOT}/src/launcher/native/SocketImageOwner.cpp")
target_include_directories(socket_network_state PRIVATE "${ROOT}/src/launcher/native")
target_compile_options(socket_network_state PRIVATE -Wall -Wextra -Werror)
target_link_options(socket_network_state PRIVATE -Wl,-export_dynamic)
add_test(NAME actual_network_terminal_state COMMAND socket_network_state $<TARGET_FILE:socket_provider>)
