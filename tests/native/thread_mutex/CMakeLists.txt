cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseThreadMutexFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Threads REQUIRED)
set(CSE_THREADTOOLS_HEADER "${CSE_ROOT}/src/public/tier0/threadtools.h" CACHE FILEPATH
  "Actual production header or an explicitly archived negative reference")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_THREADTOOLS_HEADER}")
add_executable(thread_mutex_tests mutex_tests.cpp)
target_compile_features(thread_mutex_tests PRIVATE cxx_std_14)
target_include_directories(thread_mutex_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(thread_mutex_tests PRIVATE Threads::Threads)
if(WIN32)
  configure_file(selected_header.h.in selected_header.h @ONLY)
  target_sources(thread_mutex_tests PRIVATE windows_bindings.cpp)
  target_include_directories(thread_mutex_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/win_shims"
    "${CSE_ROOT}/src/public")
  target_compile_definitions(thread_mutex_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  set(fixture_kind windows)
else()
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS extract_mutex_members.py)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_mutex_members.py"
    "${CSE_THREADTOOLS_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/actual_mutex_members.inc"
    RESULT_VARIABLE extraction_status ERROR_VARIABLE extraction_error)
  if(NOT extraction_status EQUAL 0)
    message(FATAL_ERROR "Cannot extract the production fast mutex methods: ${extraction_error}")
  endif()
  target_sources(thread_mutex_tests PRIVATE storage_model_bindings.cpp)
  target_include_directories(thread_mutex_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  set(fixture_kind storage-model)
  message(STATUS "Portable LLP64 storage/method-body fixture; this is not a Windows header/ABI qualification")
endif()
if(MSVC)
  target_compile_options(thread_mutex_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  target_compile_options(thread_mutex_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
endif()
option(CSE_THREAD_MUTEX_SANITIZERS "ASan/UBSan for portable storage-body fixtures" OFF)
if(CSE_THREAD_MUTEX_SANITIZERS AND NOT MSVC)
  target_compile_options(thread_mutex_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
    -fno-omit-frame-pointer -g)
  target_link_options(thread_mutex_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(scenario layout recursion operand-isolation high-owner-id contended-try unlock alignment)
  add_test(NAME "mutex-${fixture_kind}-${scenario}" COMMAND thread_mutex_tests "${scenario}")
endforeach()
if(WIN32)
  add_test(NAME mutex-windows-contention COMMAND thread_mutex_tests contention)
endif()
get_directory_property(mutex_test_names TESTS)
set_tests_properties(${mutex_test_names} PROPERTIES TIMEOUT 30)
