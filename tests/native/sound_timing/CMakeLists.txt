cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseSoundTiming LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_SOUND_TIMING_SOURCE "${CSE_ROOT}/src/engine/snd_dma.cpp" CACHE FILEPATH "Actual timing implementation")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_SOUND_TIMING_SOURCE}" extract_timing.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_timing.py"
    "${CSE_SOUND_TIMING_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/actual_timing.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(sound_timing_tests timing_tests.cpp)
target_compile_features(sound_timing_tests PRIVATE cxx_std_14)
target_include_directories(sound_timing_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
option(CSE_SOUND_TIMING_SANITIZERS "Instrument actual timing methods" OFF)
if(MSVC)
    target_compile_options(sound_timing_tests PRIVATE /W4 /WX /EHsc /fp:precise
        "$<$<CONFIG:Release>:/O1>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/RTC1>")
else()
    target_compile_options(sound_timing_tests PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off
        "$<$<CONFIG:Release>:-Os>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
    if(CSE_SOUND_TIMING_SANITIZERS)
        target_compile_options(sound_timing_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(sound_timing_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME actual-sound-timing COMMAND sound_timing_tests)
