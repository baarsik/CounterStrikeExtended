cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseTier0ThreadIdFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT APPLE)
  message(FATAL_ERROR "This fixture executes the macOS logical thread-ID boundary")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CSE_THREAD_ID_DATABASE "" CACHE FILEPATH "Actual generated configuration compile_commands.json")
if(NOT EXISTS "${CSE_THREAD_ID_DATABASE}")
  message(FATAL_ERROR "The generated compiler database is required")
endif()
option(CSE_THREAD_ID_SANITIZERS "Instrument both actual TUs and fixtures" OFF)
set(sanitizers)
if(CSE_THREAD_ID_SANITIZERS)
  set(sanitizers --sanitizers)
endif()
function(thread_fixture target source)
  set(object "${CMAKE_CURRENT_BINARY_DIR}/${target}-thread.o")
  set(controlled)
  if(target STREQUAL thread_id_failure_tests)
    set(controlled --controlled-id-provider)
  endif()
  add_custom_command(OUTPUT "${object}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py"
      --database "${CSE_THREAD_ID_DATABASE}" --sdk "${CSE_ROOT}/GoldSrc-SDK" --output "${object}"
      ${sanitizers} ${controlled}
    DEPENDS "${CSE_ROOT}/GoldSrc-SDK/tier0/thread.cpp" "${CSE_ROOT}/GoldSrc-SDK/public/tier0/platform.h"
      "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py" "${CSE_THREAD_ID_DATABASE}"
    DEPFILE "${object}.d" VERBATIM)
  set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  add_executable(${target} "${source}" "${object}")
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/public")
  target_compile_definitions(${target} PRIVATE TIER0_DLL POSIX OSX)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(CSE_THREAD_ID_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
      -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
thread_fixture(thread_id_tests thread_id_tests.cpp)
thread_fixture(thread_id_failure_tests failure_tests.cpp)
enable_testing()
foreach(scenario initial register concurrent primary lifecycle rights)
  add_test(NAME tier0-thread-id-${scenario} COMMAND thread_id_tests ${scenario})
  set_tests_properties(tier0-thread-id-${scenario} PROPERTIES TIMEOUT 20)
endforeach()
add_test(NAME tier0-thread-id-controlled-failure COMMAND thread_id_failure_tests)
set_tests_properties(tier0-thread-id-controlled-failure PROPERTIES TIMEOUT 20)
