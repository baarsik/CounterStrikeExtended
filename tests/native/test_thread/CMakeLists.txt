cmake_minimum_required(VERSION 3.25)
project(CseTestThreadFixtures LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CSE_TEST_THREAD_DATABASE "" CACHE FILEPATH "Actual owning tier0 compile_commands.json")
set(CSE_TEST_THREAD_SANITIZER "" CACHE STRING "Empty, address or thread")
if(NOT APPLE OR NOT EXISTS "${CSE_TEST_THREAD_DATABASE}")
    message(FATAL_ERROR "The actual macOS owning compiler database is required")
endif()
set(object "${CMAKE_CURRENT_BINARY_DIR}/actual-testthread.o")
set(sanitizer)
if(CSE_TEST_THREAD_SANITIZER)
    list(APPEND sanitizer --sanitizers "${CSE_TEST_THREAD_SANITIZER}")
endif()
add_custom_command(OUTPUT "${object}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py"
        --database "${CSE_TEST_THREAD_DATABASE}" --output "${object}" --controlled-create ${sanitizer}
    DEPENDS "${CSE_ROOT}/GoldSrc-SDK/tier0/testthread.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/compile_tu.py"
        "${CSE_TEST_THREAD_DATABASE}" DEPFILE "${object}.d" VERBATIM)
set_source_files_properties("${object}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
add_executable(test_thread_harness harness_tests.cpp "${object}")
set_target_properties(test_thread_harness PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(test_thread_harness PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
target_link_libraries(test_thread_harness PRIVATE Threads::Threads)
if(CSE_TEST_THREAD_SANITIZER)
    if(CSE_TEST_THREAD_SANITIZER STREQUAL "address")
        set(flags address,undefined)
    else()
        set(flags thread)
    endif()
    target_compile_options(test_thread_harness PRIVATE -fsanitize=${flags} -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(test_thread_harness PRIVATE -fsanitize=${flags})
endif()
foreach(scenario initial lifecycle stop failure invalid queries destruction tls spurious)
    add_test(NAME test-thread-${scenario} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_case.py"
        "$<TARGET_FILE:test_thread_harness>" "${scenario}" "${CMAKE_CURRENT_BINARY_DIR}/${scenario}")
    set_tests_properties(test-thread-${scenario} PROPERTIES TIMEOUT 12)
endforeach()
