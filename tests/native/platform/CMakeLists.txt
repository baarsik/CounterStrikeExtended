cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CsePlatformBoundaryTests LANGUAGES C CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
foreach(mode enabled ndebug)
  add_executable(utl_assert_${mode} utl_assert_tests.cpp)
  target_compile_features(utl_assert_${mode} PRIVATE cxx_std_17)
  target_include_directories(utl_assert_${mode} PRIVATE "${CSE_ROOT}/src/engine")
  target_compile_definitions(utl_assert_${mode} PRIVATE UTL_ENGINE_ASSERTS)
  if(mode STREQUAL ndebug)
    target_compile_definitions(utl_assert_${mode} PRIVATE NDEBUG)
  elseif(MSVC)
    target_compile_options(utl_assert_${mode} PRIVATE /UNDEBUG)
  else()
    target_compile_options(utl_assert_${mode} PRIVATE -UNDEBUG)
  endif()
  if(MSVC)
    target_compile_options(utl_assert_${mode} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(utl_assert_${mode} PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME utl-assert-${mode} COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/verify_assert.py" "$<TARGET_FILE:utl_assert_${mode}>")
endforeach()
add_executable(directory_tests directory_tests.cpp)
target_compile_features(directory_tests PRIVATE cxx_std_17)
add_test(NAME directory-lifecycle COMMAND directory_tests "${CMAKE_CURRENT_BINARY_DIR}/owned-directory-fixture")
if(NOT WIN32)
  add_executable(perf_counter_tests perf_counter_tests.cpp)
  target_compile_features(perf_counter_tests PRIVATE cxx_std_17)
  add_test(NAME native-clock-epoch COMMAND perf_counter_tests)
endif()
add_executable(crt_tests crt_tests.cpp)
target_compile_features(crt_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_definitions(crt_tests PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
endif()
add_test(NAME legacy-crt-text COMMAND crt_tests)

add_library(crt_c_boundary OBJECT crt_c_boundary.c)
if(MSVC)
  target_compile_definitions(crt_c_boundary PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
  target_compile_options(crt_c_boundary PRIVATE /W4 /WX)
else()
  target_compile_options(crt_c_boundary PRIVATE -Wall -Wextra -Werror)
endif()
