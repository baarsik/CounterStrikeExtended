cmake_minimum_required(VERSION 3.25)
project(CseRectClipping LANGUAGES C CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RECT_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/vgui_EngineSurfaceHW.cpp" CACHE FILEPATH "Real scalar source import")
set(RECT_GOLDEN "" CACHE FILEPATH "Pinned actual Windows reference JSONL")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_bodies.py"
  --source "${RECT_SOURCE}" --output "${CMAKE_CURRENT_BINARY_DIR}/imports" RESULT_VARIABLE extraction)
if(NOT extraction EQUAL 0)
  message(FATAL_ERROR "Actual scalar source extraction failed")
endif()
add_executable(rect_rows reference_rows.cpp "${CMAKE_CURRENT_BINARY_DIR}/imports/imported_surface.cpp"
  "${CMAKE_CURRENT_BINARY_DIR}/imports/imported_screen.cpp")
add_executable(rect_types type_arithmetic_tests.cpp)
add_library(rect_c_declaration OBJECT c_declaration_probe.c)
foreach(target rect_rows rect_types rect_c_declaration)
  target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/common" "${CSE_ROOT}/GoldSrc-SDK/engine")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
endforeach()
set_target_properties(rect_rows rect_types PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
set_target_properties(rect_c_declaration PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
option(RECT_SANITIZERS "Instrument owned pure scalar fixture" OFF)
if(RECT_SANITIZERS)
  foreach(target rect_rows rect_types)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
  endforeach()
endif()
enable_testing()
add_test(NAME record-and-arithmetic COMMAND rect_types)
add_test(NAME actual-bodies-vs-windows COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/compare_rows.py"
  "$<TARGET_FILE:rect_rows>" "${RECT_GOLDEN}" "${CMAKE_CURRENT_BINARY_DIR}/actual.jsonl")
