cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CSE_SDK_DIR "${CSE_ROOT}/src")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseSdlMouseBridgeProvider LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The provider bridge is native64 only")
endif()
option(CSE_BRIDGE_FIXTURE_LOCAL_SEAMS "Add explicitly fixture-only local lifecycle seams" OFF)
add_subdirectory("${CSE_ROOT}/cmake/dependencies/SDL3" "${CMAKE_BINARY_DIR}/dependencies/SDL3")
if(CSE_BRIDGE_FIXTURE_LOCAL_SEAMS)
  target_sources(SDL3-shared PRIVATE "${CMAKE_CURRENT_LIST_DIR}/fixture_local_seams.c")
  # Distinct identity prevents loader coalescing from hiding the real A/B dispatch test.
  set_target_properties(SDL3-shared PROPERTIES OUTPUT_NAME SDL3_cse_fixture)
endif()
add_executable(bridge_provider_probe provider_probe.cpp)
target_compile_features(bridge_provider_probe PRIVATE cxx_std_14)
target_include_directories(bridge_provider_probe PRIVATE "${CSE_SDK_DIR}/common")
target_link_libraries(bridge_provider_probe PRIVATE SDL3::Headers "${CMAKE_DL_LIBS}")
target_compile_definitions(bridge_provider_probe PRIVATE SDL_MAIN_HANDLED=1)
add_executable(bridge_import_probe import_probe.c)
target_include_directories(bridge_import_probe PRIVATE "${CSE_SDK_DIR}/common")
target_link_libraries(bridge_import_probe PRIVATE SDL3::SDL3-shared)
if(MSVC)
  target_compile_options(bridge_provider_probe PRIVATE /W4 /WX /EHsc)
  target_compile_options(bridge_import_probe PRIVATE /W4 /WX)
else()
  target_compile_options(bridge_provider_probe PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  target_compile_options(bridge_import_probe PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/provider-path-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:SDL3-shared>\n")
