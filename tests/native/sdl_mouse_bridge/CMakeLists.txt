cmake_minimum_required(VERSION 3.25)
project(CseSdlMouseBridgeFixture LANGUAGES C CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CSE_FIXTURE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT IS_DIRECTORY "${CSE_SDL_SOURCE_DIR}/src/events")
  message(FATAL_ERROR "Supply the existing pinned SDL3.4.20 source directory")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CSE_BRIDGE_SANITIZERS "Instrument controlled services and bridge" OFF)
set(CSE_BRIDGE_TEST_SOURCE "${CSE_FIXTURE_ROOT}/src/common/sdl3_native_mouse_bridge.c"
  CACHE FILEPATH "Controlled fixture bridge source; deliberate variants only")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/config")
configure_file("${CSE_SDL_SOURCE_DIR}/include/build_config/SDL_build_config_minimal.h"
  "${CMAKE_CURRENT_BINARY_DIR}/config/SDL_build_config.h" COPYONLY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_sdk.py"
  "${CSE_SDL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/imported" COMMAND_ERROR_IS_FATAL ANY)
add_library(bridge_fixture SHARED sdk_fixture.c naive_bridge.c)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_sources(bridge_fixture PRIVATE "${CSE_BRIDGE_TEST_SOURCE}")
  target_include_directories(bridge_fixture PRIVATE "${CSE_FIXTURE_ROOT}/src/common")
  add_executable(bridge_tests controlled_tests.cpp)
  target_include_directories(bridge_tests PRIVATE "${CSE_FIXTURE_ROOT}/src/common")
  target_link_libraries(bridge_tests PRIVATE bridge_fixture)
endif()
target_include_directories(bridge_fixture PRIVATE
  "${CMAKE_CURRENT_BINARY_DIR}/config" "${CMAKE_CURRENT_BINARY_DIR}/imported"
  "${CSE_SDL_SOURCE_DIR}/include" "${CSE_SDL_SOURCE_DIR}/src")
target_compile_definitions(bridge_fixture PRIVATE SDL_STATIC_LIB=1 SDL_MAIN_HANDLED=1 SDL_ASSERT_LEVEL=0)
add_executable(bridge_naive_red naive_test.cpp)
target_link_libraries(bridge_naive_red PRIVATE bridge_fixture)
add_executable(bridge_header_probe header_probe.cpp)
target_include_directories(bridge_header_probe PRIVATE "${CSE_FIXTURE_ROOT}/src/common")
set(fixture_targets bridge_fixture bridge_naive_red bridge_header_probe)
if(TARGET bridge_tests)
  list(APPEND fixture_targets bridge_tests)
endif()
foreach(target IN LISTS fixture_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /fp:precise
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-model=precise
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
  if(CSE_BRIDGE_SANITIZERS AND NOT MSVC)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
  endif()
endforeach()
