cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseDependencyPolicyFixtures LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CSE_SDK_DIR "${CSE_ROOT}/src")
set(CSE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
add_subdirectory("${CSE_ROOT}/cmake/dependencies/SDL3" dependencies/SDL3)
add_subdirectory("${CSE_ROOT}/cmake/dependencies/mpg123" dependencies/mpg123)
add_executable(audio_conversion_probe audio_conversion_probe.cpp)
target_link_libraries(audio_conversion_probe PRIVATE SDL3::Headers "${CMAKE_DL_LIBS}")
add_executable(decoder_fixture "${CSE_ROOT}/tests/music/decoder_tests.cpp"
  "${CSE_SDK_DIR}/engine/mpg123_decoder.cpp")
target_link_libraries(decoder_fixture PRIVATE libmpg123 SDL3::SDL3-shared)
add_executable(audio_stream_fixture "${CSE_ROOT}/tests/sdl3/audio_stream_tests.cpp")
target_link_libraries(audio_stream_fixture PRIVATE SDL3::SDL3-shared)
foreach(fixture audio_conversion_probe decoder_fixture audio_stream_fixture)
  target_compile_features(${fixture} PRIVATE cxx_std_17)
  set_target_properties(${fixture} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${fixture} PRIVATE /W4 /WX /EHsc /fp:precise /UNDEBUG
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(${fixture} PRIVATE -Wall -Wextra -Werror -ffp-contract=off -UNDEBUG
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/sdl-provider.txt" CONTENT "$<TARGET_FILE:SDL3-shared>")
# Provider execution is explicit; configuring/building never runs these fixtures.
