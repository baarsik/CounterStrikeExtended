cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseGameLibraryNames LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_GAME_LOADER_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/sys_dll.cpp" CACHE FILEPATH "Actual loader implementation")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_GAME_LOADER_SOURCE}" extract_loader.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_loader.py"
    "${CSE_GAME_LOADER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/actual_loader.inc" COMMAND_ERROR_IS_FATAL ANY)
option(CSE_GAME_LIBRARY_SANITIZERS "Instrument bounded native library name adaptation" OFF)
add_executable(game_library_name_tests name_tests.cpp)
set(test_targets game_library_name_tests)
if(NOT WIN32)
    foreach(platform macos linux)
        add_executable(game_library_loader_${platform} loader_tests.cpp)
        target_include_directories(game_library_loader_${platform} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        if(platform STREQUAL "linux")
            target_compile_definitions(game_library_loader_${platform} PRIVATE CSE_TEST_LINUX)
        endif()
        list(APPEND test_targets game_library_loader_${platform})
    endforeach()
endif()
foreach(target IN LISTS test_targets)
target_compile_features(${target} PRIVATE cxx_std_14)
target_include_directories(${target} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/engine")
if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc
        "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/RTC1>")
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function -Wshorten-64-to-32
        "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
    if(CSE_GAME_LIBRARY_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endif()
endforeach()
enable_testing()
add_test(NAME native-game-library-names COMMAND game_library_name_tests)
if(NOT WIN32)
    add_test(NAME actual-macos-game-library-loader COMMAND game_library_loader_macos)
    add_test(NAME actual-linux-game-library-loader COMMAND game_library_loader_linux)
endif()
