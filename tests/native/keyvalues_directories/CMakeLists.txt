cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseKeyValuesDirectories LANGUAGES CXX)
add_executable(keyvalues_directory_tests directory_tests.cpp)
target_compile_features(keyvalues_directory_tests PRIVATE cxx_std_14)
target_include_directories(keyvalues_directory_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/common")
option(CSE_KEYVALUES_DIRECTORY_SANITIZERS "Instrument native parent path selection" OFF)
if(MSVC)
    target_compile_options(keyvalues_directory_tests PRIVATE /W4 /WX /EHsc
        "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
    target_compile_options(keyvalues_directory_tests PRIVATE -Wall -Wextra -Werror
        "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
    if(CSE_KEYVALUES_DIRECTORY_SANITIZERS)
        target_compile_options(keyvalues_directory_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(keyvalues_directory_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME native-keyvalues-parent-routing COMMAND keyvalues_directory_tests)
