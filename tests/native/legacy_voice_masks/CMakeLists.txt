cmake_minimum_required(VERSION 3.25)
project(CseLegacyVoiceMasks LANGUAGES CXX)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_VOICE_MASK_SOURCE "${repo}/src/cl_dll/voice_status.cpp" CACHE FILEPATH "Actual voice source or copied control")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_methods.py"
    --repo "${repo}" --output "${CMAKE_CURRENT_BINARY_DIR}/imports" --voice-source "${CSE_VOICE_MASK_SOURCE}"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(voice_masks mask_tests.cpp "${repo}/src/common/parsemsg.cpp")
target_include_directories(voice_masks PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/imports")
foreach(folder cl_dll dlls public common pm_shared engine utils/vgui/include game_shared external)
    target_include_directories(voice_masks PRIVATE "${repo}/src/${folder}")
endforeach()
target_compile_definitions(voice_masks PRIVATE CLIENT_DLL CLIENT_WEAPONS HL_DLL)
set_target_properties(voice_masks PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
option(CSE_VOICE_MASK_SANITIZERS "Instrument only authored data and controlled services" OFF)
if(MSVC)
    target_compile_options(voice_masks PRIVATE /W3 /EHsc /fp:precise)
    target_compile_options(voice_masks PRIVATE "/FI${CMAKE_CURRENT_LIST_DIR}/visibility_neutral_fixture.h")
    target_compile_definitions(voice_masks PRIVATE _CRT_SECURE_NO_WARNINGS)
    set_property(TARGET voice_masks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(voice_masks PRIVATE -include "${repo}/src/public/cse_posix_crt.h"
        -Werror=shorten-64-to-32 -Werror=pointer-to-int-cast -ffp-model=precise -ffp-contract=off)
endif()
if(CSE_VOICE_MASK_SANITIZERS)
    target_compile_options(voice_masks PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(voice_masks PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME legacy-voice-mask-data COMMAND voice_masks)
