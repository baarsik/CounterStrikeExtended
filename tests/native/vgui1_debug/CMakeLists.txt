cmake_minimum_required(VERSION 3.25)
project(cse_vgui1_debug_fixture LANGUAGES CXX)
if(NOT APPLE)
    message(FATAL_ERROR "The owned POSIX/native profile fixture currently targets Apple ARM64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CSE_VGUI1_DEBUG_SANITIZERS "Instrument the component fixture" OFF)
option(CSE_VGUI_COLOR_DEPRECATED_COPY_ERROR "Require clean native copy declarations" ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(vgui "${repo}/GoldSrc-SDK/utils/vgui/lib-src/vgui")
set(public "${repo}/GoldSrc-SDK/utils/vgui/include")
function(fixture_flags target)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
    if(CSE_VGUI1_DEBUG_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
add_executable(vgui1_debug_native debug_driver.cpp)
target_compile_definitions(vgui1_debug_native PRIVATE CSE_VGUI1_DEBUG_NATIVE)
target_include_directories(vgui1_debug_native PRIVATE "${vgui}/include")
fixture_flags(vgui1_debug_native)
add_executable(vgui1_debug_reference debug_driver.cpp windows_transport.cpp "${vgui}/win32/Debug.cpp")
target_include_directories(vgui1_debug_reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/shims" "${vgui}/include")
fixture_flags(vgui1_debug_reference)
foreach(context IN ITEMS public private)
    add_executable(vgui1_color_${context} color_tests.cpp color_bindings.cpp "${vgui}/Color.cpp")
    if(context STREQUAL "public")
        target_include_directories(vgui1_color_${context} PRIVATE "${public}" "${vgui}/include")
    else()
        target_include_directories(vgui1_color_${context} PRIVATE "${vgui}/include" "${public}")
    endif()
    fixture_flags(vgui1_color_${context})
    target_compile_options(vgui1_color_${context} PRIVATE -Wdeprecated-copy)
    if(NOT CSE_VGUI_COLOR_DEPRECATED_COPY_ERROR)
        target_compile_options(vgui1_color_${context} PRIVATE -Wno-error=deprecated-copy)
    endif()
endforeach()
enable_testing()
add_test(NAME vgui1_debug_owned_pipes COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_debug.py"
    --native $<TARGET_FILE:vgui1_debug_native> --reference $<TARGET_FILE:vgui1_debug_reference>
    --evidence "${CMAKE_CURRENT_BINARY_DIR}/pipe-evidence")
add_test(NAME vgui1_color_public COMMAND vgui1_color_public)
add_test(NAME vgui1_color_private COMMAND vgui1_color_private)
set_tests_properties(vgui1_debug_owned_pipes PROPERTIES TIMEOUT 60)
set_tests_properties(vgui1_color_public vgui1_color_private PROPERTIES TIMEOUT 30)
