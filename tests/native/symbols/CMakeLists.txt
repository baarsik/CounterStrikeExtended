cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseNativeSymbols LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
enable_testing()
add_executable(pe_export_tests pe_export_tests.cpp)
target_compile_features(pe_export_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(pe_export_tests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME pe-export-names COMMAND pe_export_tests)
if(WIN32)
  add_executable(pe_export_file_tests pe_export_file_tests.cpp)
  target_compile_features(pe_export_file_tests PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(pe_export_file_tests PRIVATE /W4 /WX /EHsc)
  endif()
  add_test(NAME pe-export-file COMMAND pe_export_file_tests)
endif()
