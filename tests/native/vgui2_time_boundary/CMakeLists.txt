cmake_minimum_required(VERSION 3.25)
project(CseVgui2TimeBoundary LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(sdk "${CSE_ROOT}/GoldSrc-SDK")
enable_testing()
foreach(order canonical-first cs-first)
  foreach(entry engine lookup standalone cs-wrapper)
    set(target "vgui2_time_${order}_${entry}")
    add_executable(${target} boundary_tests.cpp fake_producer.cpp)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    string(REPLACE "-" "_" entry_macro "${entry}")
    target_compile_definitions(${target} PRIVATE "CSE_TIME_ENTRY_${entry_macro}=1")
    if(order STREQUAL "canonical-first")
      target_include_directories(${target} PRIVATE "${sdk}/vgui2/src/public" "${sdk}/cstrike/public")
    else()
      target_include_directories(${target} PRIVATE "${sdk}/cstrike/public" "${sdk}/vgui2/src/public")
    endif()
    target_include_directories(${target} PRIVATE "${sdk}" "${sdk}/public")
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas
        -Werror=shorten-64-to-32 -Werror=pointer-to-int-cast)
    endif()
    add_test(NAME "${target}" COMMAND "${target}")
  endforeach()
endforeach()
