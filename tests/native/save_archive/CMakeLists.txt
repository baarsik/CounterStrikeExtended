cmake_minimum_required(VERSION 3.25)
project(CseSaveArchive LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(CSE_ARCHIVE_OLD "Reproduce pinned original owner" OFF)
option(CSE_ARCHIVE_SANITIZE "Enable scoped address/undefined sanitizers" OFF)
set(CSE_ARCHIVE_OLD_SOURCE "${CSE_ROOT}/build/save-archive-20261111/old-host_cmd.cpp" CACHE FILEPATH "Pinned old source")
set(owner "${CSE_ROOT}/GoldSrc-SDK/engine/host_cmd.cpp")
set(old_arg)
if(CSE_ARCHIVE_OLD)
    set(owner "${CSE_ARCHIVE_OLD_SOURCE}")
    set(old_arg --old)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py" "${owner}"
    "${CMAKE_CURRENT_BINARY_DIR}/owner.h" ${old_arg} COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${owner}" extract.py)
foreach(width 260 1024)
    set(target archive_${width})
    add_executable(${target} archive_tests.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE MAX_PATH=${width})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CSE_ROOT}/GoldSrc-SDK/engine")
    if(CSE_ARCHIVE_OLD)
        target_compile_definitions(${target} PRIVATE ARCHIVE_OLD=1)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(CSE_ARCHIVE_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    if(CSE_ARCHIVE_OLD)
        foreach(mode writer reader short)
            add_test(NAME old-${width}-${mode} COMMAND ${target} ${mode})
        endforeach()
    else()
        add_test(NAME archive-${width} COMMAND ${target})
    endif()
endforeach()
