cmake_minimum_required(VERSION 3.25)
project(CseVgui1AppPlatform LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(vgui_source "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/lib-src/vgui")
set(CSE_APP_DECLARATION "${CSE_ROOT}/GoldSrc-SDK/utils/vgui/include/VGUI_App.h" CACHE FILEPATH "Actual App declaration")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/declaring-header")
configure_file("${CSE_APP_DECLARATION}" "${CMAKE_CURRENT_BINARY_DIR}/declaring-header/VGUI_App.h" COPYONLY)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_APP_PLATFORM_SOURCE "${vgui_source}/posix/App.cpp" CACHE FILEPATH "Actual platform source or tagged negative control")
set(CSE_APP_ADAPTER_SOURCE "${vgui_source}/posix/VGui1AppPlatform.cpp" CACHE FILEPATH "Actual adapter or tagged negative control")
option(CSE_APP_SANITIZERS "Instrument fake-service component" OFF)
add_executable(app_component component_tests.cpp controlled_dependencies.cpp controlled_surface.cpp actual_declaration_probe.cpp
  "${vgui_source}/App.cpp" "${CSE_APP_PLATFORM_SOURCE}" "${CSE_APP_ADAPTER_SOURCE}")
target_include_directories(app_component PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/declaring-header" "${vgui_source}/include" "${vgui_source}/posix")
set_target_properties(app_component PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_definitions(app_component PRIVATE VGUI_EXPORTS)
  target_compile_options(app_component PRIVATE "/FI${CMAKE_CURRENT_LIST_DIR}/visibility_neutral_fixture.h")
  target_compile_options(app_component PRIVATE /W4 /WX /EHsc /fp:precise)
  set_property(TARGET app_component PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(app_component PRIVATE -Wall -Wextra -Werror)
endif()
if(CSE_APP_SANITIZERS)
  target_compile_options(app_component PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(app_component PRIVATE -fsanitize=address,undefined)
endif()
if(APPLE)
  add_library(real_clock_compile_only OBJECT "${vgui_source}/posix/VGui1AppClock.cpp")
  target_include_directories(real_clock_compile_only PRIVATE "${vgui_source}/posix")
  set_target_properties(real_clock_compile_only PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_compile_options(real_clock_compile_only PRIVATE -Wall -Wextra -Werror)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_win_reference.py"
  "${vgui_source}/win32/App.cpp" "${CMAKE_CURRENT_BINARY_DIR}/win_reference_methods.inc"
  COMMAND_ERROR_IS_FATAL ANY)
add_executable(win_reference win_reference_tests.cpp)
target_include_directories(win_reference PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(win_reference PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
  target_compile_options(win_reference PRIVATE /W4 /WX /EHsc /fp:precise)
  set_property(TARGET win_reference PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(win_reference PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas)
endif()
if(CSE_APP_SANITIZERS)
  target_compile_options(win_reference PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(win_reference PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME controlled-windows-reference COMMAND win_reference)
add_test(NAME app-component COMMAND app_component)
add_test(NAME app-failure-children COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/failure_children.py" "$<TARGET_FILE:app_component>")
