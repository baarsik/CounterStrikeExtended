cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseNativeWebMFixture LANGUAGES C CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
include("${CSE_ROOT}/cmake/NativeThirdParty.cmake")
cse_load_native_third_party()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_WEBM_SOURCE "${CSE_ROOT}/src/engine/webm_video.cpp" CACHE FILEPATH "Actual current or preserved WebM source")
add_executable(webm_tests webm_tests.cpp)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_sources(webm_tests PRIVATE "${CSE_ROOT}/src/engine/webm_pcm_allocation.cpp")
endif()
target_compile_features(webm_tests PRIVATE cxx_std_17)
target_include_directories(webm_tests PRIVATE "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/common"
  "${CSE_ROOT}/src/public" "${CSE_ROOT}/src/pm_shared" "${CSE_ROOT}/src/dlls"
  "${CSE_ROOT}/src/external/SDL3/include")
target_include_directories(webm_tests SYSTEM PRIVATE "${CSE_NATIVE_THIRD_PARTY_PREFIX}/include/webm")
target_compile_definitions(webm_tests PRIVATE SW_RENDERER CSE_WEBM_SOURCE="${CSE_WEBM_SOURCE}"
  "$<$<PLATFORM_ID:Windows>:WIN32>" DLL_EXPORT=)
target_link_libraries(webm_tests PRIVATE unofficial::libwebm::libwebm unofficial::libvpx::libvpx Vorbis::vorbis Ogg::ogg)
set(webm_provider_inputs "")
if(WIN32)
  set(CSE_WEBM_SDL3_IMPLIB "" CACHE FILEPATH "Pinned matching CPU/config source-built SDL3 import library")
  set(CSE_WEBM_SDL3_RUNTIME "" CACHE FILEPATH "Pinned matching CPU/config source-built SDL3 runtime DLL")
  if(NOT EXISTS "${CSE_WEBM_SDL3_IMPLIB}" OR NOT EXISTS "${CSE_WEBM_SDL3_RUNTIME}")
    message(FATAL_ERROR "Windows whole-TU fixture requires CSE_WEBM_SDL3_IMPLIB and CSE_WEBM_SDL3_RUNTIME")
  endif()
  cse_verify_binary("${CSE_WEBM_SDL3_IMPLIB}" "${CSE_TARGET_ARCH}")
  cse_verify_binary("${CSE_WEBM_SDL3_RUNTIME}" "${CSE_TARGET_ARCH}")
  add_library(cse_webm_sdl3 SHARED IMPORTED)
  set_target_properties(cse_webm_sdl3 PROPERTIES IMPORTED_IMPLIB "${CSE_WEBM_SDL3_IMPLIB}"
    IMPORTED_LOCATION "${CSE_WEBM_SDL3_RUNTIME}")
  target_link_libraries(webm_tests PRIVATE cse_webm_sdl3)
  # Link the actual converter needed by the otherwise unused software player.
  # Decoder tests do not exercise this converter or claim RGB presentation.
  target_sources(webm_tests PRIVATE "${CSE_ROOT}/src/engine/yuv2rgb.cpp" unused_presentation_bindings.cpp)
  add_custom_command(TARGET webm_tests POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "$<TARGET_FILE:cse_webm_sdl3>" "$<TARGET_FILE_DIR:webm_tests>/SDL3.dll" VERBATIM)
  set(webm_provider_inputs
    "sdl3_import=$<TARGET_LINKER_FILE:cse_webm_sdl3>\nsdl3_runtime=$<TARGET_FILE:cse_webm_sdl3>\n")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selected-$<CONFIG>.txt" CONTENT
  "libwebm=$<TARGET_FILE:unofficial::libwebm::libwebm>\nlibvpx=$<TARGET_FILE:unofficial::libvpx::libvpx>\nlibvorbis=$<TARGET_FILE:Vorbis::vorbis>\nlibogg=$<TARGET_FILE:Ogg::ogg>\n${webm_provider_inputs}")
if(MSVC)
  target_compile_options(webm_tests PRIVATE /W4 /WX /EHsc /Gy)
  target_link_options(webm_tests PRIVATE /OPT:REF)
else()
  target_compile_definitions(webm_tests PRIVATE OSX POSIX)
  target_compile_options(webm_tests PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
  target_link_options(webm_tests PRIVATE -Wl,-dead_strip)
endif()
enable_testing()
add_test(NAME webm-layout COMMAND webm_tests)
foreach(codec vp8 vp9)
  add_test(NAME webm-${codec}-decode COMMAND webm_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${codec}")
  add_test(NAME webm-${codec}-threads COMMAND webm_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${codec}" threads)
  add_test(NAME webm-${codec}-lifetime COMMAND webm_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${codec}" lifetime)
  add_test(NAME webm-${codec}-select COMMAND webm_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${codec}" select)
endforeach()
add_test(NAME webm-invalid-container COMMAND webm_tests "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/truncated.webm" invalid)
add_test(NAME webm-known-fixture-hashes COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_fixtures.py")
if(WIN32)
  add_test(NAME webm-unexercised-presentation-guards COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check_presentation_guards.py" "$<TARGET_FILE:webm_tests>")
endif()
add_executable(webm_fixture_generator EXCLUDE_FROM_ALL generate_fixture.cpp)
target_compile_features(webm_fixture_generator PRIVATE cxx_std_17)
target_include_directories(webm_fixture_generator SYSTEM PRIVATE "${CSE_NATIVE_THIRD_PARTY_PREFIX}/include/webm")
target_link_libraries(webm_fixture_generator PRIVATE unofficial::libwebm::libwebm unofficial::libvpx::libvpx
  Vorbis::vorbisenc Ogg::ogg)
option(CSE_WEBM_SANITIZERS "Instrument production WebM wrapper and fixtures with ASan/UBSan" OFF)
foreach(target webm_tests webm_fixture_generator)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS
    "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
  if(CSE_WEBM_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
option(CSE_WEBM_LEAK_CHECKS "Run legal decoder lifetimes through Apple's leaks tool" OFF)
if(CSE_WEBM_LEAK_CHECKS)
  if(NOT APPLE OR CSE_WEBM_SANITIZERS)
    message(FATAL_ERROR "WebM leaks fixtures require an unsanitized macOS build")
  endif()
  find_program(CSE_LEAKS_EXECUTABLE leaks REQUIRED)
  set(CSE_WEBM_RECEIPT_DIR "${CMAKE_CURRENT_BINARY_DIR}/receipts" CACHE PATH "WebM ownership receipts")
  foreach(codec vp8 vp9)
    add_test(NAME webm-leaks-${codec} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_leaks.py"
      "${CSE_WEBM_RECEIPT_DIR}/leaks-${codec}.log" "${CSE_LEAKS_EXECUTABLE}" "$<TARGET_FILE:webm_tests>"
      "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${codec}" lifetime)
  endforeach()
endif()
