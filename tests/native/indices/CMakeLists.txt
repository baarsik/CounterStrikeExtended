cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseNativeIndexFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
add_executable(native_index_tests index_tests.cpp)
target_compile_features(native_index_tests PRIVATE cxx_std_17)
target_compile_definitions(native_index_tests PRIVATE "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
if(MSVC)
  target_compile_options(native_index_tests PRIVATE /W4 /WX /EHsc "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
  target_compile_options(native_index_tests PRIVATE "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME native-array-indices COMMAND native_index_tests)
