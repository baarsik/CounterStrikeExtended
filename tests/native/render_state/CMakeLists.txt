cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseRenderStateFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
set(native "${CSE_ROOT}/src/engine/render/native")
add_library(render_state STATIC "${native}/legacy_state.cpp" "${native}/legacy_commands.cpp"
  "${native}/legacy_texture.cpp" "${native}/frame_stream.cpp")
set_target_properties(render_state PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_include_directories(render_state PUBLIC "${native}")
add_executable(render_state_tests state_tests.cpp)
target_link_libraries(render_state_tests PRIVATE render_state)
add_executable(render_texture_tests texture_tests.cpp)
target_link_libraries(render_texture_tests PRIVATE render_state)
add_executable(render_frame_tests frame_tests.cpp)
target_link_libraries(render_frame_tests PRIVATE render_state)
target_compile_definitions(render_state PRIVATE CSE_CAPTURE_FAULT_INJECTION=1)
target_compile_definitions(render_frame_tests PRIVATE CSE_CAPTURE_FAULT_INJECTION=1)
option(CSE_RENDER_GL_ORACLE "Build actual SDL/OpenGL oracle" ON)
option(CSE_RENDER_SANITIZERS "Instrument CPU layer and fixtures" OFF)
if(CSE_RENDER_SANITIZERS AND MSVC)
  message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
endif()
set(targets render_state render_state_tests render_texture_tests render_frame_tests)
if(CSE_RENDER_GL_ORACLE)
  set(CSE_RENDER_SDL_LIBRARY "" CACHE FILEPATH "Pinned matching-config SDL3 shared/import library")
  set(CSE_RENDER_SDL_RUNTIME "" CACHE FILEPATH "Pinned matching-config SDL3 runtime")
  if(NOT EXISTS "${CSE_RENDER_SDL_LIBRARY}" OR NOT EXISTS "${CSE_RENDER_SDL_RUNTIME}")
    message(FATAL_ERROR "Actual GL oracle requires CSE_RENDER_SDL_LIBRARY and CSE_RENDER_SDL_RUNTIME")
  endif()
  add_library(render_sdl SHARED IMPORTED)
  if(WIN32)
    set_target_properties(render_sdl PROPERTIES IMPORTED_IMPLIB "${CSE_RENDER_SDL_LIBRARY}"
      IMPORTED_LOCATION "${CSE_RENDER_SDL_RUNTIME}")
  else()
    set_target_properties(render_sdl PROPERTIES IMPORTED_LOCATION "${CSE_RENDER_SDL_LIBRARY}")
  endif()
  add_executable(render_gl_oracle gl_oracle.cpp)
  target_include_directories(render_gl_oracle PRIVATE "${CSE_ROOT}/src/external/SDL3/include")
  target_link_libraries(render_gl_oracle PRIVATE render_state render_sdl)
  add_custom_command(TARGET render_gl_oracle POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${CSE_RENDER_SDL_RUNTIME}" "$<TARGET_FILE_DIR:render_gl_oracle>" VERBATIM)
  list(APPEND targets render_gl_oracle)
endif()
foreach(target IN LISTS targets)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc /fp:precise
      "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off
      "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
  if(CSE_RENDER_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
foreach(case order owned boundaries limits abort persistent allocation-failure)
  add_test(NAME render-frame-${case} COMMAND render_frame_tests ${case})
endforeach()
foreach(case matrices attributes lists ownership bounds topology vgui)
  add_test(NAME render-state-${case} COMMAND render_state_tests ${case})
endforeach()
foreach(case names uploads palettes lists-bounds formats raster lifetime)
  add_test(NAME render-texture-${case} COMMAND render_texture_tests ${case})
endforeach()
if(CSE_RENDER_GL_ORACLE)
  add_test(NAME render-real-gl-state COMMAND render_gl_oracle state)
  add_test(NAME render-real-gl-lowering COMMAND render_gl_oracle lowering)
  add_test(NAME render-real-gl-textures COMMAND render_gl_oracle textures)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/selected-$<CONFIG>.txt" CONTENT
  "sdl_library=${CSE_RENDER_SDL_LIBRARY}\nsdl_runtime=${CSE_RENDER_SDL_RUNTIME}\n")
