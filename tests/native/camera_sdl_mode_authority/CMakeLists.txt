cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Owned fixture deployment minimum")
endif()
project(CseCameraSdlModeAuthority LANGUAGES C CXX)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(CSE_SDL_SOURCE_DIR "" CACHE PATH "Pinned unmodified SDL3-3.4.20 source tree; reused read-only")
if(NOT EXISTS "${CSE_SDL_SOURCE_DIR}/src/events/SDL_mouse.c")
    message(FATAL_ERROR "Set CSE_SDL_SOURCE_DIR to the pinned source tree; no provider is built or linked")
endif()
set(CSE_MODE_MUTANT "none" CACHE STRING "Deliberate fixture-only negative variant")
option(CSE_MODE_SANITIZERS "Sanitize fake SDL/camera source bodies" OFF)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_sources.py"
    "${CSE_SDL_SOURCE_DIR}" "${CSE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/imported" "${CSE_MODE_MUTANT}"
    COMMAND_ERROR_IS_FATAL ANY)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/config")
configure_file("${CSE_SDL_SOURCE_DIR}/include/build_config/SDL_build_config_minimal.h"
    "${CMAKE_CURRENT_BINARY_DIR}/config/SDL_build_config.h" COPYONLY)
add_library(fake_sdk OBJECT sdk_fixture.c)
foreach(game cs hl)
    add_executable(mode_${game} mode_tests.cpp camera_cases.cpp caller_cases.cpp $<TARGET_OBJECTS:fake_sdk>)
    if(game STREQUAL cs)
        target_compile_definitions(mode_${game} PRIVATE CSE_CAMERA_CS=1)
    else()
        target_compile_definitions(mode_${game} PRIVATE CSE_CAMERA_CS=0)
    endif()
endforeach()
foreach(target fake_sdk mode_cs mode_hl)
    target_compile_definitions(${target} PRIVATE SDL_STATIC_LIB=1 SDL_MAIN_HANDLED=1 SDL_ASSERT_LEVEL=0)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/config"
        "${CMAKE_CURRENT_BINARY_DIR}/imported" "${CSE_SDL_SOURCE_DIR}/include"
        "${CSE_SDL_SOURCE_DIR}/src" "${CSE_SDL_SOURCE_DIR}/src/video"
        "${CSE_ROOT}/GoldSrc-SDK/common" "${CSE_ROOT}/GoldSrc-SDK/cl_dll")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:precise
            $<$<CONFIG:Release>:/O2> $<$<CONFIG:Debug>:/Od>)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter
            -Wno-overriding-option -ffp-model=precise -ffp-contract=off $<$<CONFIG:Release>:-O2> $<$<CONFIG:Debug>:-O0> $<$<CONFIG:Debug>:-g>)
        if(CSE_MODE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        endif()
    endif()
endforeach()
set_target_properties(fake_sdk PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
enable_testing()
foreach(game cs hl)
    set_target_properties(mode_${game} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(mode_${game} PRIVATE /EHs-c-)
    else()
        target_compile_options(mode_${game} PRIVATE -fno-exceptions -funwind-tables)
        if(CSE_MODE_SANITIZERS)
            target_link_options(mode_${game} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME sdl-mode-${game}-authority COMMAND mode_${game})
    set_tests_properties(sdl-mode-${game}-authority PROPERTIES TIMEOUT 15)
endforeach()
