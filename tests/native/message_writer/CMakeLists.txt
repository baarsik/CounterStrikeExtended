cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseMessageWriter LANGUAGES CXX)
set(writer_source "${CSE_ROOT}/src/common/parsemsg.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${writer_source}")
file(READ "${writer_source}" writer_translation_unit)
string(FIND "${writer_translation_unit}" "BufferWriter::BufferWriter()" writer_start)
if(writer_start LESS 0)
    message(FATAL_ERROR "Actual BufferWriter source boundary is missing")
endif()
string(SUBSTRING "${writer_translation_unit}" ${writer_start} -1 writer_methods)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual_writer.cpp"
    "#include \"parsemsg.h\"\n#include \"length_seams.h\"\n#define strlen CseFixtureLength\n${writer_methods}")
add_executable(message_writer_tests writer_tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/actual_writer.cpp")
target_compile_features(message_writer_tests PRIVATE cxx_std_14)
target_include_directories(message_writer_tests PRIVATE "${CSE_ROOT}/src/common"
    "${CSE_ROOT}/src/public" "${CMAKE_CURRENT_LIST_DIR}")
option(CSE_MESSAGE_WRITER_SANITIZERS "Instrument actual message writer methods" OFF)
if(MSVC)
    target_compile_definitions(message_writer_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(message_writer_tests PRIVATE /W4 /WX /EHsc /fp:precise
        "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
else()
    target_compile_options(message_writer_tests PRIVATE -Wall -Wextra -Werror -Werror=shorten-64-to-32
        -fno-fast-math -ffp-contract=off -fno-exceptions -funwind-tables
        "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
    if(CSE_MESSAGE_WRITER_SANITIZERS)
        target_compile_options(message_writer_tests PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        target_link_options(message_writer_tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_test(NAME actual-message-writer COMMAND message_writer_tests)
