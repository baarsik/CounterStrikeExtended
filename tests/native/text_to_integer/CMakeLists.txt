cmake_minimum_required(VERSION 3.25)
project(CseTextToInteger LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CSE_TEXT_SANITIZERS "Instrument conversion fixtures" OFF)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_corpus.py"
    --output "${CMAKE_CURRENT_BINARY_DIR}/corpus" COMMAND_ERROR_IS_FATAL ANY)
function(text_target name source)
    add_executable(${name} "${source}")
    target_include_directories(${name} PRIVATE "${CSE_ROOT}/GoldSrc-SDK/common" "${CMAKE_CURRENT_BINARY_DIR}/corpus")
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(${name} PRIVATE NOMINMAX)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion)
        if(CSE_TEXT_SANITIZERS)
            target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
                -fno-omit-frame-pointer)
            target_link_options(${name} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

        endforeach()
    endforeach()
