cmake_minimum_required(VERSION 3.25)
project(CseVguiBackdropTests LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_BACKDROP_SDL_LIBRARY "" CACHE FILEPATH "Native SDL3 shared/import library")
if(NOT EXISTS "${CSE_BACKDROP_SDL_LIBRARY}")
  message(FATAL_ERROR "Set CSE_BACKDROP_SDL_LIBRARY to the matching SDL3 library")
endif()
find_package(OpenGL REQUIRED)
add_executable(vgui_backdrop_tests backdrop_tests.cpp "${CSE_ROOT}/src/engine/vgui_background_blur_gl.cpp")
target_compile_features(vgui_backdrop_tests PRIVATE cxx_std_17)
target_include_directories(vgui_backdrop_tests PRIVATE "${CSE_ROOT}/src/engine" "${CSE_ROOT}/src/external/SDL3/include")
target_link_libraries(vgui_backdrop_tests PRIVATE "${CSE_BACKDROP_SDL_LIBRARY}" OpenGL::GL)
if(NOT MSVC)
  target_compile_options(vgui_backdrop_tests PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
endif()
enable_testing()
add_test(NAME vgui-backdrop-real-gl COMMAND vgui_backdrop_tests)
set_tests_properties(vgui-backdrop-real-gl PROPERTIES TIMEOUT 30)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_ROOT}/src/vgui2/controls/controls.cpp"
  "${CSE_ROOT}/src/cstrike/vgui2/controls/controls.cpp" extract_bindings.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract_bindings.py"
  "${CSE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/bindings" COMMAND_ERROR_IS_FATAL ANY)
foreach(owner shared cs)
  add_executable(vgui_backdrop_${owner}_bindings binding_tests.cpp)
  target_compile_features(vgui_backdrop_${owner}_bindings PRIVATE cxx_std_17)
  target_include_directories(vgui_backdrop_${owner}_bindings PRIVATE "${CSE_ROOT}/src/engine"
    "${CSE_ROOT}/src/public" "${CMAKE_CURRENT_BINARY_DIR}/bindings")
  target_compile_definitions(vgui_backdrop_${owner}_bindings PRIVATE CSE_BINDINGS_FILE="${owner}_bindings.inc")
  add_test(NAME vgui-backdrop-${owner}-bindings COMMAND vgui_backdrop_${owner}_bindings)
endforeach()
