cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseNativeFileSystem LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(manifest_sources "${CMAKE_CURRENT_BINARY_DIR}/manifest_sources.cmake")
set(CSE_FILESYSTEM_MANIFEST "${CSE_ROOT}/GoldSrc-SDK/filesystem/filesystem_stdio.vcxproj"
  CACHE FILEPATH "Owned current or pinned-reference filesystem manifest")
get_filename_component(filesystem_source_directory "${CSE_FILESYSTEM_MANIFEST}" DIRECTORY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/manifest_sources.py"
  "${CSE_ROOT}" "${CSE_TARGET_OS}" "${manifest_sources}" "${CSE_FILESYSTEM_MANIFEST}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_FILESYSTEM_MANIFEST}" "${CSE_ROOT}/tools/translate_msbuild.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/manifest_sources.py")
include("${manifest_sources}")
add_library(filesystem_stdio SHARED ${CSE_FILESYSTEM_SOURCES})
target_compile_features(filesystem_stdio PRIVATE cxx_std_17)
target_include_directories(filesystem_stdio PRIVATE "${filesystem_source_directory}")
target_compile_definitions(filesystem_stdio PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
target_link_libraries(filesystem_stdio PRIVATE "${CMAKE_DL_LIBS}")
if(MSVC)
  target_compile_options(filesystem_stdio PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  target_compile_definitions(filesystem_stdio PRIVATE WIN32 _WINDOWS _USRDLL)
  target_link_options(filesystem_stdio PRIVATE /EXPORT:CreateInterface)
else()
  target_compile_options(filesystem_stdio PRIVATE -Wall -Wextra
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
target_compile_definitions(filesystem_stdio PRIVATE "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
set_property(TARGET filesystem_stdio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set_target_properties(filesystem_stdio PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
add_executable(filesystem_tests filesystem_tests.cpp)
target_compile_features(filesystem_tests PRIVATE cxx_std_17)
target_include_directories(filesystem_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/filesystem")
target_link_libraries(filesystem_tests PRIVATE "${CMAKE_DL_LIBS}")
add_dependencies(filesystem_tests filesystem_stdio)
if(MSVC)
  target_compile_options(filesystem_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
else()
  target_compile_options(filesystem_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
set_property(TARGET filesystem_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(CSE_FILESYSTEM_SANITIZERS "Instrument actual filesystem module and fixtures with ASan/UBSan" OFF)
option(CSE_FILESYSTEM_SANITIZE_MODULE "Instrument the reloadable module as well as the driver" ON)
if(CSE_FILESYSTEM_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
  endif()
  set(sanitized_targets filesystem_tests)
  if(CSE_FILESYSTEM_SANITIZE_MODULE)
    list(APPEND sanitized_targets filesystem_stdio)
  else()
    target_compile_options(filesystem_stdio PRIVATE -fno-sanitize=all)
    target_link_options(filesystem_stdio PRIVATE -fno-sanitize=all)
  endif()
  foreach(target IN LISTS sanitized_targets)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
foreach(source IN LISTS CSE_FILESYSTEM_SOURCES)
  if(source MATCHES "/utlsymbol[.]cpp$")
    set(symbol_source "${source}")
  endif()
endforeach()
if(NOT symbol_source)
  message(FATAL_ERROR "Owned filesystem manifest did not select its symbol implementation")
endif()
add_executable(filesystem_symbol_owner_tests symbol_owner_tests.cpp "${symbol_source}")
target_compile_features(filesystem_symbol_owner_tests PRIVATE cxx_std_17)
target_include_directories(filesystem_symbol_owner_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/filesystem")
target_compile_definitions(filesystem_symbol_owner_tests PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
set_property(TARGET filesystem_symbol_owner_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
  target_compile_options(filesystem_symbol_owner_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
else()
  target_compile_options(filesystem_symbol_owner_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
if(CSE_FILESYSTEM_SANITIZERS)
  target_compile_options(filesystem_symbol_owner_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(filesystem_symbol_owner_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME filesystem-symbol-owners COMMAND filesystem_symbol_owner_tests)
set(CSE_FILESYSTEM_CASE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/fixtures" CACHE PATH "Owned case-sensitive fixture root")
foreach(scenario module search files pack find case long handles unload-find mount-lifecycle lifecycle)
  if(APPLE AND CSE_FILESYSTEM_SANITIZERS AND CSE_FILESYSTEM_SANITIZE_MODULE AND scenario STREQUAL lifecycle)
    # Apple ASan retains image-global registrations after dlclose. Keep the
    # module fully instrumented for API/mount tests; qualify actual reloads
    # separately with an instrumented driver and an uninstrumented module.
    continue()
  endif()
  if(scenario STREQUAL case)
    set(fixture_root "${CSE_FILESYSTEM_CASE_ROOT}")
  else()
    set(fixture_root "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  endif()
  add_test(NAME filesystem-${scenario} COMMAND filesystem_tests "$<TARGET_FILE:filesystem_stdio>"
    "${fixture_root}/${scenario}" "${scenario}")
endforeach()
if(NOT WIN32)
  add_executable(filesystem_prefix_tests prefix_tests.cpp)
  target_compile_features(filesystem_prefix_tests PRIVATE cxx_std_17)
  target_include_directories(filesystem_prefix_tests PRIVATE "${CSE_ROOT}/GoldSrc-SDK/filesystem")
  target_compile_options(filesystem_prefix_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  if(CSE_FILESYSTEM_SANITIZERS)
    target_compile_options(filesystem_prefix_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(filesystem_prefix_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME filesystem-prefix COMMAND filesystem_prefix_tests "${CMAKE_CURRENT_BINARY_DIR}/prefix-fixture")
endif()

option(CSE_FILESYSTEM_LEAK_CHECKS "Run actual-module valid-ownership fixtures with Apple's leaks tool" OFF)
if(CSE_FILESYSTEM_LEAK_CHECKS)
  if(NOT APPLE OR CSE_FILESYSTEM_SANITIZERS)
    message(FATAL_ERROR "The leaks fixtures require an unsanitized macOS build")
  endif()
  find_program(CSE_LEAKS_EXECUTABLE leaks REQUIRED)
  set(CSE_FILESYSTEM_RECEIPT_DIR "${CMAKE_CURRENT_BINARY_DIR}/receipts" CACHE PATH "Filesystem fixture receipts")
  foreach(scenario search pack find long unload-find lifecycle)
    add_test(NAME filesystem-leaks-${scenario} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/check_leaks.py" "${CSE_LEAKS_EXECUTABLE}"
      "$<TARGET_FILE:filesystem_tests>" "$<TARGET_FILE:filesystem_stdio>"
      "${CMAKE_CURRENT_BINARY_DIR}/leak-fixtures/${scenario}" "${scenario}"
      "${CSE_FILESYSTEM_RECEIPT_DIR}/leaks-${scenario}.log")
  endforeach()
endif()
