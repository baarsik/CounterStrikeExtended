cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseTier0HandleFixtures LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CSE_TIER0_HANDLES_SOURCE_DIR "${CSE_ROOT}/GoldSrc-SDK" CACHE PATH
  "Production SDK, or an explicitly archived source/header snapshot")
option(CSE_TIER0_HANDLES_LEGACY "Allow visible diagnostics from the archived negative reference" OFF)
option(CSE_TIER0_HANDLES_SANITIZERS "Enable ASan and UBSan" OFF)
foreach(source tier0/threadtools.cpp tier0/vprof.cpp public/tier0/threadtools.h)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_TIER0_HANDLES_SOURCE_DIR}/${source}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS extract_production.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_production.py"
  "${CSE_TIER0_HANDLES_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}"
  RESULT_VARIABLE extraction_status ERROR_VARIABLE extraction_error)
if(NOT extraction_status EQUAL 0)
  message(FATAL_ERROR "Cannot extract production tier0 boundaries: ${extraction_error}")
endif()
function(cse_handle_fixture target source)
  add_executable(${target} "${source}")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CSE_ROOT}/GoldSrc-SDK/public")
  target_link_libraries(${target} PRIVATE Threads::Threads)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
    if(NOT CSE_TIER0_HANDLES_LEGACY)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
    if(CSE_TIER0_HANDLES_LEGACY)
      target_compile_options(${target} PRIVATE -fms-extensions)
    else()
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
  if(CSE_TIER0_HANDLES_SANITIZERS AND NOT MSVC)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all
      -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
cse_handle_fixture(tier0_handle_tests handle_tests.cpp)
enable_testing()
foreach(scenario layout thread-name cached-name tls-bits serial-start scope-gates tls-isolation)
  add_test(NAME "tier0-handle-${scenario}" COMMAND tier0_handle_tests "${scenario}")
endforeach()
if(WIN32)
  add_test(NAME tier0-handle-event COMMAND tier0_handle_tests event)
else()
  # This target compiles the unchanged Windows method bodies in an owned field context.
  # It is deliberately separate from the actual native header/TLS executable.
  cse_handle_fixture(tier0_windows_diagnostic_body_tests windows_body_tests.cpp)
  add_test(NAME tier0-windows-body-name COMMAND tier0_windows_diagnostic_body_tests name)
  add_test(NAME tier0-windows-body-event COMMAND tier0_windows_diagnostic_body_tests event)
endif()
get_directory_property(handle_test_names TESTS)
set_tests_properties(${handle_test_names} PROPERTIES TIMEOUT 30)
