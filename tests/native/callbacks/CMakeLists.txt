cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseSavedCallbackTests LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
add_executable(saved_callback_tests callback_tests.cpp)
target_compile_features(saved_callback_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(saved_callback_tests PRIVATE /W4 /WX /EHsc)
endif()
enable_testing()
add_test(NAME saved-callbacks COMMAND saved_callback_tests)
foreach(game IN ITEMS hl cs)
  add_executable(${game}_callback_field_tests game_field_tests.cpp)
  target_compile_features(${game}_callback_field_tests PRIVATE cxx_std_17)
  if(game STREQUAL "cs")
    target_compile_definitions(${game}_callback_field_tests PRIVATE CSE_SAVE_GAME_CS)
  endif()
  if(MSVC)
    target_compile_options(${game}_callback_field_tests PRIVATE /W4 /WX /EHsc)
  endif()
  add_test(NAME ${game}-callback-fields COMMAND ${game}_callback_field_tests)
endforeach()
