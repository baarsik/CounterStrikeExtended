cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseA3dFloatStateFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_A3D_MIXER_SOURCE "${CSE_ROOT}/src/a3dapi/a2d.cpp" CACHE FILEPATH "Actual current or preserved mixer source")
get_filename_component(mixer_directory "${CSE_A3D_MIXER_SOURCE}" DIRECTORY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  "${CSE_ROOT}" "${CSE_A3D_MIXER_SOURCE}" "${CMAKE_BUILD_TYPE}" "${CSE_TARGET_ARCH}" "${CMAKE_CURRENT_BINARY_DIR}"
  COMMAND_ERROR_IS_FATAL ANY)
include("${CSE_ROOT}/tests/native/CMakeInventoryDependencies.cmake")
cse_fixture_inventory_dependencies("${CSE_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  "${CSE_A3D_MIXER_SOURCE}" "${mixer_directory}/A3dGeom.cpp")
include("${CMAKE_CURRENT_BINARY_DIR}/selected_policy.cmake")
add_executable(a3d_float_tests float_tests.cpp mixer_subject.cpp)
target_compile_features(a3d_float_tests PRIVATE cxx_std_17)
target_include_directories(a3d_float_tests PRIVATE "${mixer_directory}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(a3d_float_tests PRIVATE CSE_A3D_MIXER_SOURCE="${CSE_A3D_MIXER_SOURCE}"
  "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
if(MSVC)
  set_property(TARGET a3d_float_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(a3d_float_tests PRIVATE WIN32 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(a3d_float_tests PRIVATE /W4 /WX)
  set_property(SOURCE mixer_subject.cpp APPEND PROPERTY COMPILE_OPTIONS ${CSE_A3D_SUBJECT_OPTIONS})
  set_property(SOURCE float_tests.cpp APPEND PROPERTY COMPILE_OPTIONS /EHsc /fp:strict
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
else()
  target_include_directories(a3d_float_tests PRIVATE platform)
  target_compile_options(a3d_float_tests PRIVATE -Wall -Wextra -Werror -fms-extensions -ffp-contract=off)
  set_property(SOURCE mixer_subject.cpp APPEND PROPERTY COMPILE_OPTIONS -O2)
  set_property(SOURCE float_tests.cpp APPEND PROPERTY COMPILE_OPTIONS -frounding-math
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
endif()
option(CSE_A3D_FLOAT_SANITIZERS "Instrument actual mixer and state/PCM fixtures" OFF)
if(CSE_A3D_FLOAT_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
  endif()
  target_compile_options(a3d_float_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(a3d_float_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME a3d-float-state COMMAND a3d_float_tests state)
add_test(NAME a3d-float-conversion COMMAND a3d_float_tests conversion)
set(CSE_A3D_PCM_REFERENCE "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" CACHE PATH "Pinned original owned PCM receipts")
if(IS_DIRECTORY "${CSE_A3D_PCM_REFERENCE}")
  add_test(NAME a3d-float-pcm COMMAND a3d_float_tests pcm "${CSE_A3D_PCM_REFERENCE}")
endif()
if(CSE_TARGET_ARCH STREQUAL x64 AND WIN32)
  add_test(NAME a3d-float-crt-precision-probe COMMAND a3d_float_tests precision-probe)
endif()
