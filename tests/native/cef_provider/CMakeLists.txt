cmake_minimum_required(VERSION 3.25)
project(CseCefProviderFixtures LANGUAGES C CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the extracted pinned CEF distribution")
endif()
file(TO_CMAKE_PATH "${CEF_ROOT}" CEF_ROOT)
file(READ "${CSE_ROOT}/tools/qualification/fixtures/cef/lock.json" cse_cef_lock)
if(APPLE AND CSE_TARGET_ARCH STREQUAL "arm64")
  set(cse_cef_platform macosarm64)
  set(PROJECT_ARCH arm64)
elseif(WIN32 AND CSE_TARGET_ARCH STREQUAL "x64")
  set(cse_cef_platform windows64)
  set(PROJECT_ARCH x86_64)
else()
  message(FATAL_ERROR "This fixture requires a pinned native macOS ARM64 or Windows x64 provider")
endif()
set(cse_cef_pinned_files api_header runtime debug_runtime)
if(WIN32)
  list(APPEND cse_cef_pinned_files import_library)
endif()
foreach(item IN LISTS cse_cef_pinned_files)
  if(item STREQUAL "api_header")
    set(cse_cef_path include/cef_api_versions.h)
    string(JSON cse_cef_expected GET "${cse_cef_lock}" platforms ${cse_cef_platform} api_header_sha256)
  else()
    string(JSON cse_cef_path GET "${cse_cef_lock}" platforms ${cse_cef_platform} ${item} path)
    string(JSON cse_cef_expected GET "${cse_cef_lock}" platforms ${cse_cef_platform} ${item} sha256)
  endif()
  file(SHA256 "${CEF_ROOT}/${cse_cef_path}" cse_cef_actual)
  if(NOT cse_cef_actual STREQUAL cse_cef_expected)
    message(FATAL_ERROR "Pinned CEF ${item} does not match: ${cse_cef_path}")
  endif()
endforeach()
set(api_version 15600 CACHE STRING "Pinned stable CEF API" FORCE)
set(USE_SANDBOX ON CACHE BOOL "Keep the CEF process sandbox" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(APPLE)
  enable_language(OBJCXX)
  list(REMOVE_ITEM CEF_COMPILER_FLAGS "-mmacosx-version-min=13.0")
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
endif()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" "${CMAKE_BINARY_DIR}/libcef_dll_wrapper")
set(CEF_TARGET cse_cef_probe)
SET_CEF_TARGET_OUT_DIR()
if(APPLE)
  add_executable(${CEF_TARGET} MACOSX_BUNDLE probe.cpp main_mac.mm)
  SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
  target_compile_options(${CEF_TARGET} PRIVATE -fobjc-arc)
  set_target_properties(${CEF_TARGET} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
  set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
  COPY_MAC_FRAMEWORK("${CEF_TARGET}" "${CEF_BINARY_DIR}" "${CEF_APP}")
  foreach(suffixes IN LISTS CEF_HELPER_APP_SUFFIXES)
    string(REPLACE ":" ";" suffixes "${suffixes}")
    list(GET suffixes 0 name_suffix)
    list(GET suffixes 1 target_suffix)
    list(GET suffixes 2 plist_suffix)
    set(helper "cse_cef_probe_Helper${target_suffix}")
    set(helper_name "cse_cef_probe Helper${name_suffix}")
    configure_file(helper-Info.plist.in "helper${target_suffix}.plist" @ONLY)
    add_executable(${helper} MACOSX_BUNDLE "${CEF_ROOT}/tests/cefsimple/process_helper_mac.cc")
    SET_EXECUTABLE_TARGET_PROPERTIES(${helper})
    target_link_libraries(${helper} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    set_target_properties(${helper} PROPERTIES OUTPUT_NAME "${helper_name}"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/helper${target_suffix}.plist")
    add_dependencies(${CEF_TARGET} ${helper})
    add_custom_command(TARGET ${CEF_TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CEF_TARGET_OUT_DIR}/${helper_name}.app" "${CEF_APP}/Contents/Frameworks/${helper_name}.app" VERBATIM)
  endforeach()
else()
  ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_library(${CEF_TARGET} SHARED probe.cpp main_win.cpp)
  SET_LIBRARY_TARGET_PROPERTIES(${CEF_TARGET})
  target_link_libraries(${CEF_TARGET} PRIVATE libcef_lib)
  COPY_SINGLE_FILE(${CEF_TARGET} ${CEF_BINARY_DIR}/bootstrap.exe ${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.exe)
  COPY_FILES("${CEF_TARGET}" "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
  COPY_FILES("${CEF_TARGET}" "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
  SET_LPAC_ACLS("${CEF_TARGET}")
endif()
target_link_libraries(${CEF_TARGET} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
target_compile_features(${CEF_TARGET} PRIVATE cxx_std_20)
