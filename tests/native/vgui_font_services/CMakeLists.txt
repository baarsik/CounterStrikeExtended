cmake_minimum_required(VERSION 3.25)
project(CseVguiFontServicesDesign LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(vgui_source "${CSE_ROOT}/src/utils/vgui/lib-src/vgui")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(public_font_header OBJECT header_probe.cpp)
target_include_directories(public_font_header PRIVATE "${CSE_ROOT}/src/utils/vgui/include")
add_library(private_font_header OBJECT header_probe.cpp)
target_include_directories(private_font_header PRIVATE "${vgui_source}/include")
if(MSVC)
    foreach(target public_font_header private_font_header)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
        target_compile_definitions(${target} PRIVATE VGUI_EXPORTS NOMINMAX)
    endforeach()
    add_executable(controlled_gdi controlled_gdi.cpp "${vgui_source}/Font.cpp"
        "${vgui_source}/common/fileimage.cpp" "${vgui_source}/common/vfontdata.cpp")
    target_include_directories(controlled_gdi PRIVATE "${vgui_source}/include" "${vgui_source}/common"
        "${vgui_source}/win32")
    target_compile_options(controlled_gdi PRIVATE /W4 /WX /EHsc)
    target_compile_definitions(controlled_gdi PRIVATE VGUI_EXPORTS NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
    foreach(target public_font_header private_font_header)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Xclang -fdump-record-layouts)
    endforeach()
    add_library(private_service_headers OBJECT services_probe.cpp)
    target_include_directories(private_service_headers PRIVATE
        "${CSE_ROOT}/src/vgui2/vgui_surfacelib"
        "${CSE_ROOT}/src/vgui2/src/public" "${CSE_ROOT}/src/public")
    target_compile_options(private_service_headers PRIVATE -Wall -Wextra -Werror -fno-exceptions)
endif()
