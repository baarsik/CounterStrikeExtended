cmake_minimum_required(VERSION 3.25)
project(CseVgui1RealGdiReference LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This reference uses actual Windows memory GDI; no Mac font provider execution")
endif()
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_methods.py"
    "${CSE_ROOT}/src/utils/vgui/lib-src/vgui/win32/Font.cpp"
    "${CMAKE_CURRENT_BINARY_DIR}/extracted" COMMAND_ERROR_IS_FATAL ANY)
add_executable(real_gdi_reference reference.cpp)
target_include_directories(real_gdi_reference PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/extracted")
target_compile_definitions(real_gdi_reference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(real_gdi_reference PRIVATE /W4 /WX /EHsc /fp:precise)
target_link_libraries(real_gdi_reference PRIVATE gdi32 user32 bcrypt)
