cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseParticleMemoryFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(CSE_PARTICLE_MEMORY_SANITIZERS "Instrument the full actual particle module and allocation fixtures" OFF)
option(CSE_PARTICLE_MEMORY_SANITIZE_MODULE "Instrument reloadable production objects as well as test drivers" ON)
set(manifest_sources "${CMAKE_CURRENT_BINARY_DIR}/manifest_sources.cmake")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/manifest_sources.py"
  "${CSE_ROOT}" "${CSE_TARGET_OS}" "${CMAKE_BUILD_TYPE}" "${manifest_sources}" COMMAND_ERROR_IS_FATAL ANY)
include("${manifest_sources}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CSE_ROOT}/GoldSrc-SDK/projects/vs2026/particleman.vcxproj" "${CSE_ROOT}/tools/translate_msbuild.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/manifest_sources.py")
add_library(particleman_objects OBJECT ${CSE_PARTICLE_SOURCES})
set_property(TARGET particleman_objects PROPERTY POSITION_INDEPENDENT_CODE ON)
target_compile_features(particleman_objects PRIVATE cxx_std_17)
add_library(particleman SHARED $<TARGET_OBJECTS:particleman_objects>)
target_link_libraries(particleman PRIVATE "${CMAKE_DL_LIBS}")
set_property(TARGET particleman_objects PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
  target_compile_options(particleman_objects PRIVATE /W4 /WX /EHsc)
  include("${CSE_ROOT}/cmake/RuntimeTargets.cmake")
  cse_game_module_exports(particleman "${CSE_ROOT}/GoldSrc-SDK/projects/vs2026/particleman.def"
    "${CSE_ROOT}/cmake/exports/particleman.def")
else()
  target_compile_options(particleman_objects PRIVATE -Wall -Wextra -Werror=shorten-64-to-32)
  if(APPLE)
    target_compile_definitions(particleman_objects PRIVATE OSX POSIX)
  endif()
endif()
set_target_properties(particleman PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
add_executable(particle_boundary_tests boundary_tests.cpp)
target_compile_features(particle_boundary_tests PRIVATE cxx_std_17)
target_include_directories(particle_boundary_tests SYSTEM PRIVATE ${CSE_PARTICLE_INCLUDE_DIRECTORIES})
target_compile_definitions(particle_boundary_tests PRIVATE CLIENT_DLL)
if(MSVC)
  target_compile_options(particle_boundary_tests PRIVATE /W4 /WX /EHs-c-)
else()
  target_compile_options(particle_boundary_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions)
  if(APPLE)
    target_compile_definitions(particle_boundary_tests PRIVATE OSX POSIX)
  endif()
endif()
enable_testing()
add_test(NAME particle-allocation-boundary COMMAND particle_boundary_tests)
add_executable(particle_pool_tests pool_tests.cpp $<TARGET_OBJECTS:particleman_objects>)
target_link_libraries(particle_pool_tests PRIVATE "${CMAKE_DL_LIBS}")
add_executable(particle_layout_probe layout_probe.cpp)
add_executable(particle_module_tests module_tests.cpp)
target_link_libraries(particle_module_tests PRIVATE "${CMAKE_DL_LIBS}")
add_dependencies(particle_module_tests particleman)
foreach(target particle_pool_tests particle_layout_probe particle_module_tests)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} SYSTEM PRIVATE ${CSE_PARTICLE_INCLUDE_DIRECTORIES})
  target_compile_definitions(${target} PRIVATE CLIENT_DLL)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(APPLE)
      target_compile_definitions(${target} PRIVATE OSX POSIX)
    endif()
  endif()
endforeach()
foreach(scenario reset active unused reload)
  if(APPLE AND CSE_PARTICLE_MEMORY_SANITIZERS AND CSE_PARTICLE_MEMORY_SANITIZE_MODULE AND scenario STREQUAL reload)
    # The actual Apple ASan image-registration reload failure is retained in the
    # qualification receipt. Instrument the module for all other API/lifetime
    # cases and qualify true reloads separately with the uninstrumented module.
    continue()
  endif()
  add_test(NAME particle-actual-module-${scenario} COMMAND particle_module_tests "$<TARGET_FILE:particleman>" "${scenario}")
endforeach()
if(CSE_PARTICLE_MEMORY_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This fixture's ASan/UBSan mode requires Clang/GNU")
  endif()
  set(sanitized_targets particle_boundary_tests particle_pool_tests particle_layout_probe particle_module_tests)
  if(CSE_PARTICLE_MEMORY_SANITIZE_MODULE)
    list(APPEND sanitized_targets particleman_objects particleman)
  else()
    target_compile_options(particleman_objects PRIVATE -fno-sanitize=all)
    target_link_options(particleman PRIVATE -fno-sanitize=all)
  endif()
  foreach(target IN LISTS sanitized_targets)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
foreach(target particleman_objects particle_boundary_tests particle_pool_tests particle_layout_probe particle_module_tests)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS
    "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  if(MSVC)
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:/O2>"
      "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  else()
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:-O2>"
      "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  endif()
endforeach()
add_test(NAME particle-layout COMMAND particle_layout_probe)
foreach(scenario capacity construction restart phase exit-active)
  add_test(NAME particle-pool-${scenario} COMMAND particle_pool_tests "${scenario}")
endforeach()
option(CSE_PARTICLE_MEMORY_LEAK_CHECKS "Run actual pool/module ownership scenarios through Apple's leaks tool" OFF)
if(CSE_PARTICLE_MEMORY_LEAK_CHECKS)
  if(NOT APPLE OR CSE_PARTICLE_MEMORY_SANITIZERS)
    message(FATAL_ERROR "Particle leaks fixtures require an unsanitized macOS build")
  endif()
  find_program(CSE_LEAKS_EXECUTABLE leaks REQUIRED)
  set(CSE_PARTICLE_MEMORY_RECEIPT_DIR "${CMAKE_CURRENT_BINARY_DIR}/receipts" CACHE PATH "Particle leak receipts")
  foreach(scenario reset active unused reload)
    add_test(NAME particle-leaks-module-${scenario} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/check_leaks.py"
      "${CSE_PARTICLE_MEMORY_RECEIPT_DIR}/leaks-module-${scenario}.log" "Particle module ${scenario} passed"
      "${CSE_LEAKS_EXECUTABLE}" "$<TARGET_FILE:particle_module_tests>" "$<TARGET_FILE:particleman>" "${scenario}")
  endforeach()
  foreach(scenario capacity construction restart exit-active)
    add_test(NAME particle-leaks-pool-${scenario} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/check_leaks.py"
      "${CSE_PARTICLE_MEMORY_RECEIPT_DIR}/leaks-pool-${scenario}.log" "Actual particle pool ${scenario} passed"
      "${CSE_LEAKS_EXECUTABLE}" "$<TARGET_FILE:particle_pool_tests>" "${scenario}")
  endforeach()
endif()

if(NOT WIN32)
  add_test(NAME particle-core-allocation COMMAND particle_pool_tests core-allocation)
endif()
