cmake_minimum_required(VERSION 3.25)
project(CseConsoleNotify LANGUAGES CXX)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CSE_NOTIFY_SOURCE "${CSE_ROOT}/src/engine/console.cpp" CACHE FILEPATH "Actual owning console source")
option(CSE_NOTIFY_SANITIZERS "Run actual notify body with ASan and UBSan" OFF)
set(CSE_NOTIFY_SANITIZER_SET "address,undefined" CACHE STRING "Sanitizer instrumentation for isolated RED evidence")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CSE_NOTIFY_SOURCE}" extract.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/extract.py"
    --source "${CSE_NOTIFY_SOURCE}" --out "${CMAKE_CURRENT_BINARY_DIR}/actual"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(console_notify_tests notify_tests.cpp)
target_include_directories(console_notify_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/actual")
set_target_properties(console_notify_tests PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(console_notify_tests PRIVATE /W4 /WX)
else()
    target_compile_options(console_notify_tests PRIVATE -Wall -Wextra -Werror)
endif()
if(CSE_NOTIFY_SANITIZERS)
    target_compile_options(console_notify_tests PRIVATE -fsanitize=${CSE_NOTIFY_SANITIZER_SET} -fno-sanitize-recover=all)
    target_link_options(console_notify_tests PRIVATE -fsanitize=${CSE_NOTIFY_SANITIZER_SET})
endif()
enable_testing()
foreach(case short-1920 short-2560 short-3840 empty-stale short-stale full-1920 full-2560 full-3840
        columns-narrow columns-zero columns-negative high-bit expired inactive timing-boundary timing-ring
        negative-history line-maximum chat-expired chat-after-notify)
    add_test(NAME notify-${case} COMMAND console_notify_tests ${case})
endforeach()
