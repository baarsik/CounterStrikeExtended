cmake_minimum_required(VERSION 3.25)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CseA3dDeviceFixture LANGUAGES CXX)
include("${CSE_ROOT}/cmake/TargetPlatform.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CSE_A3D_DEVICE_SOURCE "${CSE_ROOT}/src/a3dapi" CACHE PATH "Current or preserved A3D source")
option(CSE_A3D_DEVICE_REFERENCE "Execute the pre-fix DWORD cast with Clang MS extension" OFF)
option(CSE_A3D_DEVICE_SANITIZERS "Instrument production initialization bodies and fixture" OFF)
option(CSE_A3D_DEVICE_BUILD_FULL_MODULE "Build and probe the full Windows A3D module" ON)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/production_bodies.inc")
add_custom_command(OUTPUT "${generated}" COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/extract_sources.py" "${CSE_A3D_DEVICE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}"
  DEPENDS extract_sources.py "${CSE_A3D_DEVICE_SOURCE}/A3dScene.cpp" "${CSE_A3D_DEVICE_SOURCE}/resman.h"
    "${CSE_A3D_DEVICE_SOURCE}/ia3dapi.h" "${CSE_A3D_DEVICE_SOURCE}/A3d3.h"
    "${CSE_A3D_DEVICE_SOURCE}/A3dGeom.cpp" VERBATIM)
add_executable(a3d_device_tests device_tests.cpp "${generated}")
target_compile_features(a3d_device_tests PRIVATE cxx_std_17)
target_compile_definitions(a3d_device_tests PRIVATE "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
target_include_directories(a3d_device_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}"
  "${CSE_ROOT}/src/a3dapi")
if(WIN32)
  target_compile_definitions(a3d_device_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(a3d_device_tests PRIVATE ole32 uuid dxguid)
endif()
if(MSVC)
  set_property(TARGET a3d_device_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_options(a3d_device_tests PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
else()
  target_compile_options(a3d_device_tests PRIVATE -Wall -Wextra -Werror
    "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>" "$<$<CONFIG:Debug>:-g>")
  if(CSE_A3D_DEVICE_REFERENCE)
    target_compile_options(a3d_device_tests PRIVATE -fms-extensions -Wno-pointer-to-int-cast)
  endif()
  if(CSE_A3D_DEVICE_SANITIZERS)
    target_compile_options(a3d_device_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(a3d_device_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
foreach(case guid default failure fallback-failure no-native missing-dal repeat repeat-failure compatibility)
  add_test(NAME a3d-device-${case} COMMAND a3d_device_tests ${case})
endforeach()
if(WIN32)
  add_test(NAME a3d-device-real-default-directsound COMMAND a3d_device_tests real-default)
endif()
if(WIN32 AND CSE_A3D_DEVICE_BUILD_FULL_MODULE)
  enable_language(RC)
  set(module_sources "${CMAKE_CURRENT_BINARY_DIR}/module_sources.cmake")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/module_sources.py"
    "${CSE_ROOT}" "${CSE_A3D_DEVICE_SOURCE}" "${CMAKE_BUILD_TYPE}" "${CSE_TARGET_ARCH}" "${module_sources}"
    COMMAND_ERROR_IS_FATAL ANY)
  include("${module_sources}")
  include("${CSE_ROOT}/tests/native/CMakeInventoryDependencies.cmake")
  cse_fixture_inventory_dependencies("${CSE_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/module_sources.py")
  add_library(a3d_device_module SHARED ${CSE_A3D_MODULE_SOURCES})
  set_property(TARGET a3d_device_module PROPERTY CXX_STANDARD 17)
  set_property(TARGET a3d_device_module PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_include_directories(a3d_device_module PRIVATE "${CSE_A3D_MODULE_INCLUDE}")
  target_compile_options(a3d_device_module PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4>" "$<$<COMPILE_LANGUAGE:CXX>:/WX>")
  target_link_libraries(a3d_device_module PRIVATE ${CSE_A3D_MODULE_LIBRARIES})
  add_executable(a3d_factory_probe factory_probe.cpp "${CSE_A3D_DEVICE_SOURCE}/a3d_guids.cpp")
  target_include_directories(a3d_factory_probe PRIVATE "${CSE_A3D_DEVICE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_features(a3d_factory_probe PRIVATE cxx_std_17)
  set_property(TARGET a3d_factory_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  target_compile_definitions(a3d_factory_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
    "$<$<CONFIG:Debug>:_DEBUG>" "$<$<CONFIG:Release>:NDEBUG>")
  target_compile_options(a3d_factory_probe PRIVATE /W4 /WX /EHsc
    "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Debug>:/Zi>")
  target_link_libraries(a3d_factory_probe PRIVATE ole32 uuid dxguid)
  add_test(NAME a3d-device-actual-factory-missing-provider COMMAND a3d_factory_probe "$<TARGET_FILE:a3d_device_module>")
endif()
