cmake_minimum_required(VERSION 3.25)
set(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
get_filename_component(CSE_ROOT "${CSE_ROOT}" ABSOLUTE)
set(CSE_SDK_DIR "${CSE_ROOT}/src")
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
include("${CSE_ROOT}/cmake/PlatformDefaults.cmake")
project(CseMusicTests LANGUAGES C CXX)
include("${CSE_ROOT}/cmake/SDL3Dependency.cmake")
include("${CSE_ROOT}/cmake/Mpg123Dependency.cmake")
find_package(Threads REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
enable_testing()
foreach(test decoder player)
  add_executable(${test}_tests "${test}_tests.cpp" "${CSE_SDK_DIR}/engine/mpg123_decoder.cpp")
  if(test STREQUAL "player")
    target_sources(${test}_tests PRIVATE "${CSE_SDK_DIR}/engine/mpg123_music.cpp")
    if(MSVC)
      # Match the production analyzer gate, including the worker's stack budget.
      set_property(SOURCE "${CSE_SDK_DIR}/engine/mpg123_music.cpp" APPEND PROPERTY
        COMPILE_OPTIONS /analyze /analyze:stacksize 16384)
    endif()
  endif()
  target_compile_features(${test}_tests PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${test}_tests PRIVATE /W4 /WX /EHsc)
  endif()
  target_link_libraries(${test}_tests PRIVATE SDL3::SDL3-shared libmpg123 Threads::Threads)
  add_custom_command(TARGET ${test}_tests POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:libmpg123>" "$<TARGET_FILE_DIR:${test}_tests>"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:SDL3::SDL3-shared>" "$<TARGET_FILE_DIR:${test}_tests>")
  if(test STREQUAL decoder)
    add_test(NAME decoder COMMAND decoder_tests "${CSE_MPG123_SOURCE}/src/tests/sweep.mp3"
      "${CMAKE_CURRENT_LIST_FILE}" "${CMAKE_BINARY_DIR}/decoded-reference.pcm")
    set_tests_properties(decoder PROPERTIES FIXTURES_SETUP cse_decoded_pcm)
  else()
    add_test(NAME ${test} COMMAND ${test}_tests "${CSE_MPG123_SOURCE}/src/tests/sweep.mp3" "${CMAKE_CURRENT_LIST_FILE}")
  endif()
endforeach()
add_executable(worker_tests worker_tests.cpp "${CSE_SDK_DIR}/engine/thread.cpp")
target_compile_features(worker_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(worker_tests PRIVATE /W4 /WX /EHsc)
endif()
target_link_libraries(worker_tests PRIVATE Threads::Threads)
add_test(NAME worker COMMAND worker_tests)

add_test(NAME decoder-pcm-reference COMMAND Python3::Interpreter
  "${CSE_ROOT}/tools/qualification/verify_pcm.py"
  --encoded "${CSE_MPG123_SOURCE}/src/tests/sweep.mp3" --pcm "${CMAKE_BINARY_DIR}/decoded-reference.pcm")
set_tests_properties(decoder-pcm-reference PROPERTIES FIXTURES_REQUIRED cse_decoded_pcm)
