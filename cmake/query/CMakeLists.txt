# selections. External providers are symbolic; this project is never built.

project(CseBuildInventory LANGUAGES C CXX)
set(CSE_BUILD_INVENTORY TRUE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CSE_SDK_DIR "${CSE_ROOT}/src")
set(CSE_TARGET_OS "${CSE_QUERY_OS}")
set(CSE_TARGET_ARCH "${CSE_QUERY_ARCH}")
set(CSE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime/$<CONFIG>")
set(CSE_BROWSER_HOST_RELATIVE_PATH "cse_cef_host")
set(WIN32 FALSE)
set(APPLE FALSE)
set(MSVC FALSE)
if(CSE_TARGET_OS STREQUAL windows)
  set(WIN32 TRUE)
  set(MSVC TRUE)
  string(TOUPPER "${CSE_TARGET_ARCH}" CSE_MSVC_MACHINE)
elseif(CSE_TARGET_OS STREQUAL macos)
  set(APPLE TRUE)
endif()
if(CSE_TARGET_ARCH STREQUAL x86)
  set(CMAKE_SIZEOF_VOID_P 4)
else()
  set(CMAKE_SIZEOF_VOID_P 8)

include("${CSE_ROOT}/cmake/RuntimeTargets.cmake")

# Bypass provider discovery and binary validation only. Keep the actual target
# definitions, their per-source properties and generator expressions unchanged.
function(cse_link_import target configuration path)
  target_link_libraries(${target} PRIVATE "$<$<CONFIG:${configuration}>:${path}>")
endfunction()
function(cse_link_sdl3 target configuration)
  target_link_libraries(${target} PRIVATE "$<$<CONFIG:${configuration}>:SDL3::SDL3-shared>")
endfunction()
function(cse_link_mpg123 target configuration)
  target_link_libraries(${target} PRIVATE "$<$<CONFIG:${configuration}>:libmpg123>")
endfunction()
function(cse_link_webp target)
  get_target_property(runtime ${target} MSVC_RUNTIME_LIBRARY)
  set(provider cse_webp_md)
  if(MSVC AND runtime AND NOT runtime MATCHES "DLL")
    set(provider cse_webp_mt)
  endif()
  target_link_libraries(${target} PRIVATE ${provider})
  target_include_directories(${target} PRIVATE "${CSE_SDK_DIR}/common")
endfunction()
add_library(cse_webp_md INTERFACE IMPORTED GLOBAL)
add_library(cse_webp_mt INTERFACE IMPORTED GLOBAL)
function(cse_verify_steam_provider path target_os architecture)
  set(CSE_STEAM_RUNTIME_NAME symbolic-steam PARENT_SCOPE)
endfunction()
set(CSE_STEAM_PROVIDER_FILE "${CMAKE_BINARY_DIR}/symbolic-steam")
add_library(SDL3::SDL3-shared INTERFACE IMPORTED GLOBAL)
add_library(libmpg123 INTERFACE IMPORTED GLOBAL)
add_library(Freetype::Freetype INTERFACE IMPORTED GLOBAL)
function(cse_target_dependency_include target configuration reference)
  target_include_directories(${target} PRIVATE "$<$<CONFIG:${configuration}>:${reference}>")
endfunction()
function(cse_source_dependency_include source configuration reference)
  set_property(SOURCE "${source}" APPEND PROPERTY INCLUDE_DIRECTORIES "$<$<CONFIG:${configuration}>:${reference}>")
endfunction()
function(cse_link_native_font_services target)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${CSE_SDK_DIR}/vgui2/vgui_surfacelib/native")
  target_link_libraries(${target} PRIVATE cse_mac_font_registration Freetype::Freetype
    "-framework CoreText" "-framework CoreFoundation")
  target_link_options(${target} PRIVATE "-Wl,-rpath,@loader_path")
endfunction()
function(cse_runtime_assets target relative)
  foreach(source IN LISTS ARGN)
    set_property(TARGET ${target} APPEND PROPERTY CSE_QUERY_RESOURCE_SOURCES "${source}")
    get_filename_component(name "${source}" NAME)
    set_property(TARGET ${target} APPEND PROPERTY CSE_QUERY_RESOURCE_DESTINATIONS "${relative}/${name}")
  endforeach()
endfunction()

function(cse_query_quote value output)
  string(REPLACE "\\" "\\\\" value "${value}")
  string(REPLACE "\"" "\\\"" value "${value}")
  string(REPLACE "\n" "\\n" value "${value}")
  string(REPLACE "\r" "\\r" value "${value}")
  string(REPLACE "\t" "\\t" value "${value}")
  set(${output} "\"${value}\"" PARENT_SCOPE)
endfunction()

file(GLOB children LIST_DIRECTORIES TRUE "${CSE_ROOT}/cmake/targets/*")
foreach(directory IN LISTS children)
  if(NOT EXISTS "${directory}/CMakeLists.txt")
    continue()
  endif()
  get_filename_component(target "${directory}" NAME)
  if(target STREQUAL "csextended")
    # The isolated AMXX package is not a game-relative runtime target.
    continue()
  endif()
  add_subdirectory("${directory}" "${CMAKE_BINARY_DIR}/targets/${target}")
  # A derived native export file may be absent precisely because the caller is
  # regenerating it. Original x86 definition inputs still must exist.
  get_target_property(native_definition ${target} CSE_NATIVE_EXPORT_DEFINITION)
  if(native_definition)
    set_source_files_properties("${native_definition}" DIRECTORY "${directory}" PROPERTIES GENERATED TRUE)
  endif()
  # A host without the Windows resource compiler can still inspect the selected
  # resource inputs. Objective-C++ inputs are inspected as host C++ recipes.
  get_target_property(sources ${target} SOURCES)
  set(resources "[]")
  set(resource_count 0)
  foreach(source IN LISTS sources)
    if(source MATCHES "\\.rc$")
      # The File API has no compile group for a host-disabled RC input. Capture
      # its configured preprocessor/include properties in the owning directory.
      set(resource "{}")
      cse_query_quote("${source}" quoted)
      string(JSON resource SET "${resource}" path "${quoted}")
      foreach(key INCLUDE_DIRECTORIES COMPILE_DEFINITIONS)
        get_target_property(target_value ${target} ${key})
        get_source_file_property(source_value "${source}" DIRECTORY "${directory}" ${key})
        if(target_value STREQUAL "target_value-NOTFOUND")
          set(target_value "")
        endif()
        if(source_value STREQUAL "NOTFOUND")
          set(source_value "")
        endif()
        cse_query_quote("${target_value};${source_value}" quoted)
        string(JSON resource SET "${resource}" "${key}" "${quoted}")
      endforeach()
      string(JSON resources SET "${resources}" ${resource_count} "${resource}")
      math(EXPR resource_count "${resource_count} + 1")
      set_source_files_properties("${source}" DIRECTORY "${directory}" PROPERTIES HEADER_FILE_ONLY TRUE)
    elseif(source MATCHES "\\.mm$")
      set_source_files_properties("${source}" DIRECTORY "${directory}" PROPERTIES LANGUAGE CXX)
    endif()
  endforeach()
  set(properties "{}")
  foreach(key OUTPUT_NAME TYPE MSVC_RUNTIME_LIBRARY CXX_STANDARD C_STANDARD CXX_EXTENSIONS C_EXTENSIONS
      POSITION_INDEPENDENT_CODE VISIBILITY_INLINES_HIDDEN CXX_VISIBILITY_PRESET C_VISIBILITY_PRESET
      LINK_LIBRARIES LINK_OPTIONS
      RUNTIME_OUTPUT_DIRECTORY LIBRARY_OUTPUT_DIRECTORY CSE_APPLE_MAIN_IMAGE_PUBLISHER
      CSE_QUERY_RESOURCE_SOURCES CSE_QUERY_RESOURCE_DESTINATIONS)
    get_target_property(value ${target} ${key})
    if(value STREQUAL "value-NOTFOUND")
      set(value "")
    endif()
    cse_query_quote("${value}" quoted)
    string(JSON properties SET "${properties}" "${key}" "${quoted}")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/properties/${target}-$<CONFIG>.json"
    CONTENT "${properties}\n" TARGET ${target})
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/properties/${target}-$<CONFIG>-resources.json"
    CONTENT "${resources}\n" TARGET ${target})
endforeach()
