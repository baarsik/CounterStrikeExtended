cmake_minimum_required(VERSION 3.25)
set(archive "${CSE_SDK_DIR}/external/libwebp-source/libwebp-1.6.0.tar.gz")
file(SHA256 "${archive}" archive_hash)
if(NOT archive_hash STREQUAL "e4ab7009bf0629fd11982d4c2aa83964cf244cffba7347ecd39019a9e38c4564")
  message(FATAL_ERROR "libwebp source archive hash mismatch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${archive}")
set(source_root "${CMAKE_CURRENT_BINARY_DIR}/source")
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${source_root}")
set(upstream_source "${source_root}/libwebp-1.6.0")
set(upstream_binary "${CMAKE_CURRENT_BINARY_DIR}/upstream")
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(BUILD_SHARED_LIBS OFF)
set(WEBP_LINK_STATIC ON)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS WEBP_JS FUZZTEST)
  set(WEBP_BUILD_${option} OFF)
endforeach()
set(WEBP_ENABLE_WUNUSED_RESULT OFF)
find_package(Threads REQUIRED)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_subdirectory("${upstream_source}" "${upstream_binary}" EXCLUDE_FROM_ALL)

# Reuse upstream's configured source inventory and SIMD flags. Build one archive
# per CRT instead of mixing /MT VGUI1 with the /MD engine's static dependencies.
set(codec_sources)
foreach(component webpdecode webpencode webpdsp webputils sharpyuv)
  get_target_property(component_sources ${component} SOURCES)
  foreach(source IN LISTS component_sources)
    if(NOT IS_ABSOLUTE "${source}")
      set(source "${upstream_source}/${source}")
    endif()
    list(APPEND codec_sources "${source}")
    get_source_file_property(flags "${source}" DIRECTORY "${upstream_source}" COMPILE_FLAGS)
    if(flags)
      set_source_files_properties("${source}" PROPERTIES COMPILE_FLAGS "${flags}")
    endif()
  endforeach()
endforeach()
list(REMOVE_DUPLICATES codec_sources)
get_directory_property(codec_definitions DIRECTORY "${upstream_source}" COMPILE_DEFINITIONS)
get_target_property(codec_links webp LINK_LIBRARIES)
get_target_property(sharpyuv_links sharpyuv LINK_LIBRARIES)
list(REMOVE_ITEM codec_links sharpyuv)
if(sharpyuv_links)
  list(APPEND codec_links ${sharpyuv_links})
endif()
set(variants md)
if(MSVC)
  list(APPEND variants mt)
endif()
foreach(variant IN LISTS variants)
  add_library(cse_webp_${variant} STATIC ${codec_sources})
  target_include_directories(cse_webp_${variant} PRIVATE "${upstream_source}" "${upstream_binary}"
    PUBLIC "${upstream_source}/src")
  target_compile_definitions(cse_webp_${variant} PRIVATE ${codec_definitions})
  target_link_libraries(cse_webp_${variant} PRIVATE ${codec_links})
  set_target_properties(cse_webp_${variant} PROPERTIES POSITION_INDEPENDENT_CODE ON
    C_VISIBILITY_PRESET hidden ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>/webp/${variant}")
  if(MSVC)
    if(variant STREQUAL mt)
      set_property(TARGET cse_webp_${variant} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    target_compile_options(cse_webp_${variant} PRIVATE "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
  else()
    target_compile_options(cse_webp_${variant} PRIVATE "$<$<CONFIG:Release>:-O2>" "$<$<CONFIG:Debug>:-O0>")
  endif()
endforeach()
