if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the extracted pinned CEF distribution")
endif()
file(TO_CMAKE_PATH "${CEF_ROOT}" CEF_ROOT)
file(READ "${CSE_ROOT}/tools/qualification/fixtures/cef/lock.json" cse_cef_lock)
if(APPLE AND CSE_TARGET_ARCH STREQUAL "arm64")
  set(cse_cef_platform macosarm64)
  set(PROJECT_ARCH arm64)
elseif(WIN32 AND CSE_TARGET_ARCH STREQUAL "x64")
  set(cse_cef_platform windows64)
  set(PROJECT_ARCH x86_64)
else()
  message(FATAL_ERROR "The browser host requires a pinned native macOS ARM64 or Windows x64 provider")
endif()
set(cse_cef_pinned_files api_header runtime debug_runtime)
if(WIN32)
  list(APPEND cse_cef_pinned_files import_library)
endif()
foreach(item IN LISTS cse_cef_pinned_files)
  if(item STREQUAL "api_header")
    set(cse_cef_path include/cef_api_versions.h)
    string(JSON cse_cef_expected GET "${cse_cef_lock}" platforms ${cse_cef_platform} api_header_sha256)
  else()
    string(JSON cse_cef_path GET "${cse_cef_lock}" platforms ${cse_cef_platform} ${item} path)
    string(JSON cse_cef_expected GET "${cse_cef_lock}" platforms ${cse_cef_platform} ${item} sha256)
  endif()
  file(SHA256 "${CEF_ROOT}/${cse_cef_path}" cse_cef_actual)
  if(NOT cse_cef_actual STREQUAL cse_cef_expected)
    message(FATAL_ERROR "Pinned CEF ${item} does not match: ${cse_cef_path}")
  endif()
endforeach()
set(api_version 15600 CACHE STRING "Pinned stable CEF API" FORCE)
set(USE_SANDBOX ON CACHE BOOL "Keep the CEF process sandbox" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(APPLE)
  enable_language(OBJCXX)
  list(REMOVE_ITEM CEF_COMPILER_FLAGS "-mmacosx-version-min=13.0")
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" "${CMAKE_BINARY_DIR}/libcef_dll_wrapper")
set(CEF_TARGET cse_cef_host)
set(CSE_HOST_SOURCE "${CSE_ROOT}/GoldSrc-SDK/engine/html/native")
SET_CEF_TARGET_OUT_DIR()
# A complete runtime stages the host beside the engine. Standalone fixtures
# retain their own isolated provider directory. CEF flags remain in this subtree.
if(CSE_CEF_STAGE_DIRECTORY)
  if(NOT IS_ABSOLUTE "${CSE_CEF_STAGE_DIRECTORY}")
    message(FATAL_ERROR "CSE_CEF_STAGE_DIRECTORY must be an absolute runtime path")
  endif()
  set(CEF_TARGET_OUT_DIR "${CSE_CEF_STAGE_DIRECTORY}")
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CEF_TARGET_OUT_DIR}")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CEF_TARGET_OUT_DIR}")
endif()
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/imports/$<CONFIG>")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/symbols/$<CONFIG>")
set(CMAKE_COMPILE_PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/symbols/$<CONFIG>")
if(APPLE)
  add_executable(${CEF_TARGET} MACOSX_BUNDLE "${CSE_HOST_SOURCE}/cef_host.cpp" "${CSE_HOST_SOURCE}/cef_host_mac.mm")
  SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
  target_compile_options(${CEF_TARGET} PRIVATE -fobjc-arc)
  target_compile_definitions(${CEF_TARGET} PRIVATE CSE_CEF_DEBUG_PROFILE=$<IF:$<CONFIG:Debug>,1,0>)
  target_link_libraries(${CEF_TARGET} PRIVATE "-framework Security")
  set_target_properties(${CEF_TARGET} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
  set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
  # The vendor copy macro's ln -sf follows existing directory symlinks on relink.
  # Recreate only our staged framework before copying to preserve its signed layout.
  add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E rm -rf "${CEF_APP}/Contents/Frameworks/Chromium Embedded Framework.framework"
    VERBATIM)
  COPY_MAC_FRAMEWORK("${CEF_TARGET}" "${CEF_BINARY_DIR}" "${CEF_APP}")
  foreach(suffixes IN LISTS CEF_HELPER_APP_SUFFIXES)
    string(REPLACE ":" ";" suffixes "${suffixes}")
    list(GET suffixes 0 name_suffix)
    list(GET suffixes 1 target_suffix)
    list(GET suffixes 2 plist_suffix)
    set(helper "cse_cef_host_Helper${target_suffix}")
    set(helper_name "cse_cef_host Helper${name_suffix}")
    configure_file(helper-Info.plist.in "helper${target_suffix}.plist" @ONLY)
    add_executable(${helper} MACOSX_BUNDLE "${CSE_HOST_SOURCE}/cef_host_renderer.cpp" "${CSE_HOST_SOURCE}/cef_host_helper_mac.mm")
    SET_EXECUTABLE_TARGET_PROPERTIES(${helper})
    target_link_libraries(${helper} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    target_compile_features(${helper} PRIVATE cxx_std_20)
    set_target_properties(${helper} PROPERTIES OUTPUT_NAME "${helper_name}"
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/helper${target_suffix}.plist")
    add_custom_command(TARGET ${helper} POST_BUILD
      COMMAND /usr/bin/codesign --force --sign - "${CEF_TARGET_OUT_DIR}/${helper_name}.app" VERBATIM)
    add_dependencies(${CEF_TARGET} ${helper})
    add_custom_command(TARGET ${CEF_TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CEF_TARGET_OUT_DIR}/${helper_name}.app" "${CEF_APP}/Contents/Frameworks/${helper_name}.app" VERBATIM)
  endforeach()
else()
  ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_library(${CEF_TARGET} SHARED "${CSE_HOST_SOURCE}/cef_host.cpp" "${CSE_HOST_SOURCE}/cef_host_win.cpp")
  SET_LIBRARY_TARGET_PROPERTIES(${CEF_TARGET})
  target_link_libraries(${CEF_TARGET} PRIVATE libcef_lib)
  COPY_SINGLE_FILE(${CEF_TARGET} ${CEF_BINARY_DIR}/bootstrap.exe ${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.exe)
  COPY_FILES("${CEF_TARGET}" "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
  COPY_FILES("${CEF_TARGET}" "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
  SET_LPAC_ACLS("${CEF_TARGET}")
endif()
if(MSVC)
  # MSVC places an incremental-link database beside the DLL even when PDB/import
  # paths are isolated. Runtime directories contain only staged runtime files.
  target_link_options(${CEF_TARGET} PRIVATE /INCREMENTAL:NO)
endif()
target_link_libraries(${CEF_TARGET} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
target_compile_features(${CEF_TARGET} PRIVATE cxx_std_20)

target_sources(${CEF_TARGET} PRIVATE "${CSE_HOST_SOURCE}/cef_host_renderer.cpp")

# Retain the provenance of the owned entry points and the separately distributed CEF provider.
if(APPLE)
  set(CSE_HOST_LICENSE_DIR "${CEF_APP}/Contents/Resources")
else()
  set(CSE_HOST_LICENSE_DIR "${CEF_TARGET_OUT_DIR}")
endif()
add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CSE_HOST_LICENSE_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.cef"
    "${CSE_HOST_LICENSE_DIR}/LICENSE.cef-host.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE"
    "${CSE_HOST_LICENSE_DIR}/NOTICE.cef-host.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/LICENSE.txt"
    "${CSE_HOST_LICENSE_DIR}/LICENSE.cef-provider.txt"
  VERBATIM)
if(APPLE)
  # Seal only our host after all owned helpers/resources are staged. Preserve
  # the pinned vendor framework and its signature; never deep-resign it.
  add_custom_command(TARGET ${CEF_TARGET} POST_BUILD
    COMMAND /usr/bin/codesign --force --sign - "${CEF_APP}" VERBATIM)
endif()
