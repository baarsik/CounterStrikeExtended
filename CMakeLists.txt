cmake_minimum_required(VERSION 4.2)
cmake_policy(SET CMP0091 NEW)
set(CSE_ROOT "${CMAKE_CURRENT_LIST_DIR}")
set(CSE_SDK_DIR "${CSE_ROOT}/GoldSrc-SDK")
if(NOT CMAKE_HOST_WIN32)
  message(FATAL_ERROR "CounterStrikeExtended currently requires a Windows MSVC x86 builder. macOS/Linux runtime builds are not supported.")
endif()
set(CMAKE_USER_MAKE_RULES_OVERRIDE "${CSE_ROOT}/cmake/MSVCDefaults.cmake")
project(CounterStrikeExtended VERSION 0.1 LANGUAGES C CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The baseline requires MSVC targeting Windows x86/Win32. x64 and ARM64 ports are not implemented.")
endif()
if(DEFINED MSVC_TOOLSET_VERSION AND NOT MSVC_TOOLSET_VERSION STREQUAL "145")
  message(FATAL_ERROR "The inherited baseline uses MSVC v145; selected toolset is ${MSVC_TOOLSET_VERSION}.")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
elseif(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Debug or Release" FORCE)
elseif(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
  message(FATAL_ERROR "Only Debug and Release have translated MSBuild selections.")
endif()
option(CSE_HARDWARE_ONLY "Build only hw.dll and its source-built VGUI1 dependency" OFF)
option(CSE_BUILD_DEDICATED "Build hlds.exe and swds.dll" ON)
option(CSE_BUILD_BROADCASTING "Build standalone hltv.exe and proxy.dll" OFF)
option(CSE_BUILD_A3D "Build legacy a3dapi.dll" OFF)
include("${CSE_ROOT}/cmake/RuntimeTargets.cmake")
include("${CSE_ROOT}/cmake/SDL3Dependency.cmake")
include("${CSE_ROOT}/cmake/Mpg123Dependency.cmake")
foreach(config Debug Release)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/symbols/${config}" "${CMAKE_BINARY_DIR}/lib/${config}")
endforeach()
# VGUI1 must exist before consumers reference its compatibility import library.
set(runtime_targets vgui hw_engine)
if(NOT CSE_HARDWARE_ONLY)
  list(APPEND runtime_targets tier0 vgui2 filesystem_stdio launcher cs_client cs_server
    hl_client hl_server gameui particleman hltv_core demoplayer)
  if(CSE_BUILD_DEDICATED)
    list(APPEND runtime_targets hlds swds)
  endif()
  if(CSE_BUILD_BROADCASTING)
    list(APPEND runtime_targets hltv hltv_proxy)
  endif()
  if(CSE_BUILD_A3D)
    list(APPEND runtime_targets a3dapi)
  endif()
endif()
foreach(target IN LISTS runtime_targets)
  add_subdirectory("${CSE_ROOT}/cmake/targets/${target}" "${CMAKE_BINARY_DIR}/targets/${target}")
endforeach()
add_custom_target(cse_hardware DEPENDS hw_engine)
add_custom_target(cse_runtime DEPENDS ${runtime_targets})
message(STATUS "Runtime outputs: ${CMAKE_BINARY_DIR}/runtime/<Configuration>")
message(STATUS "No tests or runtime launches are part of this build.")
